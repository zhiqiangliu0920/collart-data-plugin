-- Collart Android 过去3天 DAU 变化分析
-- 生成时间: 2026-04-20
-- 数据范围: 过去3天 (2026-04-17 至 2026-04-19)

WITH daily_dau AS (
  SELECT
    event_date,
    COUNT(DISTINCT user_pseudo_id) AS total_dau,
    COUNT(DISTINCT CASE WHEN is_new = 'New' THEN user_pseudo_id END) AS new_user_dau,
    COUNT(DISTINCT CASE WHEN is_new = 'Old' THEN user_pseudo_id END) AS old_user_dau
  FROM `aidata2025.dwd.dwd_oper_user_collart_di`
  WHERE event_date BETWEEN DATE_SUB(CURRENT_DATE(), INTERVAL 3 DAY) AND DATE_SUB(CURRENT_DATE(), INTERVAL 1 DAY)
    AND package_name = 'free.ai.photo.generator.collart.ai'
  GROUP BY event_date
),
dau_with_change AS (
  SELECT
    event_date,
    total_dau,
    new_user_dau,
    old_user_dau,
    ROUND(new_user_dau / total_dau * 100, 2) AS new_user_pct,
    ROUND(old_user_dau / total_dau * 100, 2) AS old_user_pct,
    -- 环比变化（与前一天比较）
    total_dau - LAG(total_dau) OVER (ORDER BY event_date) AS total_dau_change,
    ROUND((total_dau - LAG(total_dau) OVER (ORDER BY event_date)) / LAG(total_dau) OVER (ORDER BY event_date) * 100, 2) AS total_dau_change_pct
  FROM daily_dau
)
SELECT
  FORMAT_DATE('%Y-%m-%d (%a)', event_date) AS date_label,
  total_dau,
  new_user_dau,
  old_user_dau,
  new_user_pct || '%' AS new_user_ratio,
  old_user_pct || '%' AS old_user_ratio,
  COALESCE(CAST(total_dau_change AS STRING), '-') AS dau_change,
  COALESCE(CAST(total_dau_change_pct AS STRING) || '%', '-') AS dau_change_pct
FROM dau_with_change
ORDER BY event_date DESC;

-- ============================================
-- 简化版查询（快速查看）
-- ============================================
SELECT
  FORMAT_DATE('%Y-%m-%d (%a)', event_date) AS date,
  COUNT(DISTINCT user_pseudo_id) AS DAU,
  COUNT(DISTINCT CASE WHEN is_new = 'New' THEN user_pseudo_id END) AS 新用户,
  COUNT(DISTINCT CASE WHEN is_new = 'Old' THEN user_pseudo_id END) AS 老用户
FROM `aidata2025.dwd.dwd_oper_user_collart_di`
WHERE event_date >= DATE_SUB(CURRENT_DATE(), INTERVAL 7 DAY)
  AND package_name = 'free.ai.photo.generator.collart.ai'
GROUP BY event_date
ORDER BY event_date DESC
LIMIT 10;
