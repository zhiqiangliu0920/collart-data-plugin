-- 验证口径：分别查看 DAU 和 template_click 的原始数据
-- 时间范围：2026-03-24 至 2026-03-30

-- 1. 检查 DAU 表的数据分布
SELECT
  date,
  COUNT(DISTINCT user_id) as user_count,
  COUNT(*) as total_rows
FROM `aidata2025.story_web_android.dau_active_table`
WHERE _TABLE_SUFFIX BETWEEN '20260324' AND '20260330'
  AND platform = 'Web'
GROUP BY date
ORDER BY date;

-- 2. 检查 template_click 事件的数据分布
SELECT
  event_date,
  COUNT(DISTINCT user_pseudo_id) as unique_users,
  COUNT(*) as total_clicks,
  COUNT(DISTINCT device.web_info.hostname) as hostnames
FROM `aidata2025.story_web_android.events_*`
WHERE _TABLE_SUFFIX BETWEEN '20260324' AND '20260330'
  AND event_name = 'template_click'
  AND app_info.id IS NULL
GROUP BY event_date
ORDER BY event_date;
