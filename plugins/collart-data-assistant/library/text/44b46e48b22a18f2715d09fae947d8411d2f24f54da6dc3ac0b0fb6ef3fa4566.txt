-- 从 events_* 提取单个 event_params（标量子查询，避免全量 UNNEST 膨胀）
-- 示例：Android ai_service_start 的 service_type / function_type 分布
SELECT
  PARSE_DATE('%Y%m%d', event_date) AS event_date,
  (
    SELECT ep.value.string_value FROM UNNEST(event_params) ep
    WHERE ep.key = 'service_type' LIMIT 1
  ) AS service_type,
  (
    SELECT ep.value.string_value FROM UNNEST(event_params) ep
    WHERE ep.key = 'function_type' LIMIT 1
  ) AS function_type,
  COUNT(*) AS pv,
  COUNT(DISTINCT user_pseudo_id) AS uv
FROM `storytemplate-10a27.analytics_232977577.events_*`
WHERE _TABLE_SUFFIX BETWEEN '20260801' AND '20260807'
  AND app_info.id = 'free.ai.photo.generator.collart.ai'
  AND event_name = 'ai_service_start'
GROUP BY ALL
ORDER BY pv DESC;
