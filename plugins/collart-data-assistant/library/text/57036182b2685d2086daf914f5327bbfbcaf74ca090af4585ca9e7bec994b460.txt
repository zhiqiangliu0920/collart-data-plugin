# -*- coding: utf-8 -*-
"""
Vibedance 新产品核心指标探索分析
项目: vibedance-540a0
表: analytics_533044528.events_*
"""
import os
os.environ['PYTHONIOENCODING'] = 'utf-8'

from google.cloud import bigquery
from google.oauth2 import service_account

# 尝试用 aidata2025 的服务账号访问 vibedance 项目
KEY_PATH = "knowledge/service-accounts/aidata2025.json"
credentials = service_account.Credentials.from_service_account_file(KEY_PATH)
client = bigquery.Client(project="vibedance-540a0", credentials=credentials)

print("=" * 70)
print("Vibedance 产品数据探索")
print("=" * 70)

# ============ 1. 查看可用的表 ============
print("\n[1] 查看数据集可用表...")
try:
    tables = list(client.list_tables("vibedance-540a0.analytics_533044528"))
    print("找到 {} 个表:".format(len(tables)))
    for t in tables[:20]:
        print("  - {}".format(t.table_id))
except Exception as e:
    print("访问失败: {}".format(e))
    print("服务账号可能没有 vibedance-540a0 项目权限")

# ============ 2. 查看 events 表最近数据日期范围 ============
print("\n[2] 查看 events 表数据时间范围...")
try:
    query_range = """
    SELECT
      MIN(_TABLE_SUFFIX) AS earliest_date,
      MAX(_TABLE_SUFFIX) AS latest_date,
      COUNT(DISTINCT _TABLE_SUFFIX) AS total_days
    FROM `vibedance-540a0.analytics_533044528.events_*`
    WHERE _TABLE_SUFFIX >= '20260101'
    """
    rows = list(client.query(query_range).result())
    for row in rows:
        print("最早日期: {}, 最新日期: {}, 总天数: {}".format(
            row.earliest_date, row.latest_date, row.total_days))
except Exception as e:
    print("查询失败: {}".format(e))

# ============ 3. 查看有哪些事件 ============
print("\n[3] 查看主要事件分布...")
try:
    query_events = """
    SELECT
      event_name,
      COUNT(*) AS event_count,
      COUNT(DISTINCT user_pseudo_id) AS user_count
    FROM `vibedance-540a0.analytics_533044528.events_*`
    WHERE _TABLE_SUFFIX >= '20260101'
    GROUP BY event_name
    ORDER BY event_count DESC
    LIMIT 30
    """
    rows = list(client.query(query_events).result())
    print("{:<40} {:>12} {:>12}".format("事件名", "事件次数", "用户去重"))
    print("-" * 70)
    for row in rows:
        print("{:<40} {:>12,} {:>12,}".format(row.event_name, row.event_count, row.user_count))
except Exception as e:
    print("查询失败: {}".format(e))

# ============ 4. 核心指标概览 ============
print("\n[4] 按日核心指标趋势 (最近30天)...")
try:
    query_daily = """
    SELECT
      event_date,
      COUNT(DISTINCT CASE WHEN event_name = 'session_start' THEN user_pseudo_id END) AS dau,
      COUNT(DISTINCT user_pseudo_id) AS active_users,
      COUNT(DISTINCT CASE WHEN event_name = 'first_open' THEN user_pseudo_id END) AS new_users,
      COUNT(*) AS total_events,
      COUNT(DISTINCT device.mobile_os_hardware_model) AS device_models,
      COUNT(DISTINCT geo.country) AS countries
    FROM `vibedance-540a0.analytics_533044528.events_*`
    WHERE _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 30 DAY))
      AND _TABLE_SUFFIX <= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 1 DAY))
    GROUP BY event_date
    ORDER BY event_date
    """
    rows = list(client.query(query_daily).result())
    print("{:<12} {:>8} {:>8} {:>8} {:>10} {:>8} {:>8}".format(
        "日期", "DAU", "活跃用户", "新用户", "总事件", "设备数", "国家数"))
    print("-" * 70)
    for row in rows:
        print("{:<12} {:>8,} {:>8,} {:>8,} {:>10,} {:>8,} {:>8,}".format(
            row.event_date, row.dau, row.active_users, row.new_users, 
            row.total_events, row.device_models, row.countries))
except Exception as e:
    print("查询失败: {}".format(e))

# ============ 5. 国家分布 TOP15 ============
print("\n[5] 国家分布 TOP15 (全部时间)...")
try:
    query_geo = """
    SELECT
      geo.country,
      COUNT(DISTINCT user_pseudo_id) AS users,
      COUNT(*) AS events
    FROM `vibedance-540a0.analytics_533044528.events_*`
    WHERE _TABLE_SUFFIX >= '20260101'
    GROUP BY geo.country
    ORDER BY users DESC
    LIMIT 15
    """
    rows = list(client.query(query_geo).result())
    print("{:<25} {:>10} {:>12}".format("国家", "用户数", "事件数"))
    print("-" * 50)
    for row in rows:
        print("{:<25} {:>10,} {:>12,}".format(row.country, row.users, row.events))
except Exception as e:
    print("查询失败: {}".format(e))

# ============ 6. 平台/设备分布 ============
print("\n[6] 平台与设备分布...")
try:
    query_device = """
    SELECT
      device.operating_system AS os,
      device.mobile_brand_name AS brand,
      COUNT(DISTINCT user_pseudo_id) AS users,
      COUNT(*) AS events
    FROM `vibedance-540a0.analytics_533044528.events_*`
    WHERE _TABLE_SUFFIX >= '20260101'
    GROUP BY device.operating_system, device.mobile_brand_name
    ORDER BY users DESC
    LIMIT 15
    """
    rows = list(client.query(query_device).result())
    print("{:<12} {:<20} {:>10} {:>12}".format("系统", "品牌", "用户数", "事件数"))
    print("-" * 60)
    for row in rows:
        print("{:<12} {:<20} {:>10,} {:>12,}".format(
            row.os or 'N/A', row.brand or 'N/A', row.users, row.events))
except Exception as e:
    print("查询失败: {}".format(e))

# ============ 7. 留存分析 (如果有足够数据) ============
print("\n[7] D1 留存率 (最近14天新用户)...")
try:
    query_retention = """
    WITH new_users AS (
      SELECT DISTINCT
        event_date AS install_date,
        user_pseudo_id
      FROM `vibedance-540a0.analytics_533044528.events_*`
      WHERE event_date >= DATE_SUB(CURRENT_DATE(), INTERVAL 14 DAY)
        AND event_date < DATE_SUB(CURRENT_DATE(), INTERVAL 1 DAY)
        AND event_name = 'first_open'
    ),
    d1_active AS (
      SELECT DISTINCT
        DATE_SUB(event_date, INTERVAL 1 DAY) AS install_date,
        user_pseudo_id
      FROM `vibedance-540a0.analytics_533044528.events_*`
      WHERE event_date >= DATE_SUB(CURRENT_DATE(), INTERVAL 13 DAY)
        AND event_date < CURRENT_DATE()
        AND event_name = 'session_start'
    )
    SELECT
      n.install_date,
      COUNT(DISTINCT n.user_pseudo_id) AS new_users,
      COUNT(DISTINCT d1.user_pseudo_id) AS d1_returned,
      ROUND(COUNT(DISTINCT d1.user_pseudo_id) * 100.0 / COUNT(DISTINCT n.user_pseudo_id), 1) AS d1_retention_pct
    FROM new_users n
    LEFT JOIN d1_active d1 
      ON n.install_date = d1.install_date AND n.user_pseudo_id = d1.user_pseudo_id
    GROUP BY n.install_date
    ORDER BY n.install_date
    """
    rows = list(client.query(query_retention).result())
    if rows:
        print("{:<12} {:>10} {:>10} {:>10}".format("日期", "新用户", "D1回访", "D1留存%"))
        print("-" * 45)
        for row in rows:
            print("{:<12} {:>10,} {:>10,} {:>10.1f}".format(
                str(row.install_date), row.new_users, row.d1_returned, row.d1_retention_pct))
    else:
        print("暂无足够数据计算留存")
except Exception as e:
    print("查询失败: {}".format(e))

print("\n=== 探索完成 ===")
