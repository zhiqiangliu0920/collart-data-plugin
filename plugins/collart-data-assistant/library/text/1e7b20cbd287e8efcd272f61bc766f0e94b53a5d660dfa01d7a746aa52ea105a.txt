-- Collart Android TOP 10 点击率模板分析 (2026-04-02)
-- 数据源：storytemplate-10a27.analytics_232977577.events_20260402
-- 统计时间：2026-04-03 11:01
-- 排序：按点击率降序
-- 用途：识别高点击率模板，优化推荐策略

WITH template_events AS (
  SELECT
    user_pseudo_id,
    (SELECT value.string_value FROM UNNEST(event_params) WHERE key = 'id') AS temp_id,
    (SELECT value.string_value FROM UNNEST(event_params) WHERE key = 'name') AS temp_name,
    event_name
  FROM `storytemplate-10a27.analytics_232977577.events_20260402`
  WHERE app_info.id = 'free.ai.photo.generator.collart.ai'
    AND user_pseudo_id IS NOT NULL
    AND event_name IN ('Home_Template_Show', 'Home_Template_Click', 'ai_service_start')
),

template_stats AS (
  SELECT
    temp_id,
    MAX(temp_name) AS temp_name,
    COUNT(DISTINCT CASE WHEN event_name = 'Home_Template_Show' THEN user_pseudo_id END) AS show_uv,
    COUNT(DISTINCT CASE WHEN event_name = 'Home_Template_Click' THEN user_pseudo_id END) AS click_uv,
    COUNT(DISTINCT CASE WHEN event_name = 'ai_service_start' THEN user_pseudo_id END) AS generate_uv
  FROM template_events
  WHERE temp_id IS NOT NULL
  GROUP BY temp_id
  HAVING show_uv >= 10
)

SELECT
  temp_id,
  temp_name,
  show_uv,
  click_uv,
  ROUND(click_uv * 100.0 / NULLIF(show_uv, 0), 1) AS click_rate,
  generate_uv,
  ROUND(generate_uv * 100.0 / NULLIF(click_uv, 0), 1) AS generate_rate
FROM template_stats
ORDER BY click_rate DESC, show_uv DESC
LIMIT 10
