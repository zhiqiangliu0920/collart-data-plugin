-- 把 event_name / event_params / agg 换成规则行；单次建议 ≤30 天
-- Android/iOS: params_info；Web/Fashion: web_params_info + package_name 过滤

-- 2) count_events：事件次数 + UV
SELECT
  e.event_date,
  COUNT(DISTINCT e.user_pseudo_id) AS uv,
  COUNT(*) AS metric_value
FROM `aidata2025.dm.dm_collart_android_user_event_di` AS e
INNER JOIN `aidata2025.ads_collartandroid.ads_oper_user_active_di` AS a
  USING (event_date, user_pseudo_id)
WHERE e.event_date BETWEEN from_d AND to_d
  AND a.is_active
  AND e.event_name = 'ai_tool_service_execute_start'
  -- AND LOWER(e.params_info.service_type) LIKE '%video%'
GROUP BY e.event_date
ORDER BY e.event_date;
