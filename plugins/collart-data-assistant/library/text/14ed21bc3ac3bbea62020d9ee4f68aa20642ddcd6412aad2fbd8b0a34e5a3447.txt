import json, collections
from pathlib import Path
R=Path(__file__).resolve().parent
jobs=json.loads((R/'jobs.json').read_text(encoding='utf-8'))['rows']
stats=collections.defaultdict(lambda:{'parent_runs':0,'leaf_jobs':0,'bytes':0,'slot_ms':0,'errors':[],'runs':[]})
for j in jobs:
    labels={x['key']:x['value'] for x in j['labels']}
    action=labels.get('dataform_workflow_execution_action_id_name','')
    if not action: continue
    s=stats[action]
    if j['parent_job_id'] is None:
        s['parent_runs']+=1
        s['runs'].append({'job':j['job_id'],'start':j['start_time'],'end':j['end_time'],'invocation':labels.get('dataform_workflow_execution_id')})
    if j['statement_type']!='SCRIPT':
        s['leaf_jobs']+=1; s['bytes']+=j['total_bytes_processed'] or 0; s['slot_ms']+=j['total_slot_ms'] or 0
    if j['error_result']: s['errors'].append({'job':j['job_id'],'time':j['creation_time'],'error':j['error_result']})
(R/'job_summary.json').write_text(json.dumps(stats,ensure_ascii=False,indent=2),encoding='utf-8')
for k,v in sorted(stats.items(),key=lambda x:x[1]['bytes'],reverse=True):
    print(json.dumps({'action':k,**{kk:vv for kk,vv in v.items() if kk!='runs'}},ensure_ascii=False))
