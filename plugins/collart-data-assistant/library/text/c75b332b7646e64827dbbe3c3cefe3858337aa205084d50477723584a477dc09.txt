-- Web页面转化分析
WITH home_visitors AS (
  SELECT DISTINCT user_pseudo_id
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX >= '20260301' AND _TABLE_SUFFIX <= '20260331'
    AND app_info.id IS NULL
    AND event_name = 'page_view'
),
video_visitors AS (
  SELECT DISTINCT user_pseudo_id
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX >= '20260301' AND _TABLE_SUFFIX <= '20260331'
    AND app_info.id IS NULL
    AND event_name = 'aivideo_edit_show'
)
SELECT 
  COUNT(DISTINCT h.user_pseudo_id) AS home_uv,
  COUNT(DISTINCT v.user_pseudo_id) AS video_uv,
  ROUND(COUNT(DISTINCT v.user_pseudo_id) * 100.0 / COUNT(DISTINCT h.user_pseudo_id), 2) AS conversion_rate
FROM home_visitors h
LEFT JOIN video_visitors v ON h.user_pseudo_id = v.user_pseudo_id;
