-- Collart Web: 周末 fashion_video_generate_click 用户新老判定
-- 查询日期: 2026-06-29
-- 行为口径: fashion_video_generate_click = 使用穿搭带货功能
-- 周末窗口: 2026-06-27 ~ 2026-06-28（北京时间）
-- 新用户口径: aidata2025.ads_collart.ads_oper_user_new_collartweb_di 命中生成当天 user_pseudo_id
-- 内部用户: 按 collart_web/sql_presets/internal_user_filter.sql 排除可识别 user_id

DECLARE start_date DATE DEFAULT DATE '2026-06-27';
DECLARE end_date DATE DEFAULT DATE '2026-06-28';
DECLARE campaign_name STRING DEFAULT '20260624_sea_collartfashionai_sales_webregistration';

WITH internal_users AS (
  SELECT user_id
  FROM UNNEST([
    '{INTERNAL_USER_ID_15}',
    '{INTERNAL_USER_ID_14}',
    '{INTERNAL_USER_ID_2}',
    '{INTERNAL_USER_ID_11}',
    '{INTERNAL_USER_ID_7}',
    '{INTERNAL_USER_ID_4}',
    '{INTERNAL_USER_ID_18}',
    '{INTERNAL_USER_ID_1}',
    '{INTERNAL_USER_ID_8}',
    '{INTERNAL_USER_ID_9}',
    '{INTERNAL_USER_ID_3}',
    '{INTERNAL_USER_ID_17}',
    '{INTERNAL_USER_ID_13}',
    '{INTERNAL_USER_ID_6}',
    '{INTERNAL_USER_ID_5}',
    '{INTERNAL_USER_ID_16}',
    '{INTERNAL_USER_ID_12}',
    '{INTERNAL_USER_ID_10}'
  ]) AS user_id
),

event_base AS (
  SELECT
    PARSE_DATE('%Y%m%d', event_date) AS event_dt,
    DATE(DATETIME(TIMESTAMP_MICROS(event_timestamp), 'Asia/Shanghai')) AS event_date_cn,
    DATETIME(TIMESTAMP_MICROS(event_timestamp), 'Asia/Shanghai') AS event_time_cn,
    event_timestamp,
    user_pseudo_id,
    user_id,
    event_name,
    geo.country AS country,
    device.category AS device_category,
    traffic_source.source AS source,
    traffic_source.medium AS medium,
    traffic_source.name AS campaign,
    (
      SELECT ep.value.string_value
      FROM UNNEST(event_params) ep
      WHERE ep.key = 'page_location'
    ) AS page_location
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN FORMAT_DATE('%Y%m%d', start_date)
                          AND FORMAT_DATE('%Y%m%d', DATE_ADD(end_date, INTERVAL 1 DAY))
    AND app_info.id IS NULL
),

click_events AS (
  SELECT *
  FROM event_base
  WHERE event_name = 'fashion_video_generate_click'
    AND event_date_cn BETWEEN start_date AND end_date
),

identity_map AS (
  SELECT
    user_pseudo_id,
    ARRAY_AGG(user_id IGNORE NULLS ORDER BY event_timestamp DESC LIMIT 1)[SAFE_OFFSET(0)] AS resolved_user_id
  FROM event_base
  WHERE user_id IS NOT NULL
    AND user_id != ''
  GROUP BY user_pseudo_id
),

new_user AS (
  SELECT DISTINCT
    event_date,
    user_pseudo_id
  FROM `aidata2025.ads_collart.ads_oper_user_new_collartweb_di`
  WHERE event_date BETWEEN start_date AND end_date
),

user_rollup AS (
  SELECT
    c.user_pseudo_id,
    ARRAY_AGG(c.event_date_cn ORDER BY c.event_timestamp LIMIT 1)[SAFE_OFFSET(0)] AS first_click_date_cn,
    ARRAY_AGG(c.event_time_cn ORDER BY c.event_timestamp LIMIT 1)[SAFE_OFFSET(0)] AS first_click_time_cn,
    ARRAY_AGG(c.event_time_cn ORDER BY c.event_timestamp DESC LIMIT 1)[SAFE_OFFSET(0)] AS last_click_time_cn,
    ARRAY_AGG(c.country IGNORE NULLS ORDER BY c.event_timestamp LIMIT 1)[SAFE_OFFSET(0)] AS country,
    ARRAY_AGG(c.device_category IGNORE NULLS ORDER BY c.event_timestamp LIMIT 1)[SAFE_OFFSET(0)] AS device_category,
    ARRAY_AGG(c.source IGNORE NULLS ORDER BY c.event_timestamp LIMIT 1)[SAFE_OFFSET(0)] AS source,
    ARRAY_AGG(c.medium IGNORE NULLS ORDER BY c.event_timestamp LIMIT 1)[SAFE_OFFSET(0)] AS medium,
    ARRAY_AGG(c.campaign IGNORE NULLS ORDER BY c.event_timestamp LIMIT 1)[SAFE_OFFSET(0)] AS campaign,
    ARRAY_AGG(c.page_location IGNORE NULLS ORDER BY c.event_timestamp LIMIT 1)[SAFE_OFFSET(0)] AS first_page_location,
    COUNT(*) AS generate_click_cnt
  FROM click_events c
  GROUP BY c.user_pseudo_id
),

classified AS (
  SELECT
    u.*,
    m.resolved_user_id AS user_id,
    IF(nu.user_pseudo_id IS NOT NULL, 'new', 'old') AS new_old_type,
    LOWER(COALESCE(u.campaign, '')) = campaign_name
      OR LOWER(COALESCE(u.first_page_location, '')) LIKE CONCAT('%utm_campaign=', campaign_name, '%') AS is_sea_campaign
  FROM user_rollup u
  LEFT JOIN identity_map m
    USING (user_pseudo_id)
  LEFT JOIN new_user nu
    ON u.first_click_date_cn = nu.event_date
   AND u.user_pseudo_id = nu.user_pseudo_id
  WHERE m.resolved_user_id IS NULL
     OR m.resolved_user_id NOT IN (SELECT user_id FROM internal_users)
)

SELECT
  user_pseudo_id,
  user_id,
  first_click_date_cn,
  first_click_time_cn,
  last_click_time_cn,
  new_old_type,
  is_sea_campaign,
  country,
  device_category,
  source,
  medium,
  campaign,
  generate_click_cnt,
  first_page_location
FROM classified
ORDER BY is_sea_campaign DESC, first_click_time_cn;
