-- Collart Web: 近一周端内 FB 投放来源用户 ID
-- 查询日期: 2026-07-10
-- 时间窗口: 2026-07-03 ~ 2026-07-09（最近 7 个完整自然日，北京时间）
-- 表: aidata2025.dwd.dwd_oper_user_collartweb_di
-- 口径: traffic_src_type = 'delivery' 且 traffic_src_source / traffic_src_name 命中 facebook 或 fb

DECLARE start_date DATE DEFAULT DATE '2026-07-03';
DECLARE end_date DATE DEFAULT DATE '2026-07-09';

WITH internal_users AS (
  SELECT user_id
  FROM UNNEST([
    '{INTERNAL_USER_ID_15}',
    '{INTERNAL_USER_ID_14}',
    '{INTERNAL_USER_ID_2}',
    '{INTERNAL_USER_ID_11}',
    '{INTERNAL_USER_ID_7}',
    '{INTERNAL_USER_ID_4}',
    '{INTERNAL_USER_ID_18}',
    '{INTERNAL_USER_ID_1}',
    '{INTERNAL_USER_ID_8}',
    '{INTERNAL_USER_ID_9}',
    '{INTERNAL_USER_ID_3}',
    '{INTERNAL_USER_ID_17}',
    '{INTERNAL_USER_ID_13}',
    '{INTERNAL_USER_ID_6}',
    '{INTERNAL_USER_ID_5}',
    '{INTERNAL_USER_ID_16}',
    '{INTERNAL_USER_ID_12}',
    '{INTERNAL_USER_ID_10}'
  ]) AS user_id
),

base AS (
  SELECT
    event_date,
    user_pseudo_id,
    user_id,
    country,
    device_category,
    operating_system,
    browser,
    traffic_src_source,
    traffic_src_name,
    traffic_src_type,
    is_new,
    is_login,
    is_vip
  FROM `aidata2025.dwd.dwd_oper_user_collartweb_di`
  WHERE event_date BETWEEN start_date AND end_date
    AND LOWER(COALESCE(traffic_src_type, '')) = 'delivery'
    AND (
      REGEXP_CONTAINS(LOWER(COALESCE(traffic_src_source, '')), r'facebook|(^|[^a-z])fb([^a-z]|$)')
      OR REGEXP_CONTAINS(LOWER(COALESCE(traffic_src_name, '')), r'facebook|(^|[^a-z])fb([^a-z]|$)')
    )
    AND user_id IS NOT NULL
    AND user_id != ''
    AND user_id NOT IN (SELECT user_id FROM internal_users)
)

SELECT
  user_id,
  ARRAY_AGG(DISTINCT user_pseudo_id IGNORE NULLS LIMIT 20) AS user_pseudo_ids,
  MIN(event_date) AS first_seen_date_in_window,
  MAX(event_date) AS last_seen_date_in_window,
  COUNT(DISTINCT event_date) AS active_days_in_window,
  ARRAY_AGG(DISTINCT country IGNORE NULLS LIMIT 10) AS countries,
  ARRAY_AGG(DISTINCT device_category IGNORE NULLS LIMIT 10) AS device_categories,
  ARRAY_AGG(DISTINCT traffic_src_source IGNORE NULLS LIMIT 10) AS traffic_sources,
  ARRAY_AGG(DISTINCT traffic_src_name IGNORE NULLS LIMIT 20) AS traffic_campaigns,
  LOGICAL_OR(is_new) AS is_new_in_window,
  LOGICAL_OR(is_login) AS is_login_in_window,
  LOGICAL_OR(is_vip) AS is_vip_in_window
FROM base
GROUP BY user_id
ORDER BY first_seen_date_in_window, user_id;
