-- Collart Android 昨日 DAU 查询
-- 数据来源: aidata2025.ads_collart.ads_oper_basic_indicator_collart_di
-- 更新频率: 每日

SELECT 
  event_date,
  SUM(dau) as dau
FROM `aidata2025.ads_collart.ads_oper_basic_indicator_collart_di`
WHERE event_date = DATE('2026-06-04')
GROUP BY event_date
;

-- 查最近7天趋势
SELECT 
  event_date,
  SUM(dau) as dau
FROM `aidata2025.ads_collart.ads_oper_basic_indicator_collart_di`
WHERE event_date >= DATE_SUB(CURRENT_DATE(), INTERVAL 8 DAY)
GROUP BY event_date
ORDER BY event_date DESC
;
