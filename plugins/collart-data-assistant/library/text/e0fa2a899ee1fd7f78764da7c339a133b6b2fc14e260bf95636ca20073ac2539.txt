-- Collart Web 顶部 Banner 点击率分析
-- 时间范围：过去3天（2026-03-23 至 2026-03-25）
-- 分母：DAU (session_start 去重用户)
-- 
-- Banner 说明：截图中的顶部 Banner 包含 "AI Video"、"AI Image"、"AI Tools" 三个入口
-- 对应事件：explore_card_click，参数 type 标识具体类型

WITH dau_base AS (
  -- DAU 计算：每日活跃用户数（分母）
  SELECT 
    PARSE_DATE('%Y%m%d', event_date) AS dt,
    COUNT(DISTINCT user_pseudo_id) AS dau
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN '20260323' AND '20260325'
    AND app_info.id IS NULL
    AND event_name = 'session_start'
  GROUP BY dt
),

banner_click AS (
  -- Banner 点击数（explore_card_click - 顶部功能卡片点击）
  SELECT 
    PARSE_DATE('%Y%m%d', event_date) AS dt,
    (SELECT ep.value.string_value FROM UNNEST(event_params) ep WHERE ep.key = 'type') AS banner_type,
    COUNT(*) AS click_cnt,
    COUNT(DISTINCT user_pseudo_id) AS click_uv
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN '20260323' AND '20260325'
    AND app_info.id IS NULL
    AND event_name = 'explore_card_click'
  GROUP BY dt, banner_type
)

-- Banner 点击率结果
SELECT 
  b.dt AS date,
  b.banner_type,
  b.click_cnt,
  b.click_uv,
  d.dau,
  ROUND(b.click_cnt * 100.0 / d.dau, 2) AS ctr_by_click,
  ROUND(b.click_uv * 100.0 / d.dau, 2) AS ctr_by_uv
FROM banner_click b
JOIN dau_base d ON b.dt = d.dt
ORDER BY b.dt DESC, b.click_cnt DESC;
