config {
    type: "operations",
    tags: ["collart_fashion"],
    description: "Fashion daily revenue aggregated from user-day Stripe revenue",
    dependencies: ["fashion_user_revenue"]
}

js {
  const cfg = require("../../../../includes/collart_fashion/constants");
}

DECLARE from_date DATE DEFAULT ${dataform.projectConfig.vars.biz_date} - ${cfg.REVENUE_BACKFILL_DAYS};
DECLARE to_date   DATE DEFAULT ${dataform.projectConfig.vars.biz_date};

DELETE FROM `aidata2025.ads_collart.ads_oper_revenue_collart_fashion_di`
WHERE event_date BETWEEN from_date AND to_date;

INSERT INTO `aidata2025.ads_collart.ads_oper_revenue_collart_fashion_di`

SELECT
  event_date,
  "${cfg.PACKAGE_NAME}" AS package_name,
  ROUND(SUM(revenue_usd), 2) AS revenue_usd,
  SUM(pay_cnt) AS pay_cnt,
  CURRENT_TIMESTAMP() AS etl_time
FROM `aidata2025.ads_collart.ads_oper_user_revenue_collart_fashion_di`
WHERE event_date BETWEEN from_date AND to_date
  AND package_name = "${cfg.PACKAGE_NAME}"
GROUP BY event_date;
