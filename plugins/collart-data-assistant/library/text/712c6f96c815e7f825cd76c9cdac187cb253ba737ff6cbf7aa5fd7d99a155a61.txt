-- =============================================
-- Collart Android 首日订阅率趋势分析
-- 口径定义：当日新增用户中，在当天完成订阅转化的比例
-- 时间范围：过去30天（2026-03-22 至 2026-04-21）
-- =============================================

WITH daily_new_users AS (
  -- 每日新增用户（首次打开应用）
  SELECT
    dt,
    COUNT(DISTINCT user_pseudo_id) AS new_users
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN '20260322' AND '20260421'
    AND event_name = 'first_open'
    AND platform = 'ANDROID'
  GROUP BY dt
),

daily_first_day_subs AS (
  -- 当日新增且当日完成订阅的用户
  SELECT
    n.dt,
    COUNT(DISTINCT n.user_pseudo_id) AS first_day_subscribers
  FROM (
    SELECT
      dt,
      user_pseudo_id
    FROM `storytemplate-10a27.analytics_232977577.events_*`
    WHERE _TABLE_SUFFIX BETWEEN '20260322' AND '20260421'
      AND event_name = 'first_open'
      AND platform = 'ANDROID'
  ) n
  INNER JOIN (
    SELECT
      user_pseudo_id,
      MIN(dt) AS sub_dt
    FROM `storytemplate-10a27.analytics_232977577.events_*`
    WHERE _TABLE_SUFFIX BETWEEN '20260322' AND '20260421'
      AND event_name = 'in_app_purchase'
      AND platform = 'ANDROID'
      AND (
        event_params.value.string_value LIKE '%sub%'
        OR event_params.value.string_value LIKE '%会员%'
        OR event_params.value.string_value LIKE '%vip%'
      )
    GROUP BY user_pseudo_id
  ) s
  ON n.user_pseudo_id = s.user_pseudo_id AND n.dt = s.sub_dt
  GROUP BY n.dt
)

SELECT
  n.dt,
  n.new_users,
  COALESCE(s.first_day_subscribers, 0) AS first_day_subscribers,
  ROUND(COALESCE(s.first_day_subscribers, 0) * 100.0 / n.new_users, 2) AS first_day_sub_rate_pct
FROM daily_new_users n
LEFT JOIN daily_first_day_subs s ON n.dt = s.dt
ORDER BY n.dt DESC
