config {
    type: "operations",
    tags: ["collart_fashion"],
    description: "Fashion daily new/retention/DAU + revenue",
    dependencies: ["fashion_revenue"]
}

js {
  const cfg = require("../../../../includes/collart_fashion/constants");
}

DECLARE from_date DATE DEFAULT ${dataform.projectConfig.vars.biz_date} - ${cfg.METRIC_BACKFILL_DAYS};
DECLARE to_date   DATE DEFAULT ${dataform.projectConfig.vars.biz_date};

DELETE FROM `aidata2025.ads_collart.ads_oper_basic_metric_collart_fashion_di`
WHERE event_date BETWEEN from_date AND to_date;

INSERT INTO `aidata2025.ads_collart.ads_oper_basic_metric_collart_fashion_di`

WITH indicator AS (
  SELECT
    event_date,
    SUM(IF(is_new, dau, 0)) AS new_users,
    SAFE_DIVIDE(SUM(IF(is_new, retain.d2, 0)), SUM(IF(is_new, dau, 0))) AS retain_2,
    SAFE_DIVIDE(SUM(IF(is_new, retain.d7, 0)), SUM(IF(is_new, dau, 0))) AS retain_7,
    SUM(IF(is_new AND traffic_src_type = "organic", dau, 0)) AS new_users_organic,
    SUM(dau) AS active_users
  FROM `aidata2025.ads_collartfashion.ads_oper_basic_indicator_attr_di`
  WHERE event_date BETWEEN from_date AND to_date
    AND package_name = "${cfg.PACKAGE_NAME}"
  GROUP BY event_date
)

SELECT
  i.event_date,
  "${cfg.PACKAGE_NAME}" AS package_name,
  i.new_users,
  i.retain_2,
  i.retain_7,
  i.new_users_organic,
  i.active_users,
  IFNULL(r.revenue_usd, 0) AS revenue_usd,
  IFNULL(r.pay_cnt, 0) AS pay_cnt,
  CURRENT_TIMESTAMP() AS etl_time
FROM indicator i
LEFT JOIN `aidata2025.ads_collart.ads_oper_revenue_collart_fashion_di` r
  USING (event_date);
