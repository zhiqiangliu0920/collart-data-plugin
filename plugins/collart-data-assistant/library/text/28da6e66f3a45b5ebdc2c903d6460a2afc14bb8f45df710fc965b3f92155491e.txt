-- 查询日期: 2026-05-13
-- 目标: Collart Web 昨日收入 (2026-05-12)
-- 来源表: pubdata2025.dwd.dwd_cdct_revenue_stripe_di
-- 注意事项: 该表有重复数据，需以 transaction_Id 去重

SELECT
  event_date,
  COUNT(DISTINCT transaction_Id) AS order_cnt,
  ROUND(SUM(revneue), 2) AS total_revenue,
  ROUND(SUM(CASE WHEN description_type = 'Subscription creation' AND revneue > 0 THEN revneue END), 2) AS new_sub_revenue,
  ROUND(SUM(CASE WHEN description_type = 'Subscription update' AND revneue > 0 THEN revneue END), 2) AS renew_revenue,
  ROUND(SUM(CASE WHEN description_type = 'credit' AND revneue > 0 THEN revneue END), 2) AS pack_revenue
FROM (
  SELECT DISTINCT
    event_date,
    transaction_Id,
    revneue,
    description_type
  FROM pubdata2025.dwd.dwd_cdct_revenue_stripe_di
) t
WHERE event_date = '2026-05-12'
  AND revneue > 0
GROUP BY event_date
