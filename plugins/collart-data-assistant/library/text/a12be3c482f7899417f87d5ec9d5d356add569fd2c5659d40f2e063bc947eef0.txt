-- ============================================================
-- SQL: 美国 Trending 模块推荐模板稳定性分析
-- 数据范围: 2026-05-20 ~ 2026-05-26 (7天)
-- 数据源: aidata2025.dws.dws_template_recommend_score_di
-- 关联: aidata2025.ods.template
-- ============================================================

-- Query 1: 每日 TOP 30 推荐模板变化透视
WITH ranked AS (
  SELECT
    a.event_date,
    a.country_code AS country,
    a.temp_id AS temp_name,
    a.temp_score,
    ROW_NUMBER() OVER (
      PARTITION BY a.event_date, a.country_code, a.traffic_src_type
      ORDER BY a.temp_score DESC
    ) AS rank_num
  FROM aidata2025.dws.dws_template_recommend_score_di a
  LEFT JOIN (
    SELECT id, unique_id
    FROM `aidata2025.ods.template`
    WHERE platform = 'collart-android'
  ) c ON a.temp_id = c.unique_id
  WHERE a.event_date BETWEEN DATE '2026-05-20' AND DATE '2026-05-26'
    AND a.traffic_src_type = 'all'
    AND a.country_code = 'US'
)
SELECT
  rank_num,
  string_agg(CASE WHEN event_date = DATE '2026-05-20' THEN temp_name END) AS `2026_05_20`,
  string_agg(CASE WHEN event_date = DATE '2026-05-21' THEN temp_name END) AS `2026_05_21`,
  string_agg(CASE WHEN event_date = DATE '2026-05-22' THEN temp_name END) AS `2026_05_22`,
  string_agg(CASE WHEN event_date = DATE '2026-05-23' THEN temp_name END) AS `2026_05_23`,
  string_agg(CASE WHEN event_date = DATE '2026-05-24' THEN temp_name END) AS `2026_05_24`,
  string_agg(CASE WHEN event_date = DATE '2026-05-25' THEN temp_name END) AS `2026_05_25`,
  string_agg(CASE WHEN event_date = DATE '2026-05-26' THEN temp_name END) AS `2026_05_26`
FROM ranked
WHERE rank_num <= 30
GROUP BY 1
ORDER BY 1;

-- Query 2: 模板生命周期查询 (在 TOP 30 出现的天数)
WITH ranked AS (
  SELECT
    a.event_date,
    a.country_code AS country,
    a.temp_id AS temp_name,
    ROW_NUMBER() OVER (
      PARTITION BY a.event_date, a.country_code, a.traffic_src_type
      ORDER BY a.temp_score DESC
    ) AS rank_num
  FROM aidata2025.dws.dws_template_recommend_score_di a
  LEFT JOIN (
    SELECT id, unique_id
    FROM `aidata2025.ods.template`
    WHERE platform = 'collart-android'
  ) c ON a.temp_id = c.unique_id
  WHERE a.event_date BETWEEN DATE '2026-05-20' AND DATE '2026-05-26'
    AND a.traffic_src_type = 'all'
    AND a.country_code = 'US'
)
SELECT
  temp_name,
  COUNT(DISTINCT event_date) AS lifespan_days,
  COUNT(*) AS total_appearances,
  AVG(rank_num) AS avg_rank,
  MIN(rank_num) AS best_rank,
  MAX(rank_num) AS worst_rank
FROM ranked
WHERE rank_num <= 30
GROUP BY 1
ORDER BY lifespan_days DESC, avg_rank ASC
LIMIT 50;
