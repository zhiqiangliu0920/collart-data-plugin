-- ============================================
-- Collart Web 大R用户分析（累计充值≥50美元）
-- 创建日期：2026-03-15
-- ============================================

-- 第一步：识别大R用户（累计充值≥50美元）
WITH whale_users AS (
  SELECT 
    user_id,
    SUM(revneue) AS total_revenue,
    MIN(event_date) AS first_pay_date,
    MAX(event_date) AS last_pay_date,
    COUNT(DISTINCT transaction_id) AS pay_times,
    COUNT(DISTINCT DATE_TRUNC(event_date, MONTH)) AS pay_months
  FROM (
    SELECT DISTINCT 
      event_date,
      transaction_id,
      stripe_customer_id AS user_id,
      revneue
    FROM `pubdata2025.dwd.dwd_cdct_revenue_stripe_di`
    WHERE revneue > 0
  )
  GROUP BY user_id
  HAVING total_revenue >= 50
),

-- 第二步：计算每日新增大R（按首次付费日期）
whale_daily AS (
  SELECT 
    first_pay_date AS event_date,
    COUNT(DISTINCT user_id) AS new_whale_users,
    SUM(total_revenue) AS new_whale_revenue
  FROM whale_users
  GROUP BY first_pay_date
),

-- 第三步：计算累计大R用户数和收入
daily_summary AS (
  SELECT 
    event_date,
    new_whale_users,
    new_whale_revenue,
    SUM(new_whale_users) OVER (ORDER BY event_date) AS cumulative_whale_users,
    SUM(new_whale_revenue) OVER (ORDER BY event_date) AS cumulative_whale_revenue
  FROM whale_daily
)

-- 查询1：大R增长趋势（按日）
SELECT * FROM daily_summary
ORDER BY event_date DESC
LIMIT 1000;


-- ============================================
-- 查询2：大R用户月度趋势聚合
-- ============================================
WITH whale_users AS (
  SELECT 
    user_id,
    SUM(revneue) AS total_revenue,
    MIN(event_date) AS first_pay_date,
    MAX(event_date) AS last_pay_date,
    COUNT(DISTINCT transaction_id) AS pay_times
  FROM (
    SELECT DISTINCT 
      event_date,
      transaction_id,
      stripe_customer_id AS user_id,
      revneue
    FROM `pubdata2025.dwd.dwd_cdct_revenue_stripe_di`
    WHERE revneue > 0
  )
  GROUP BY user_id
  HAVING total_revenue >= 50
)
SELECT 
  DATE_TRUNC(first_pay_date, MONTH) AS month,
  COUNT(DISTINCT user_id) AS new_whale_users,
  AVG(total_revenue) AS avg_whale_ltv,
  PERCENTILE_CONT(total_revenue, 0.5) OVER () AS median_whale_ltv,
  AVG(pay_times) AS avg_pay_times
FROM whale_users
GROUP BY 1
ORDER BY 1 DESC;


-- ============================================
-- 查询3：大R用户行为特征分析
-- ============================================
WITH whale_users AS (
  SELECT 
    user_id,
    SUM(revneue) AS total_revenue,
    MIN(event_date) AS first_pay_date,
    MAX(event_date) AS last_pay_date,
    COUNT(DISTINCT transaction_id) AS pay_times
  FROM (
    SELECT DISTINCT 
      event_date,
      transaction_id,
      stripe_customer_id AS user_id,
      revneue
    FROM `pubdata2025.dwd.dwd_cdct_revenue_stripe_di`
    WHERE revneue > 0
  )
  GROUP BY user_id
  HAVING total_revenue >= 50
),
-- 关联用户行为数据
whale_behavior AS (
  SELECT 
    w.user_id,
    w.total_revenue,
    w.first_pay_date,
    w.last_pay_date,
    w.pay_times,
    u.country,
    u.device_category,
    u.operating_system,
    u.traffic_src_source,
    u.traffic_src_type
  FROM whale_users w
  LEFT JOIN `aidata2025.dwd.dwd_oper_user_collartweb_di` u
    ON w.user_id = u.user_id
    AND u.event_date BETWEEN w.first_pay_date AND w.first_pay_date
  WHERE u.user_id IS NOT NULL
  QUALIFY ROW_NUMBER() OVER (PARTITION BY w.user_id ORDER BY u.event_date) = 1
)
SELECT 
  -- 国家分布
  country,
  COUNT(DISTINCT user_id) AS whale_user_count,
  ROUND(COUNT(DISTINCT user_id) * 100.0 / SUM(COUNT(DISTINCT user_id)) OVER (), 2) AS pct_of_total,
  AVG(total_revenue) AS avg_revenue,
  AVG(pay_times) AS avg_pay_times
FROM whale_behavior
WHERE country IS NOT NULL
GROUP BY country
ORDER BY whale_user_count DESC
LIMIT 20;


-- ============================================
-- 查询4：大R用户设备与渠道分布
-- ============================================
WITH whale_users AS (
  SELECT 
    user_id,
    SUM(revneue) AS total_revenue,
    MIN(event_date) AS first_pay_date,
    COUNT(DISTINCT transaction_id) AS pay_times
  FROM (
    SELECT DISTINCT 
      event_date,
      transaction_id,
      stripe_customer_id AS user_id,
      revneue
    FROM `pubdata2025.dwd.dwd_cdct_revenue_stripe_di`
    WHERE revneue > 0
  )
  GROUP BY user_id
  HAVING total_revenue >= 50
),
whale_first_visit AS (
  SELECT 
    w.user_id,
    w.total_revenue,
    w.pay_times,
    u.device_category,
    u.operating_system,
    u.browser,
    u.traffic_src_source,
    u.traffic_src_type,
    u.is_new
  FROM whale_users w
  LEFT JOIN `aidata2025.dwd.dwd_oper_user_collartweb_di` u
    ON w.user_id = u.user_id
    AND u.event_date <= w.first_pay_date
  WHERE u.user_id IS NOT NULL
  QUALIFY ROW_NUMBER() OVER (PARTITION BY w.user_id ORDER BY u.event_date DESC) = 1
)
-- 设备类型分布
SELECT 
  'Device Category' AS dimension,
  device_category AS value,
  COUNT(DISTINCT user_id) AS whale_count,
  ROUND(AVG(total_revenue), 2) AS avg_ltv
FROM whale_first_visit
WHERE device_category IS NOT NULL
GROUP BY device_category
UNION ALL
-- 操作系统分布
SELECT 
  'Operating System' AS dimension,
  operating_system AS value,
  COUNT(DISTINCT user_id) AS whale_count,
  ROUND(AVG(total_revenue), 2) AS avg_ltv
FROM whale_first_visit
WHERE operating_system IS NOT NULL
GROUP BY operating_system
UNION ALL
-- 流量来源类型
SELECT 
  'Traffic Type' AS dimension,
  traffic_src_type AS value,
  COUNT(DISTINCT user_id) AS whale_count,
  ROUND(AVG(total_revenue), 2) AS avg_ltv
FROM whale_first_visit
WHERE traffic_src_type IS NOT NULL
GROUP BY traffic_src_type
ORDER BY dimension, whale_count DESC;


-- ============================================
-- 查询5：大R用户付费频次与金额分层
-- ============================================
WITH whale_users AS (
  SELECT 
    user_id,
    SUM(revneue) AS total_revenue,
    MIN(event_date) AS first_pay_date,
    MAX(event_date) AS last_pay_date,
    COUNT(DISTINCT transaction_id) AS pay_times,
    DATE_DIFF(MAX(event_date), MIN(event_date), DAY) AS lifespan_days
  FROM (
    SELECT DISTINCT 
      event_date,
      transaction_id,
      stripe_customer_id AS user_id,
      revneue
    FROM `pubdata2025.dwd.dwd_cdct_revenue_stripe_di`
    WHERE revneue > 0
  )
  GROUP BY user_id
  HAVING total_revenue >= 50
)
SELECT 
  CASE 
    WHEN total_revenue >= 500 THEN '超R (≥$500)'
    WHEN total_revenue >= 200 THEN '大R ($200-$499)'
    WHEN total_revenue >= 100 THEN '中R ($100-$199)'
    ELSE '小R ($50-$99)'
  END AS revenue_tier,
  COUNT(DISTINCT user_id) AS user_count,
  ROUND(AVG(total_revenue), 2) AS avg_revenue,
  ROUND(AVG(pay_times), 1) AS avg_pay_times,
  ROUND(AVG(lifespan_days), 0) AS avg_lifespan_days,
  ROUND(SUM(total_revenue), 2) AS total_revenue_contrib,
  ROUND(SUM(total_revenue) * 100.0 / SUM(SUM(total_revenue)) OVER (), 2) AS revenue_pct
FROM whale_users
GROUP BY revenue_tier
ORDER BY avg_revenue DESC;


-- ============================================
-- 查询6：大R用户收入贡献占比（vs 全体用户）
-- ============================================
WITH all_users AS (
  SELECT 
    stripe_customer_id AS user_id,
    SUM(revneue) AS total_revenue
  FROM (
    SELECT DISTINCT 
      event_date,
      transaction_id,
      stripe_customer_id,
      revneue
    FROM `pubdata2025.dwd.dwd_cdct_revenue_stripe_di`
    WHERE revneue > 0
  )
  GROUP BY user_id
),
user_segments AS (
  SELECT 
    user_id,
    total_revenue,
    CASE 
      WHEN total_revenue >= 50 THEN 'Whale (≥$50)'
      WHEN total_revenue >= 10 THEN 'Dolphin ($10-$49)'
      WHEN total_revenue >= 1 THEN 'Minnow ($1-$9)'
      ELSE 'Tiny (<$1)'
    END AS user_segment
  FROM all_users
)
SELECT 
  user_segment,
  COUNT(DISTINCT user_id) AS user_count,
  ROUND(COUNT(DISTINCT user_id) * 100.0 / SUM(COUNT(DISTINCT user_id)) OVER (), 2) AS user_pct,
  ROUND(SUM(total_revenue), 2) AS revenue_contrib,
  ROUND(SUM(total_revenue) * 100.0 / SUM(SUM(total_revenue)) OVER (), 2) AS revenue_pct,
  ROUND(AVG(total_revenue), 2) AS avg_revenue_per_user
FROM user_segments
GROUP BY user_segment
ORDER BY avg_revenue_per_user DESC;


-- ============================================
-- 查询7：近期大R用户留存/复购分析
-- ============================================
WITH whale_users AS (
  SELECT 
    user_id,
    SUM(revneue) AS total_revenue,
    MIN(event_date) AS first_pay_date,
    MAX(event_date) AS last_pay_date,
    COUNT(DISTINCT transaction_id) AS pay_times
  FROM (
    SELECT DISTINCT 
      event_date,
      transaction_id,
      stripe_customer_id AS user_id,
      revneue
    FROM `pubdata2025.dwd.dwd_cdct_revenue_stripe_di`
    WHERE revneue > 0
  )
  GROUP BY user_id
  HAVING total_revenue >= 50
),
monthly_activity AS (
  SELECT 
    w.user_id,
    DATE_TRUNC(s.event_date, MONTH) AS activity_month,
    SUM(s.revneue) AS monthly_revenue
  FROM whale_users w
  JOIN (
    SELECT DISTINCT 
      event_date,
      transaction_id,
      stripe_customer_id AS user_id,
      revneue
    FROM `pubdata2025.dwd.dwd_cdct_revenue_stripe_di`
    WHERE revneue > 0
  ) s ON w.user_id = s.user_id
  GROUP BY w.user_id, DATE_TRUNC(s.event_date, MONTH)
)
SELECT 
  activity_month,
  COUNT(DISTINCT user_id) AS active_whale_users,
  ROUND(SUM(monthly_revenue), 2) AS monthly_whale_revenue,
  ROUND(AVG(monthly_revenue), 2) AS avg_revenue_per_active_whale
FROM monthly_activity
GROUP BY activity_month
ORDER BY activity_month DESC
LIMIT 24;
