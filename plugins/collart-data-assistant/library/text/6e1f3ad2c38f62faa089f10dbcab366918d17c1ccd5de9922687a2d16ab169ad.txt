WITH module_show AS (
  SELECT 
    (
      SELECT ep.value.string_value 
      FROM UNNEST(event_params) ep 
      WHERE ep.key = 'module_id'
    ) AS module_id,
    COUNT(*) AS show_count
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX >= '20260306' AND _TABLE_SUFFIX <= '20260312'
    AND app_info.id = 'free.ai.photo.generator.collart.ai'
    AND event_name = 'home_module_show'
  GROUP BY module_id
),
module_click AS (
  SELECT 
    (
      SELECT ep.value.string_value 
      FROM UNNEST(event_params) ep 
      WHERE ep.key = 'module_id'
    ) AS module_id,
    COUNT(*) AS click_count
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX >= '20260306' AND _TABLE_SUFFIX <= '20260312'
    AND app_info.id = 'free.ai.photo.generator.collart.ai'
    AND event_name = 'home_module_click'
  GROUP BY module_id
)
SELECT 
  s.module_id,
  s.show_count,
  c.click_count,
  ROUND(c.click_count * 100.0 / s.show_count, 2) AS ctr_pct
FROM module_show s
LEFT JOIN module_click c ON s.module_id = c.module_id
ORDER BY s.show_count DESC
LIMIT 20
