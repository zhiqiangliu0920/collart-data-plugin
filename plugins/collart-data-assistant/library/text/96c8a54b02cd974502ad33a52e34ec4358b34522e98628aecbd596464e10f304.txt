-- Collart Web 过去3天分 Campaign 新增用户
-- 日期范围: 2026-03-08 至 2026-03-10
-- 创建时间: 2026-03-11

SELECT 
  event_date,
  traffic_src_name,
  traffic_src_source,
  traffic_src_type,
  COUNT(DISTINCT user_pseudo_id) AS new_users
FROM aidata2025.ads_collart.ads_oper_user_new_collartweb_di
WHERE event_date BETWEEN '2026-03-08' AND '2026-03-10'
GROUP BY event_date, traffic_src_name, traffic_src_source, traffic_src_type
ORDER BY event_date DESC, new_users DESC;
