-- 分析 4月1日以来新用户的付费路径
-- 1. 直接付费：没有看过激励广告就付费
-- 2. 激励广告转化：先看激励广告，之后才付费
-- 创建时间：2026-04-14

WITH
-- 1. 4月1日以来的新用户
new_users AS (
  SELECT
    user_pseudo_id,
    MIN(event_date) AS first_install_date
  FROM aidata2025.dwd.dwd_oper_user_collart_di
  WHERE is_new = TRUE
    AND package_name = 'free.ai.photo.generator.collart.ai'
    AND event_date >= '2026-04-01'
  GROUP BY user_pseudo_id
),

-- 2. 新用户中的付费事件
paid_users AS (
  SELECT
    e.user_pseudo_id,
    MIN(TIMESTAMP_MICROS(e.event_timestamp)) AS first_pay_time
  FROM `storytemplate-10a27.analytics_232977577.events_*` e
  INNER JOIN new_users n ON e.user_pseudo_id = n.user_pseudo_id
  WHERE e._TABLE_SUFFIX BETWEEN '20260401' AND FORMAT_DATE('%Y%m%d', CURRENT_DATE())
    AND e.app_info.id = 'free.ai.photo.generator.collart.ai'
    AND e.event_name = 'vip_subscribe_succeed'
  GROUP BY e.user_pseudo_id
),

-- 3. 首次付费前的激励广告展示
ad_watch_before_pay AS (
  SELECT
    p.user_pseudo_id,
    MIN(TIMESTAMP_MICROS(e.event_timestamp)) AS first_ad_show_time
  FROM paid_users p
  INNER JOIN `storytemplate-10a27.analytics_232977577.events_*` e
    ON p.user_pseudo_id = e.user_pseudo_id
  WHERE e._TABLE_SUFFIX BETWEEN '20260401' AND FORMAT_DATE('%Y%m%d', CURRENT_DATE())
    AND e.app_info.id = 'free.ai.photo.generator.collart.ai'
    AND e.event_name = 'generate_watch_ad_show'
    AND TIMESTAMP_MICROS(e.event_timestamp) < p.first_pay_time
  GROUP BY p.user_pseudo_id
),

-- 4. 分类统计
classified AS (
  SELECT
    p.user_pseudo_id,
    p.first_pay_time,
    CASE
      WHEN a.user_pseudo_id IS NOT NULL THEN 'ad_converted'
      ELSE 'direct_pay'
    END AS pay_path
  FROM paid_users p
  LEFT JOIN ad_watch_before_pay a ON p.user_pseudo_id = a.user_pseudo_id
)

SELECT
  pay_path,
  CASE pay_path
    WHEN 'direct_pay' THEN '直接付费（未看激励广告）'
    WHEN 'ad_converted' THEN '激励广告转化'
  END AS path_name,
  COUNT(*) AS user_count,
  ROUND(COUNT(*) * 100.0 / SUM(COUNT(*)) OVER(), 2) AS pct
FROM classified
GROUP BY pay_path
ORDER BY
  CASE pay_path
    WHEN 'direct_pay' THEN 1
    WHEN 'ad_converted' THEN 2
  END
