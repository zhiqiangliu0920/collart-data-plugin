-- img2video 与 img2img 服务成功率趋势（5月1日-5月13日，分国家 Top10）
WITH submit_events AS (
  SELECT
    PARSE_DATE('%Y%m%d', event_date) AS dt,
    geo.country AS country,
    event_name,
    COUNT(*) AS cnt
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN '20260501' AND '20260513'
    AND app_info.id = 'free.ai.photo.generator.collart.ai'
    AND event_name IN ('ai_service_submit_start', 'ai_service_submit_success', 'ai_service_submit_failed')
    AND (SELECT value.string_value FROM UNNEST(event_params) WHERE key = 'service_type') IN ('img2video', 'img2img')
  GROUP BY 1, 2, 3
)
SELECT
  dt,
  country,
  SUM(CASE WHEN event_name = 'ai_service_submit_start' THEN cnt ELSE 0 END) AS submit_start,
  SUM(CASE WHEN event_name = 'ai_service_submit_success' THEN cnt ELSE 0 END) AS submit_success,
  ROUND(SUM(CASE WHEN event_name = 'ai_service_submit_success' THEN cnt ELSE 0 END) / NULLIF(SUM(CASE WHEN event_name = 'ai_service_submit_start' THEN cnt ELSE 0 END), 0), 4) AS success_rate
FROM submit_events
GROUP BY dt, country
HAVING submit_start > 0
ORDER BY dt, submit_start DESC;
