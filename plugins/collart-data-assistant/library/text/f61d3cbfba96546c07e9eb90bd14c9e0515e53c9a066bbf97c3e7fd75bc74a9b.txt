WITH daily_dau AS (
  SELECT 
    DATE(TIMESTAMP_MICROS(event_timestamp)) AS dt,
    COUNT(DISTINCT user_pseudo_id) AS dau
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX >= '20260306' AND _TABLE_SUFFIX <= '20260312'
    AND app_info.id = 'free.ai.photo.generator.collart.ai'
    AND event_name = 'session_start'
  GROUP BY dt
),
feature_uv AS (
  SELECT 
    DATE(TIMESTAMP_MICROS(event_timestamp)) AS dt,
    (
      SELECT ep.value.string_value 
      FROM UNNEST(event_params) ep 
      WHERE ep.key = 'feature'
    ) AS feature,
    COUNT(DISTINCT user_pseudo_id) AS uv
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX >= '20260306' AND _TABLE_SUFFIX <= '20260312'
    AND app_info.id = 'free.ai.photo.generator.collart.ai'
    AND event_name = 'home_feature_click'
  GROUP BY dt, feature
)
SELECT 
  f.feature,
  SUM(f.uv) AS total_uv,
  ROUND(SUM(f.uv) * 100.0 / SUM(d.dau), 2) AS ctr_pct
FROM feature_uv f
JOIN daily_dau d ON f.dt = d.dt
GROUP BY f.feature
ORDER BY ctr_pct DESC
