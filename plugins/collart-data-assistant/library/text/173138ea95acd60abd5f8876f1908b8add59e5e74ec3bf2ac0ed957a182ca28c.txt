-- 查询: Collart Android 违规警告弹窗埋点 PV/UV（2026-05-08 以来）
-- 作者: 强强
-- 日期: 2026-05-28
-- 数据源: storytemplate-10a27.analytics_232977577.events_*

SELECT
  event_name,
  event_date,
  COUNT(*) AS pv,
  COUNT(DISTINCT user_pseudo_id) AS uv
FROM `storytemplate-10a27.analytics_232977577.events_*`
WHERE _TABLE_SUFFIX BETWEEN '20260508' AND FORMAT_DATE('%Y%m%d', CURRENT_DATE())
  AND app_info.id = 'free.ai.photo.generator.collart.ai'
  AND event_name IN (
    'violation_guard_dialog_warn_show',        -- 违规警告弹窗：提示还有N次会被临时限制
    'violation_guard_dialog_restricting_show',  -- 违规警告弹窗：功能限制中，等解封倒计时
    'violation_guard_dialog_restrict_show',     -- 即将封号警告弹窗：提示还有N次会被封号
    'violation_guard_dialog_ban_show'          -- 封号弹窗展示：点击生成按钮，弹窗提示已封号
  )
GROUP BY event_name, event_date
ORDER BY event_date DESC, pv DESC
