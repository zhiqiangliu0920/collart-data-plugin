-- ============================================================
-- Collart Android: For You vs Trending 首页模块分析
-- 数据范围: 2026-05-21 ~ 2026-05-27 (最近7天)
-- 数据源: storytemplate-10a27.analytics_232977577.events_*
-- ============================================================
-- 埋点事件说明:
--   home_module_show  : 首页模块曝光
--   home_module_click : 首页模块点击
--   Home_Template_Show: 模块内具体模板曝光
--   Home_Template_Click: 模块内具体模板点击
--   event_params.module_id  标识模块:
--     - recommend_module = For You
--     - trending_module  = Trending (含 Most Popular / Trending 标题)
-- ============================================================

-- ------------------------------------------------
-- Query 1: 模块级汇总 (For You vs Trending)
-- ------------------------------------------------
WITH module_show AS (
  SELECT
    IFNULL((SELECT ep.value.string_value FROM UNNEST(event_params) ep WHERE ep.key = 'module_id'), 'NULL') AS module_id,
    IFNULL((SELECT ep.value.string_value FROM UNNEST(event_params) ep WHERE ep.key = 'module_title'), 'NULL') AS module_title,
    COUNT(*) AS show_pv,
    COUNT(DISTINCT user_pseudo_id) AS show_uv
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN '20260521' AND '20260527'
    AND app_info.id = 'free.ai.photo.generator.collart.ai'
    AND event_name = 'home_module_show'
    AND (SELECT ep.value.string_value FROM UNNEST(event_params) ep WHERE ep.key = 'module_id') IN ('recommend_module', 'trending_module')
  GROUP BY 1, 2
),
module_click AS (
  SELECT
    IFNULL((SELECT ep.value.string_value FROM UNNEST(event_params) ep WHERE ep.key = 'module_id'), 'NULL') AS module_id,
    IFNULL((SELECT ep.value.string_value FROM UNNEST(event_params) ep WHERE ep.key = 'module_title'), 'NULL') AS module_title,
    COUNT(*) AS click_pv,
    COUNT(DISTINCT user_pseudo_id) AS click_uv
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN '20260521' AND '20260527'
    AND app_info.id = 'free.ai.photo.generator.collart.ai'
    AND event_name = 'home_module_click'
    AND (SELECT ep.value.string_value FROM UNNEST(event_params) ep WHERE ep.key = 'module_id') IN ('recommend_module', 'trending_module')
  GROUP BY 1, 2
)
SELECT
  s.module_id,
  s.module_title,
  s.show_pv,
  s.show_uv,
  IFNULL(c.click_pv, 0) AS click_pv,
  IFNULL(c.click_uv, 0) AS click_uv,
  ROUND(IFNULL(c.click_pv, 0) * 100.0 / s.show_pv, 2) AS ctr_pct,
  ROUND(s.show_pv * 1.0 / s.show_uv, 2) AS pv_per_uv
FROM module_show s
LEFT JOIN module_click c ON s.module_id = c.module_id AND s.module_title = c.module_title
ORDER BY s.show_pv DESC;

-- ------------------------------------------------
-- Query 2: 每日趋势
-- ------------------------------------------------
WITH show_daily AS (
  SELECT
    _TABLE_SUFFIX AS dt,
    IFNULL((SELECT ep.value.string_value FROM UNNEST(event_params) ep WHERE ep.key = 'module_id'), 'NULL') AS module_id,
    COUNT(*) AS show_pv,
    COUNT(DISTINCT user_pseudo_id) AS show_uv
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN '20260521' AND '20260527'
    AND app_info.id = 'free.ai.photo.generator.collart.ai'
    AND event_name = 'home_module_show'
    AND (SELECT ep.value.string_value FROM UNNEST(event_params) ep WHERE ep.key = 'module_id') IN ('recommend_module', 'trending_module')
  GROUP BY 1, 2
),
click_daily AS (
  SELECT
    _TABLE_SUFFIX AS dt,
    IFNULL((SELECT ep.value.string_value FROM UNNEST(event_params) ep WHERE ep.key = 'module_id'), 'NULL') AS module_id,
    COUNT(*) AS click_pv,
    COUNT(DISTINCT user_pseudo_id) AS click_uv
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN '20260521' AND '20260527'
    AND app_info.id = 'free.ai.photo.generator.collart.ai'
    AND event_name = 'home_module_click'
    AND (SELECT ep.value.string_value FROM UNNEST(event_params) ep WHERE ep.key = 'module_id') IN ('recommend_module', 'trending_module')
  GROUP BY 1, 2
)
SELECT
  s.dt,
  s.module_id,
  s.show_pv,
  s.show_uv,
  IFNULL(c.click_pv, 0) AS click_pv,
  IFNULL(c.click_uv, 0) AS click_uv,
  ROUND(IFNULL(c.click_pv, 0) * 100.0 / s.show_pv, 2) AS ctr_pct
FROM show_daily s
LEFT JOIN click_daily c ON s.dt = c.dt AND s.module_id = c.module_id
ORDER BY s.dt, s.module_id;
