-- Web 按登录 UUID 查画像（一行一台设备）
-- 红线（2026-09-07）：不要 WHERE p.user_id = @uid
-- 表文档：tables/aidata2025.ads_collartweb.ads_oper_user_profile_df.md

DECLARE uid STRING DEFAULT '{REDACTED_EXAMPLE_ID}';

SELECT
  p.user_pseudo_id,
  p.user_id AS last_login_user_id,
  ARRAY_LENGTH(p.user_ids) AS n_login_ids,
  p.user_ids,
  p.first_open_date,
  p.last_active_date,
  p.current_active_days,
  p.country,
  p.traffic_src_type,
  p.is_vip,
  p.revenue
FROM `aidata2025.ads_collartweb.ads_oper_user_profile_df` AS p
WHERE uid IN UNNEST(p.user_ids)
