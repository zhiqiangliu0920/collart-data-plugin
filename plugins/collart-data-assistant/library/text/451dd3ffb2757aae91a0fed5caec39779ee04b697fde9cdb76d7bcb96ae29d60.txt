-- 查看 ai_service_submit_failed 的错误码/错误信息
SELECT DISTINCT
  (SELECT value.string_value FROM UNNEST(event_params) WHERE key = 'service_type') AS service_type,
  (SELECT value.string_value FROM UNNEST(event_params) WHERE key = 'error_msg') AS error_msg,
  (SELECT value.string_value FROM UNNEST(event_params) WHERE key = 'error') AS error,
  (SELECT value.int_value FROM UNNEST(event_params) WHERE key = 'error_code') AS error_code,
  COUNT(*) AS cnt
FROM `storytemplate-10a27.analytics_232977577.events_*`
WHERE _TABLE_SUFFIX BETWEEN '20260513' AND '20260513'
  AND app_info.id = 'free.ai.photo.generator.collart.ai'
  AND event_name = 'ai_service_submit_failed'
GROUP BY 1, 2, 3, 4
ORDER BY cnt DESC
LIMIT 20;
