## 三、SQL 代码（优化版）

### 1. Claim 按钮点击漏斗（PC端）

```sql
-- 一次扫描，条件聚合
WITH base AS (
  SELECT
    user_pseudo_id,
    event_name
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 14 DAY))
    AND FORMAT_DATE('%Y%m%d', CURRENT_DATE())
    AND app_info.id IS NULL
    AND device.category = 'desktop'
    AND event_name IN ('session_start', 'dev_free_credit_claim_click', 'login_show', 'login_account_choose', 'login_success')
),
claim_users AS (
  SELECT DISTINCT user_pseudo_id
  FROM base
  WHERE event_name = 'dev_free_credit_claim_click'
)
SELECT
  COUNTIF(event_name = 'session_start') AS dau,
  (SELECT COUNT(*) FROM claim_users) AS claim_click_uv,
  COUNT(DISTINCT CASE WHEN event_name = 'login_show' AND user_pseudo_id IN (SELECT user_pseudo_id FROM claim_users) THEN user_pseudo_id END) AS login_show_uv,
  COUNT(DISTINCT CASE WHEN event_name = 'login_account_choose' AND user_pseudo_id IN (SELECT user_pseudo_id FROM claim_users) THEN user_pseudo_id END) AS login_choose_uv,
  COUNT(DISTINCT CASE WHEN event_name = 'login_success' AND user_pseudo_id IN (SELECT user_pseudo_id FROM claim_users) THEN user_pseudo_id END) AS login_success_uv
FROM base
WHERE event_name = 'session_start' 
   OR (event_name IN ('login_show', 'login_account_choose', 'login_success') AND user_pseudo_id IN (SELECT user_pseudo_id FROM claim_users));
```

### 2. 登录入口漏斗（PC端）

```sql
-- 宽表先行 + 条件聚合
WITH base AS (
  SELECT
    user_pseudo_id,
    event_name,
    (SELECT ep.value.string_value FROM UNNEST(event_params) ep WHERE ep.key = 'from') AS login_from
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 14 DAY))
    AND FORMAT_DATE('%Y%m%d', CURRENT_DATE())
    AND app_info.id IS NULL
    AND device.category = 'desktop'
    AND event_name IN ('login_show', 'login_account_choose', 'login_success')
),
-- 记录每个用户首次登录弹窗的入口
user_first_show AS (
  SELECT user_pseudo_id, login_from
  FROM (
    SELECT user_pseudo_id, login_from,
           ROW_NUMBER() OVER (PARTITION BY user_pseudo_id ORDER BY event_name = 'login_show' DESC) AS rn
    FROM base
    WHERE event_name = 'login_show'
  )
  WHERE rn = 1
)
SELECT
  COALESCE(u.login_from, '(null)') AS login_from,
  COUNT(DISTINCT CASE WHEN b.event_name = 'login_show' THEN b.user_pseudo_id END) AS show_uv,
  COUNT(DISTINCT CASE WHEN b.event_name = 'login_account_choose' THEN b.user_pseudo_id END) AS choose_uv,
  COUNT(DISTINCT CASE WHEN b.event_name = 'login_success' THEN b.user_pseudo_id END) AS success_uv,
  -- 转化率
  ROUND(COUNT(DISTINCT CASE WHEN b.event_name = 'login_account_choose' THEN b.user_pseudo_id END) * 100.0 
    / NULLIF(COUNT(DISTINCT CASE WHEN b.event_name = 'login_show' THEN b.user_pseudo_id END), 0), 2) AS show_to_choose_pct,
  ROUND(COUNT(DISTINCT CASE WHEN b.event_name = 'login_success' THEN b.user_pseudo_id END) * 100.0 
    / NULLIF(COUNT(DISTINCT CASE WHEN b.event_name = 'login_account_choose' THEN b.user_pseudo_id END), 0), 2) AS choose_to_success_pct,
  ROUND(COUNT(DISTINCT CASE WHEN b.event_name = 'login_success' THEN b.user_pseudo_id END) * 100.0 
    / NULLIF(COUNT(DISTINCT CASE WHEN b.event_name = 'login_show' THEN b.user_pseudo_id END), 0), 2) AS overall_pct
FROM base b
LEFT JOIN user_first_show u ON b.user_pseudo_id = u.user_pseudo_id
GROUP BY ALL
ORDER BY show_uv DESC;
```

### 3. 登录方式漏斗（PC端）

```sql
-- 一次扫描，条件聚合
WITH base AS (
  SELECT
    user_pseudo_id,
    event_name,
    (SELECT ep.value.string_value FROM UNNEST(event_params) ep WHERE ep.key = 'type') AS login_type
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 14 DAY))
    AND FORMAT_DATE('%Y%m%d', CURRENT_DATE())
    AND app_info.id IS NULL
    AND device.category = 'desktop'
    AND event_name IN ('session_start', 'login_account_choose', 'login_success')
),
choose_users AS (
  SELECT DISTINCT user_pseudo_id, login_type
  FROM base
  WHERE event_name = 'login_account_choose'
)
SELECT
  COUNTIF(event_name = 'session_start') AS dau,
  (SELECT COUNT(*) FROM choose_users WHERE login_type = 'google') AS google_choose_uv,
  (SELECT COUNT(*) FROM choose_users WHERE login_type = 'email') AS email_choose_uv,
  COUNT(DISTINCT CASE WHEN event_name = 'login_success' 
    AND user_pseudo_id IN (SELECT user_pseudo_id FROM choose_users WHERE login_type = 'google') 
    THEN user_pseudo_id END) AS google_success_uv,
  COUNT(DISTINCT CASE WHEN event_name = 'login_success' 
    AND user_pseudo_id IN (SELECT user_pseudo_id FROM choose_users WHERE login_type = 'email') 
    THEN user_pseudo_id END) AS email_success_uv
FROM base
WHERE event_name = 'session_start';
```

> **注**：移动端查询只需将 `device.category = 'desktop'` 改为 `device.category IN ('mobile', 'tablet')`
