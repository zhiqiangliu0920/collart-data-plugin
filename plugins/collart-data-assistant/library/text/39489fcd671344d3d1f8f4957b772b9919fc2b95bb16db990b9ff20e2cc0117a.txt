-- Collart Web 过去一周 DAU 用户机型分布
-- 生成时间: 2026-04-22
-- 数据范围: 过去7天 (2026-04-15 ~ 2026-04-21)

WITH dau_users AS (
  SELECT DISTINCT
    user_pseudo_id,
    device.mobile_brand_name AS brand,
    device.mobile_model_name AS model,
    device.operating_system AS os,
    -- 原始 user_agent 用于排查
    (SELECT value.string_value FROM UNNEST(event_params) WHERE key = 'user_agent') AS user_agent
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN '20260415' AND '20260421'
    AND app_info.id IS NULL  -- Web 端
    AND user_pseudo_id IS NOT NULL
)

-- 品牌+机型分布
SELECT
  brand,
  model,
  os,
  COUNT(DISTINCT user_pseudo_id) AS dau_users,
  ROUND(100.0 * COUNT(DISTINCT user_pseudo_id) / SUM(COUNT(DISTINCT user_pseudo_id)) OVER (), 2) AS pct
FROM dau_users
GROUP BY brand, model, os
ORDER BY dau_users DESC
