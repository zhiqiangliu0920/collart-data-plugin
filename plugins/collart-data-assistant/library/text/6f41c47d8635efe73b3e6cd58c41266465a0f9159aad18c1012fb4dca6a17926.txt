-- 用户粒度收入：只读 Stripe T+1（不要 ods.orders，不要空表 HI）
-- Web：LOWER(app_name) LIKE 'collart%web%'
-- 移动端俄罗斯：UPPER(currency)='RUB'，Android=collart_android，iOS=vidart/vidart_ios/vidart-ios
-- 当天未进 T+1：等次日，或明确写「T+1 口径」

SELECT
  s.event_date AS pay_date,
  TRIM(CAST(s.user_id AS STRING)) AS user_id,
  CAST(s.transaction_id AS STRING) AS order_id,
  CAST(COALESCE(s.product_name, s.product_id) AS STRING) AS product_name,
  s.payment_channel,
  s.order_status AS status,
  DATETIME(TIMESTAMP_SECONDS(s.event_timestamp), 'Asia/Shanghai') AS paid_time_bj,
  ROUND(IFNULL(s.revneue, 0) + IFNULL(s.fee, 0), 2) AS amount_usd
FROM `pubdata2025.dwd.dwd_cdct_revenue_stripe_di` AS s
WHERE s.user_id IS NOT NULL
  AND TRIM(CAST(s.user_id AS STRING)) <> ''
  AND s.transaction_id IS NOT NULL
  AND IFNULL(s.revneue, 0) > 0
QUALIFY ROW_NUMBER() OVER (
  PARTITION BY CAST(s.transaction_id AS STRING)
  ORDER BY s.event_timestamp DESC, s.revneue DESC
) = 1
