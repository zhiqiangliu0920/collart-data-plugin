"""Read-only Dataform/BigQuery audit. Does not execute model SQL or cloud writes."""
from __future__ import annotations
import argparse, base64, concurrent.futures, hashlib, json, re, threading
from datetime import datetime, timezone
from pathlib import Path
from google.auth.transport.requests import AuthorizedSession
from google.oauth2 import service_account
from google.cloud import bigquery

ROOT = Path(__file__).resolve().parent
KNOWLEDGE = ROOT.parent.parent
REPO = 'projects/aidata2025/locations/us-east1/repositories/aidata'
API = 'https://dataform.googleapis.com/v1beta1/'
WORKSPACE = REPO + '/workspaces/script'
LOCAL = threading.local()

def credentials():
    idx = json.loads((KNOWLEDGE/'0global/service-accounts/credential_index.json').read_text(encoding='utf-8-sig'))
    key = idx['accounts'][idx['default_account']]['credential_path']
    return service_account.Credentials.from_service_account_file(key, scopes=['https://www.googleapis.com/auth/cloud-platform'])

def get(path, params=None):
    if not hasattr(LOCAL, 'session'):
        LOCAL.session = AuthorizedSession(credentials())
    r = LOCAL.session.get(API + path, params=params or {}, timeout=45)
    if not r.ok:
        raise RuntimeError(f'GET {path}: HTTP {r.status_code} {r.text[:400]}')
    return r.json()

def pages(path, key, params=None, max_pages=100):
    params=dict(params or {}); out=[]
    for _ in range(max_pages):
        data=get(path,params); out.extend(data.get(key,[]))
        if not data.get('nextPageToken'): break
        params['pageToken']=data['nextPageToken']
    return out

def save(name, obj):
    (ROOT/name).write_text(json.dumps(obj,ensure_ascii=False,indent=2,default=str),encoding='utf-8')

def collect():
    queue=['']; files=[]
    while queue:
        folder=queue.pop(0)
        entries=pages(WORKSPACE+':queryDirectoryContents','directoryEntries',{'path':folder,'pageSize':1000})
        for x in entries:
            if 'directory' in x:
                part=x['directory']; path=part if part.startswith(folder+'/') else (folder+'/'+part).strip('/')
                if path.split('/')[0] in ('definitions','includes'): queue.append(path)
            if 'file' in x:
                part=x['file']; path=part if part.startswith(folder+'/') else (folder+'/'+part).strip('/')
                files.append(path)
    save('workspace_file_index.json',files)
    candidates=[p for p in files if re.search(r'fashion|dm_collart_web_user_event|cdct_revenue_stripe|cdct_delivery_cost|workflow_settings|dataform.json|includes/',p,re.I)]
    def read(p):
        raw=base64.b64decode(get(WORKSPACE+':readFile',{'path':p})['fileContents'])
        return {'path':p,'sha256':hashlib.sha256(raw).hexdigest(),'content':raw.decode('utf-8'),'fetched_at':datetime.now(timezone.utc).isoformat()}
    with concurrent.futures.ThreadPoolExecutor(max_workers=6) as ex:
        sources=list(ex.map(read,candidates))
    save('workspace_sources.json',sources)
    for item in sources:
        if item['path'].endswith('.sqlx'):
            dest=ROOT/'sources'/item['path']; dest.parent.mkdir(parents=True,exist_ok=True); dest.write_text(item['content'],encoding='utf-8')
    print(json.dumps({'files':len(files),'sources':len(sources),'paths':[s['path'] for s in sources]},ensure_ascii=False),flush=True)
    for endpoint,key,limit in [('workflowConfigs','workflowConfigs',100),('releaseConfigs','releaseConfigs',100),('workflowInvocations','workflowInvocations',1)]:
        try:
            data=pages(REPO+'/'+endpoint,key,{'pageSize':100},max_pages=limit)
            save(endpoint+'.json',data)
            print(endpoint,len(data),flush=True)
        except Exception as e: save(endpoint+'_error.json',{'error':str(e)}); print(str(e),flush=True)

def tables():
    client=bigquery.Client(project='aidata2025',credentials=credentials())
    ids=[x.table_id for x in client.list_tables('aidata2025.ads_collartfashion')]
    full=['aidata2025.ads_collartfashion.'+x for x in ids]+[
        'aidata2025.dm.dm_collart_web_user_event_di',
        'aidata2025.ads_collart.ads_oper_user_revenue_collart_fashion_di',
        'aidata2025.ads_collart.ads_oper_revenue_collart_fashion_di',
        'pubdata2025.dwd.dwd_cdct_revenue_stripe_di',
        'aidata2025.dwd.dwd_cdct_delivery_cost_di']
    out=[]
    for name in full:
        try:
            t=client.get_table(name); out.append({'table':name,'rows':t.num_rows,'bytes':t.num_bytes,'modified':t.modified,'partitioning':t.time_partitioning.to_api_repr() if t.time_partitioning else None,'clustering':t.clustering_fields,'require_partition_filter':t.require_partition_filter,'schema':[f.to_api_repr() for f in t.schema],'type':t.table_type})
        except Exception as e: out.append({'table':name,'error':str(e)[:500]})
    save('table_metadata.json',out)
    print(json.dumps([{k:v for k,v in x.items() if k!='schema'} for x in out],default=str,ensure_ascii=False),flush=True)

def runtime():
    release=json.loads((ROOT/'releaseConfigs.json').read_text(encoding='utf-8'))[0]
    comp=release['releaseCompilationResult']
    save('compiled_result.json',get(comp))
    actions=pages(comp+':query','compilationResultActions',{'pageSize':1000})
    relevant=[a for a in actions if re.search(r'fashion|h_dm_collart_web_user_event',json.dumps(a),re.I)]
    save('compiled_fashion_actions.json',relevant)
    print('compiled_actions',len(actions),'fashion_related',len(relevant),flush=True)
    wf=json.loads((ROOT/'workflowConfigs.json').read_text(encoding='utf-8'))
    runs=[]
    for w in wf:
        if w['name'].endswith('/daily_wf_am'):
            for rec in w.get('recentScheduledExecutionRecords',[])[:3]:
                name=rec['workflowInvocation']; run=get(name)
                aa=pages(name+':query','workflowInvocationActions',{'pageSize':1000})
                selected=[a for a in aa if re.search(r'fashion|h_dm_collart_web_user_event',json.dumps(a),re.I)]
                run['fashion_actions']=selected; runs.append(run)
    save('morning_runs.json',runs)
    print(json.dumps([{'name':r['name'],'state':r['state'],'timing':r.get('invocationTiming'),'actions':[{k:v for k,v in a.items() if k!='bigqueryAction'} for a in r['fashion_actions']]} for r in runs],ensure_ascii=False),flush=True)

def query_file(path):
    query=Path(path).read_text(encoding='utf-8-sig').strip()
    if not re.match(r'^(SELECT|WITH)\b',query,re.I): raise ValueError('Only SELECT/WITH queries are supported')
    if re.search(r'\b(INSERT|UPDATE|DELETE|MERGE|CREATE|DROP|ALTER|EXPORT|CALL)\s',query,re.I): raise ValueError('Mutation keywords forbidden')
    client=bigquery.Client(project='aidata2025',credentials=credentials())
    dry=client.query(query,job_config=bigquery.QueryJobConfig(dry_run=True,use_query_cache=False),location='US')
    estimated=dry.total_bytes_processed or 0
    print('dry_run_bytes',estimated,flush=True)
    cap=8*1024**3
    if estimated>cap: raise ValueError(f'Estimated {estimated} exceeds 8 GiB audit cap')
    job=client.query(query,job_config=bigquery.QueryJobConfig(maximum_bytes_billed=cap,labels={'purpose':'fashion-readonly-audit'}),location='US')
    rows=[dict(r) for r in job.result(timeout=120)]
    save(Path(path).stem+'.json',{'job_id':job.job_id,'bytes_processed':job.total_bytes_processed,'bytes_billed':job.total_bytes_billed,'slot_millis':job.slot_millis,'rows':rows})
    print(json.dumps({'row_count':len(rows),'rows':rows[:30]},ensure_ascii=False,default=str),flush=True)

def trigger_trace():
    stats=json.loads((ROOT/'job_summary.json').read_text(encoding='utf-8'))
    runs=stats['ads_collartfashion_oper_user_active_di']['runs'][:3]
    out=[]
    for item in runs:
        name=REPO+'/workflowInvocations/'+item['invocation']
        record=get(name)
        actions=pages(name+':query','workflowInvocationActions',{'pageSize':1000})
        record['fashion_actions']=[a for a in actions if 'fashion' in a['target']['name'] or a['target']['name']=='h_dm_collart_web_user_event_di']
        out.append(record)
    save('trigger_trace.json',out)
    for r in out:
        print(json.dumps({k:v for k,v in r.items() if k!='fashion_actions'},ensure_ascii=False),flush=True)
    known={s['path'] for s in json.loads((ROOT/'workspace_sources.json').read_text(encoding='utf-8'))}
    compiled=json.loads((ROOT/'compiled_fashion_actions.json').read_text(encoding='utf-8'))
    extra=[]
    for path in sorted({a['filePath'] for a in compiled}-known):
        raw=base64.b64decode(get(WORKSPACE+':readFile',{'path':path})['fileContents'])
        item={'path':path,'content':raw.decode('utf-8'),'sha256':hashlib.sha256(raw).hexdigest(),'fetched_at':datetime.now(timezone.utc).isoformat()}
        extra.append(item)
        dest=ROOT/'sources'/path
        dest.parent.mkdir(parents=True,exist_ok=True)
        dest.write_text(item['content'],encoding='utf-8')
    save('additional_sources.json',extra)
    print('additional_sources',len(extra),flush=True)

if __name__=='__main__':
    p=argparse.ArgumentParser(); p.add_argument('mode',choices=['collect','tables','runtime','query','trigger_trace']); p.add_argument('--sql'); args=p.parse_args()
    if args.mode=='collect': collect()
    elif args.mode=='tables': tables()
    elif args.mode=='runtime': runtime()
    elif args.mode=='trigger_trace': trigger_trace()
    else: query_file(args.sql)
