# -*- coding: utf-8 -*-
"""
Collart Android DAU 30天趋势诊断
修正版 - is_new 为 BOOL 类型
"""
import sys, os
os.environ['PYTHONIOENCODING'] = 'utf-8'

from google.cloud import bigquery
from google.oauth2 import service_account

KEY_PATH = "knowledge/service-accounts/aidata2025.json"
credentials = service_account.Credentials.from_service_account_file(KEY_PATH)
client = bigquery.Client(project="aidata2025", credentials=credentials)

# ============ 查询1：过去30天 DAU 趋势 ============
print("=" * 60)
print("查询1：过去30天 DAU + 新用户趋势")
print("=" * 60)

query1 = """
SELECT
  event_date,
  COUNT(DISTINCT user_pseudo_id) AS dau,
  COUNT(DISTINCT CASE WHEN is_new THEN user_pseudo_id END) AS new_users,
  COUNT(DISTINCT CASE WHEN NOT is_new THEN user_pseudo_id END) AS old_users,
  ROUND(SAFE_DIVIDE(COUNT(DISTINCT CASE WHEN is_new THEN user_pseudo_id END),
    COUNT(DISTINCT user_pseudo_id)) * 100, 1) AS new_user_pct
FROM `aidata2025.dwd.dwd_oper_user_collart_di`
WHERE event_date >= DATE_SUB(CURRENT_DATE(), INTERVAL 30 DAY)
  AND event_date < CURRENT_DATE()
  AND package_name = 'free.ai.photo.generator.collart.ai'
GROUP BY event_date
ORDER BY event_date
"""

try:
    rows = list(client.query(query1).result())
    print("{:<12} {:>8} {:>8} {:>8} {:>8}".format("日期", "DAU", "新用户", "老用户", "新占比%"))
    print("-" * 50)
    dau_values, new_values = [], []
    for row in rows:
        print("{:<12} {:>8,} {:>8,} {:>8,} {:>8.1f}".format(
            str(row.event_date), row.dau, row.new_users, row.old_users, row.new_user_pct))
        dau_values.append(row.dau)
        new_values.append(row.new_users)
    if dau_values:
        print("\nDAU 均值: {:,.0f}".format(sum(dau_values)/len(dau_values)))
        print("DAU 范围: {:,} ~ {:,}".format(min(dau_values), max(dau_values)))
        print("新用户均值: {:,.0f}".format(sum(new_values)/len(new_values)))
except Exception as e:
    print("查询1失败: {}".format(e))

# ============ 查询2：过去90天周度趋势 ============
print()
print("=" * 60)
print("查询2：过去90天周度 DAU（周五数据对比）")
print("=" * 60)

query2 = """
SELECT
  event_date,
  COUNT(DISTINCT user_pseudo_id) AS dau,
  COUNT(DISTINCT CASE WHEN is_new THEN user_pseudo_id END) AS new_users
FROM `aidata2025.dwd.dwd_oper_user_collart_di`
WHERE event_date >= DATE_SUB(CURRENT_DATE(), INTERVAL 90 DAY)
  AND event_date < CURRENT_DATE()
  AND package_name = 'free.ai.photo.generator.collart.ai'
  AND EXTRACT(DAYOFWEEK FROM event_date) = 6
GROUP BY event_date
ORDER BY event_date
"""

try:
    rows = list(client.query(query2).result())
    print("{:<16} {:>8} {:>8} {:>8}".format("日期(周五)", "DAU", "新用户", "趋势"))
    print("-" * 45)
    prev = None
    for row in rows:
        trend = "-"
        if prev is not None:
            pct = (row.dau - prev) * 100.0 / prev
            trend = "{:+.1f}%".format(pct)
        print("{:<16} {:>8,} {:>8,} {:>8}".format(str(row.event_date), row.dau, row.new_users, trend))
        prev = row.dau
except Exception as e:
    print("查询2失败: {}".format(e))

# ============ 查询3：D1 留存率趋势 ============
print()
print("=" * 60)
print("查询3：过去30天 D1 留存率趋势")
print("=" * 60)

query3 = """
WITH new_users AS (
  SELECT DISTINCT
    event_date AS install_date,
    user_pseudo_id
  FROM `aidata2025.dwd.dwd_oper_user_collart_di`
  WHERE event_date >= DATE_SUB(CURRENT_DATE(), INTERVAL 30 DAY)
    AND event_date < DATE_SUB(CURRENT_DATE(), INTERVAL 1 DAY)
    AND package_name = 'free.ai.photo.generator.collart.ai'
    AND is_new
),
d1_active AS (
  SELECT DISTINCT
    DATE_SUB(event_date, INTERVAL 1 DAY) AS install_date,
    user_pseudo_id
  FROM `aidata2025.dwd.dwd_oper_user_collart_di`
  WHERE event_date >= DATE_SUB(CURRENT_DATE(), INTERVAL 29 DAY)
    AND event_date < CURRENT_DATE()
    AND package_name = 'free.ai.photo.generator.collart.ai'
)
SELECT
  n.install_date,
  COUNT(DISTINCT n.user_pseudo_id) AS new_users,
  COUNT(DISTINCT d1.user_pseudo_id) AS d1_returned,
  ROUND(COUNT(DISTINCT d1.user_pseudo_id) * 100.0 / COUNT(DISTINCT n.user_pseudo_id), 1) AS d1_retention_pct
FROM new_users n
LEFT JOIN d1_active d1 ON n.install_date = d1.install_date AND n.user_pseudo_id = d1.user_pseudo_id
GROUP BY n.install_date
ORDER BY n.install_date
"""

try:
    rows = list(client.query(query3).result())
    print("{:<12} {:>10} {:>10} {:>10}".format("安装日期", "新增用户", "D1回访", "D1留存%"))
    print("-" * 50)
    ret_values = []
    for row in rows:
        print("{:<12} {:>10,} {:>10,} {:>10.1f}".format(
            str(row.install_date), row.new_users, row.d1_returned, row.d1_retention_pct))
        ret_values.append(row.d1_retention_pct)
    if ret_values:
        print("\nD1 留存率均值: {:.1f}%".format(sum(ret_values)/len(ret_values)))
        print("D1 留存率范围: {:.1f}% ~ {:.1f}%".format(min(ret_values), max(ret_values)))
except Exception as e:
    print("查询3失败: {}".format(e))
