WITH raw AS (
  SELECT *
  FROM `pubdata2025.dwd.dwd_cdct_revenue_stripe_di`
  WHERE event_date BETWEEN '2026-08-28' AND '2026-09-10' AND app_name='collart_web'
), dedup AS (
  SELECT * FROM raw
  QUALIFY ROW_NUMBER() OVER (PARTITION BY transaction_id ORDER BY event_timestamp DESC,revneue DESC)=1
), external_orders AS (
  SELECT *, IF(event_date>='2026-09-04','current','previous') AS period,
    CASE description_type WHEN 'credit' THEN 'credit' WHEN 'Subscription creation' THEN 'new_subscription'
      WHEN 'Subscription update' THEN 'renewal' ELSE CONCAT('other:',COALESCE(description_type,'NULL')) END AS revenue_type
  FROM dedup
  WHERE NOT EXISTS(SELECT 1 FROM UNNEST(@internal_user_ids) i WHERE UPPER(i)=UPPER(dedup.user_id))
)
SELECT period, dim.dimension, dim.value,
  COUNT(*) AS transactions, COUNT(DISTINCT user_id) AS payers,
  CAST(SUM(revneue) AS FLOAT64) AS net_revenue,
  CAST(SUM(IF(UPPER(currency)='USD',amount,revneue+IFNULL(fee,0))) AS FLOAT64) AS gross_usd,
  CAST(MAX(revneue) AS FLOAT64) AS largest_transaction,
  COUNTIF(revneue>=100) AS high_value_transactions,
  CAST(SUM(IF(revneue>=100,revneue,0)) AS FLOAT64) AS high_value_revenue,
  COUNTIF(user_id IS NULL) AS missing_user,
  CAST(SUM(IF(revneue<0,revneue,0)) AS FLOAT64) AS negative_revenue
FROM external_orders
CROSS JOIN UNNEST([
  STRUCT('total' AS dimension,'all' AS value),
  STRUCT('date',CAST(event_date AS STRING)),
  STRUCT('type',revenue_type),
  STRUCT('country',COALESCE(country,'unknown')),
  STRUCT('product',COALESCE(product_name,'unknown')),
  STRUCT('date_type',CONCAT(CAST(event_date AS STRING),'|',revenue_type))
]) dim
GROUP BY 1,2,3
ORDER BY dimension,period,value;
