-- 过去20天：新用户、活跃用户、explore_page_show UV
-- 日期: 2026-04-09

WITH 
-- 新用户
new_users AS (
  SELECT 
    PARSE_DATE('%Y%m%d', event_date) AS stats_date,
    COUNT(DISTINCT user_pseudo_id) AS new_users
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 21 DAY))
    AND app_info.id IS NULL
    AND event_name = 'first_visit'
  GROUP BY stats_date
),
-- 活跃用户
active_users AS (
  SELECT 
    PARSE_DATE('%Y%m%d', event_date) AS stats_date,
    COUNT(DISTINCT user_pseudo_id) AS active_users
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 21 DAY))
    AND app_info.id IS NULL
    AND event_name = 'session_start'
  GROUP BY stats_date
),
-- explore_page_show UV
explore_users AS (
  SELECT 
    PARSE_DATE('%Y%m%d', event_date) AS stats_date,
    COUNT(DISTINCT user_pseudo_id) AS explore_uv
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 21 DAY))
    AND app_info.id IS NULL
    AND event_name = 'explore_page_show'
  GROUP BY stats_date
)
SELECT 
  COALESCE(n.stats_date, a.stats_date, e.stats_date) AS date,
  IFNULL(n.new_users, 0) AS new_users,
  IFNULL(a.active_users, 0) AS active_users,
  IFNULL(e.explore_uv, 0) AS explore_uv
FROM new_users n
FULL OUTER JOIN active_users a USING (stats_date)
FULL OUTER JOIN explore_users e USING (stats_date)
WHERE COALESCE(n.stats_date, a.stats_date, e.stats_date) IS NOT NULL
ORDER BY date DESC
