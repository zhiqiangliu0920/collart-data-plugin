#!/usr/bin/env python3
"""Check ALL param keys across the entire dataset for template-related keys"""
import os
from google.cloud import bigquery
from google.oauth2 import service_account

os.environ['PYTHONIOENCODING'] = 'utf-8'
KEY_PATH = r"{LOCAL_PATH}"

creds = service_account.Credentials.from_service_account_file(KEY_PATH, scopes=["https://www.googleapis.com/auth/cloud-platform"])
client = bigquery.Client(credentials=creds, project="vidart-8b8ca")

# 1. ALL unique param keys in entire dataset (last 90 days) - search for template-related
q1 = """
SELECT p.key, COUNT(*) as cnt
FROM `vidart-8b8ca.analytics_528583115.events_*`, UNNEST(event_params) p
WHERE _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 90 DAY))
GROUP BY p.key
ORDER BY cnt DESC
"""
print("=== ALL Param Keys in Dataset (Top 100) ===")
rows = list(client.query(q1).result())
for r in rows[:100]:
    print(f"  {r.key:40} count={r.cnt}")

print(f"\n--- Template-related keys filtered ---")
tmpl_keys = [r for r in rows if 'template' in r.key.lower() or 'temp_' in r.key.lower() or 'material' in r.key.lower()]
if tmpl_keys:
    for r in tmpl_keys:
        print(f"  >>> {r.key:40} count={r.cnt}")
else:
    print("  (no keys with 'template' or 'material' in name)")

# 2. Check if 'material_id' could be template-related
print("\n=== material_id param values (sample) ===")
q2 = """
WITH vip_users AS (
  SELECT DISTINCT user_pseudo_id
  FROM `vidart-8b8ca.analytics_528583115.events_*`
  WHERE event_name = 'vip_subscribe_succeed'
    AND app_info.install_source <> 'manual_install'
    AND _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 90 DAY))
)
SELECT
  e.event_name,
  (SELECT p.value.string_value FROM UNNEST(e.event_params) p WHERE p.key = 'material_id') as material_id,
  COUNT(*) as cnt
FROM `vidart-8b8ca.analytics_528583115.events_*` e
JOIN vip_users v ON e.user_pseudo_id = v.user_pseudo_id
WHERE e._TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 90 DAY))
  AND (SELECT p.value.string_value FROM UNNEST(e.event_params) p WHERE p.key = 'material_id') IS NOT NULL
GROUP BY e.event_name, material_id
ORDER BY cnt DESC
LIMIT 30
"""
rows = list(client.query(q2).result())
if rows:
    for r in rows:
        print(f"  event={r.event_name:35} material_id={r.material_id:30} cnt={r.cnt}")
else:
    print("  (no material_id values found)")

# 3. Check ALL event_names that contain 'template' 
print("\n=== Event names containing 'template' ===")
q3 = """
SELECT event_name, COUNT(*) as cnt
FROM `vidart-8b8ca.analytics_528583115.events_*`
WHERE _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 90 DAY))
  AND LOWER(event_name) LIKE '%template%'
GROUP BY event_name
ORDER BY cnt DESC
"""
rows = list(client.query(q3).result())
if rows:
    for r in rows:
        print(f"  event_name={r.event_name} count={r.cnt}")
else:
    print("  (no event names with 'template'")

# 4. Check ALL event_names that contain 'select', 'choose', 'apply', 'use'
print("\n=== Event names with select/choose/apply/use ===")
q4 = """
SELECT event_name, COUNT(*) as cnt
FROM `vidart-8b8ca.analytics_528583115.events_*`
WHERE _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 90 DAY))
  AND (LOWER(event_name) LIKE '%select%' OR LOWER(event_name) LIKE '%choose%' OR LOWER(event_name) LIKE '%apply%' OR LOWER(event_name) LIKE '%use%')
GROUP BY event_name
ORDER BY cnt DESC
"""
rows = list(client.query(q4).result())
for r in rows:
    print(f"  event_name={r.event_name:50} count={r.cnt}")

# 5. Check any event param values that look like template IDs (UUID-like strings)
print("\n=== Sample param values for 'material_id' across ALL users ===")
q5 = """
SELECT
  (SELECT p.value.string_value FROM UNNEST(event_params) p WHERE p.key = 'material_id') as material_id,
  COUNT(DISTINCT user_pseudo_id) as users,
  COUNT(*) as cnt
FROM `vidart-8b8ca.analytics_528583115.events_*`
WHERE _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 90 DAY))
  AND (SELECT p.value.string_value FROM UNNEST(event_params) p WHERE p.key = 'material_id') IS NOT NULL
GROUP BY material_id
ORDER BY cnt DESC
LIMIT 20
"""
rows = list(client.query(q5).result())
for r in rows:
    print(f"  material_id={r.material_id:40} users={r.users:4} events={r.cnt:4}")

print("\n=== Done ===")
