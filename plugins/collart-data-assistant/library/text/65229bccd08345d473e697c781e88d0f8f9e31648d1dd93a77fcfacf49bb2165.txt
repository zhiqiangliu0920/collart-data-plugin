-- Collart Android ai_service_start 事件 function_type 分布
-- 时间范围：过去 7 天
-- 输出：日期、function_type、uv、pv

SELECT
  PARSE_DATE('%Y%m%d', event_date) AS stats_date,
  (SELECT value.string_value FROM UNNEST(event_params) WHERE key = 'function_type') AS function_type,
  COUNT(*) AS pv,
  COUNT(DISTINCT user_pseudo_id) AS uv
FROM `storytemplate-10a27.analytics_232977577.events_*`
WHERE _TABLE_SUFFIX BETWEEN FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 7 DAY))
  AND FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 1 DAY))
  AND app_info.id = 'free.ai.photo.generator.collart.ai'
  AND event_name = 'ai_service_start'
GROUP BY stats_date, function_type
ORDER BY stats_date DESC, pv DESC
