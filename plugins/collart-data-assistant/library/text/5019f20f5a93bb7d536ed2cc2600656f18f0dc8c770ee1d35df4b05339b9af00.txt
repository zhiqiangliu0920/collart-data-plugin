SELECT job_id, parent_job_id, creation_time, start_time, end_time, statement_type, state,
  total_bytes_processed, total_bytes_billed, total_slot_ms, cache_hit, error_result,
  labels, referenced_tables,
  REGEXP_EXTRACT(query, r'(?i)(?:INTO|MERGE|FROM)\s+`(aidata2025\.(?:ads_collartfashion|ads_collart|dm)\.[^`]+)`') AS first_target,
  SUBSTR(query, 1, 450) AS query_start
FROM `aidata2025.region-us.INFORMATION_SCHEMA.JOBS_BY_PROJECT`
WHERE creation_time >= TIMESTAMP('2026-09-06 00:00:00', 'Asia/Shanghai')
  AND creation_time < TIMESTAMP('2026-09-10 00:00:00', 'Asia/Shanghai')
  AND REGEXP_CONTAINS(LOWER(query), r'collart_fashion|ads_collartfashion|dm_collart_web_user_event_di')
  AND NOT EXISTS (SELECT 1 FROM UNNEST(labels) l WHERE l.key='purpose' AND l.value='fashion-readonly-audit')
ORDER BY creation_time DESC
