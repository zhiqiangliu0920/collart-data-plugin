CREATE TABLE IF NOT EXISTS `aidata2025.ads_collart.ads_oper_user_revenue_collart_fashion_di` (
  event_date                 DATE      NOT NULL OPTIONS(description = "Stripe 收入日期"),
  package_name               STRING    NOT NULL OPTIONS(description = "collart_fashion"),
  user_id                    STRING    NOT NULL OPTIONS(description = "Collart 登录用户 ID"),
  user_pseudo_id             STRING             OPTIONS(description = "匹配支付成功事件的 GA4 用户 ID"),
  first_fashion_succeed_date DATE      NOT NULL OPTIONS(description = "首次可映射的 Fashion vip_subscribe_succeed 日期"),
  revenue_usd                FLOAT64            OPTIONS(description = "Stripe 用户当日收入 USD"),
  pay_cnt                    INT64              OPTIONS(description = "Stripe 去重交易笔数"),
  etl_time                   TIMESTAMP          OPTIONS(description = "入库时间")
)
PARTITION BY event_date
CLUSTER BY user_id
OPTIONS (
  description = "Fashion 用户日 Stripe 收入；每笔 Stripe 交易匹配前后 10 分钟内最近的 Fashion 支付成功事件",
  require_partition_filter = TRUE
);
