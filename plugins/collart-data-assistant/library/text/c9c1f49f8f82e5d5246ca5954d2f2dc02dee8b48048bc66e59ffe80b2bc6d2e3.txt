WITH aiedit_base AS (
  SELECT COUNT(DISTINCT user_pseudo_id) AS total_uv
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX >= '20260306' AND _TABLE_SUFFIX <= '20260312'
    AND app_info.id = 'free.ai.photo.generator.collart.ai'
    AND event_name = 'aiedit_show'
),
step_funnel AS (
  SELECT 
    event_name,
    COUNT(DISTINCT user_pseudo_id) AS uv
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX >= '20260306' AND _TABLE_SUFFIX <= '20260312'
    AND app_info.id = 'free.ai.photo.generator.collart.ai'
    AND event_name IN ('aiedit_show', 'aiedit_retake', 'aiedit_faceretouch', 
                       'aiedit_improve', 'aiedit_remove', 'aiedit_save_success')
  GROUP BY event_name
)
SELECT 
  s.event_name,
  s.uv,
  b.total_uv AS denominator,
  ROUND(s.uv * 100.0 / b.total_uv, 2) AS conversion_rate_pct
FROM step_funnel s
CROSS JOIN aiedit_base b
ORDER BY s.uv DESC
