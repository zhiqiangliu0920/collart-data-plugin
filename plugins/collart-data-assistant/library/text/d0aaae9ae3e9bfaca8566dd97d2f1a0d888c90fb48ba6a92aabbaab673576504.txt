-- 每天触发 ai_service_start 的次数（Android 端）
-- 日期范围：最近 30 天
SELECT
  PARSE_DATE('%Y%m%d', event_date) AS stats_date,
  COUNT(*) AS event_count,
  COUNT(DISTINCT user_pseudo_id) AS user_count
FROM `storytemplate-10a27.analytics_232977577.events_*`
WHERE _TABLE_SUFFIX BETWEEN FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 30 DAY))
  AND FORMAT_DATE('%Y%m%d', CURRENT_DATE())
  AND app_info.id = 'free.ai.photo.generator.collart.ai'
  AND event_name = 'ai_service_start'
GROUP BY stats_date
ORDER BY stats_date DESC
