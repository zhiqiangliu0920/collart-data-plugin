-- Collart Web: /soft-allure 访问与付费交集诊断
-- 查询日期: 2026-06-26

DECLARE start_date DATE DEFAULT DATE '2026-06-19';
DECLARE end_date DATE DEFAULT DATE '2026-06-25';

WITH base AS (
  SELECT
    PARSE_DATE('%Y%m%d', event_date) AS event_dt,
    user_pseudo_id,
    user_id,
    event_name,
    (
      SELECT ep.value.string_value
      FROM UNNEST(event_params) ep
      WHERE ep.key = 'page_location'
    ) AS page_location
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN FORMAT_DATE('%Y%m%d', start_date)
                          AND FORMAT_DATE('%Y%m%d', end_date)
    AND app_info.id IS NULL
    AND (event_name = 'page_view' OR user_id IS NOT NULL)
),

page_exact AS (
  SELECT *
  FROM base
  WHERE event_name = 'page_view'
    AND REGEXP_CONTAINS(
      LOWER(COALESCE(page_location, '')),
      r'(^https?://([^/]+\.)?collart\.ai)?/soft-allure([/?#]|$)'
    )
),

page_soft_like AS (
  SELECT *
  FROM base
  WHERE event_name = 'page_view'
    AND LOWER(COALESCE(page_location, '')) LIKE '%soft%'
),

ids AS (
  SELECT DISTINCT
    user_pseudo_id,
    user_id
  FROM base
  WHERE user_pseudo_id IS NOT NULL
    AND user_id IS NOT NULL
),

paid AS (
  SELECT
    user_id,
    COUNT(*) AS paid_order_cnt,
    SUM(total_price) AS total_price_raw
  FROM `aidata2025.ods.orders`
  WHERE DATE(created_at) BETWEEN start_date AND end_date
    AND status IN ('active', 'paid', 'active_ending')
  GROUP BY user_id
)

SELECT
  '01_page_exact' AS metric,
  COUNT(*) AS row_cnt,
  COUNT(DISTINCT user_pseudo_id) AS pseudo_users,
  COUNT(DISTINCT user_id) AS direct_user_ids
FROM page_exact

UNION ALL
SELECT
  '02_page_soft_like' AS metric,
  COUNT(*) AS row_cnt,
  COUNT(DISTINCT user_pseudo_id) AS pseudo_users,
  COUNT(DISTINCT user_id) AS direct_user_ids
FROM page_soft_like

UNION ALL
SELECT
  '03_page_exact_with_identity' AS metric,
  COUNT(*) AS row_cnt,
  COUNT(DISTINCT page_exact.user_pseudo_id) AS pseudo_users,
  COUNT(DISTINCT ids.user_id) AS direct_user_ids
FROM page_exact
LEFT JOIN ids
  USING (user_pseudo_id)

UNION ALL
SELECT
  '04_paid_orders' AS metric,
  SUM(paid_order_cnt) AS row_cnt,
  COUNT(DISTINCT user_id) AS pseudo_users,
  NULL AS direct_user_ids
FROM paid

UNION ALL
SELECT
  '05_overlap' AS metric,
  COUNT(*) AS row_cnt,
  COUNT(DISTINCT page_exact.user_pseudo_id) AS pseudo_users,
  COUNT(DISTINCT paid.user_id) AS direct_user_ids
FROM page_exact
JOIN ids
  USING (user_pseudo_id)
JOIN paid
  ON ids.user_id = paid.user_id

UNION ALL
SELECT
  '06_paid_users_seen_in_ga4' AS metric,
  COUNT(*) AS row_cnt,
  COUNT(DISTINCT paid.user_id) AS pseudo_users,
  COUNT(DISTINCT ids.user_id) AS direct_user_ids
FROM paid
LEFT JOIN ids
  ON paid.user_id = ids.user_id;
