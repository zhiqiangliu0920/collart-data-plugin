-- Collart Web: 周末 fashion_video_generate_click 用户中真实使用 fashion_video 的人数
-- 查询日期: 2026-06-29
-- 分母: 2026-06-27 ~ 2026-06-28 触发过 fashion_video_generate_click 的 user_pseudo_id
-- 真实使用口径: ai_service_start 且 event_params.service_type = 'fashion_video'
-- 新用户口径: aidata2025.ads_collart.ads_oper_user_new_collartweb_di 命中首次点击当天 user_pseudo_id
-- 内部用户: 可映射到 user_id 时按 collart_web/sql_presets/internal_user_filter.sql 排除

DECLARE start_date DATE DEFAULT DATE '2026-06-27';
DECLARE end_date DATE DEFAULT DATE '2026-06-28';

WITH internal_users AS (
  SELECT user_id
  FROM UNNEST([
    '{INTERNAL_USER_ID_15}',
    '{INTERNAL_USER_ID_14}',
    '{INTERNAL_USER_ID_2}',
    '{INTERNAL_USER_ID_11}',
    '{INTERNAL_USER_ID_7}',
    '{INTERNAL_USER_ID_4}',
    '{INTERNAL_USER_ID_18}',
    '{INTERNAL_USER_ID_1}',
    '{INTERNAL_USER_ID_8}',
    '{INTERNAL_USER_ID_9}',
    '{INTERNAL_USER_ID_3}',
    '{INTERNAL_USER_ID_17}',
    '{INTERNAL_USER_ID_13}',
    '{INTERNAL_USER_ID_6}',
    '{INTERNAL_USER_ID_5}',
    '{INTERNAL_USER_ID_16}',
    '{INTERNAL_USER_ID_12}',
    '{INTERNAL_USER_ID_10}'
  ]) AS user_id
),

event_base AS (
  SELECT
    PARSE_DATE('%Y%m%d', event_date) AS event_dt,
    DATE(DATETIME(TIMESTAMP_MICROS(event_timestamp), 'Asia/Shanghai')) AS event_date_cn,
    DATETIME(TIMESTAMP_MICROS(event_timestamp), 'Asia/Shanghai') AS event_time_cn,
    event_timestamp,
    user_pseudo_id,
    user_id,
    event_name,
    geo.country AS country,
    device.category AS device_category,
    traffic_source.source AS source,
    traffic_source.medium AS medium,
    traffic_source.name AS campaign,
    (
      SELECT ep.value.string_value
      FROM UNNEST(event_params) ep
      WHERE ep.key = 'page_location'
    ) AS page_location,
    (
      SELECT ep.value.string_value
      FROM UNNEST(event_params) ep
      WHERE ep.key = 'service_type'
    ) AS service_type
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN FORMAT_DATE('%Y%m%d', start_date)
                          AND FORMAT_DATE('%Y%m%d', end_date)
    AND app_info.id IS NULL
),

click_users AS (
  SELECT
    user_pseudo_id,
    MIN(event_dt) AS first_click_dt,
    MIN(event_time_cn) AS first_click_time_cn,
    MAX(event_time_cn) AS last_click_time_cn,
    COUNT(*) AS click_cnt
  FROM event_base
  WHERE event_name = 'fashion_video_generate_click'
    AND event_date_cn BETWEEN start_date AND end_date
  GROUP BY user_pseudo_id
),

identity_map AS (
  SELECT
    user_pseudo_id,
    ARRAY_AGG(user_id IGNORE NULLS ORDER BY event_timestamp DESC LIMIT 1)[SAFE_OFFSET(0)] AS user_id
  FROM event_base
  WHERE user_id IS NOT NULL
    AND user_id != ''
  GROUP BY user_pseudo_id
),

real_use AS (
  SELECT
    user_pseudo_id,
    MIN(event_time_cn) AS first_real_use_time_cn,
    COUNT(*) AS real_use_cnt
  FROM event_base
  WHERE event_name = 'ai_service_start'
    AND LOWER(COALESCE(service_type, '')) = 'fashion_video'
    AND event_date_cn BETWEEN start_date AND end_date
  GROUP BY user_pseudo_id
),

first_context AS (
  SELECT
    e.user_pseudo_id,
    ARRAY_AGG(e.country IGNORE NULLS ORDER BY e.event_timestamp LIMIT 1)[SAFE_OFFSET(0)] AS country,
    ARRAY_AGG(e.device_category IGNORE NULLS ORDER BY e.event_timestamp LIMIT 1)[SAFE_OFFSET(0)] AS device_category,
    ARRAY_AGG(e.source IGNORE NULLS ORDER BY e.event_timestamp LIMIT 1)[SAFE_OFFSET(0)] AS source,
    ARRAY_AGG(e.medium IGNORE NULLS ORDER BY e.event_timestamp LIMIT 1)[SAFE_OFFSET(0)] AS medium,
    ARRAY_AGG(e.campaign IGNORE NULLS ORDER BY e.event_timestamp LIMIT 1)[SAFE_OFFSET(0)] AS campaign,
    ARRAY_AGG(e.page_location IGNORE NULLS ORDER BY e.event_timestamp LIMIT 1)[SAFE_OFFSET(0)] AS first_page_location
  FROM event_base e
  JOIN click_users c
    USING (user_pseudo_id)
  GROUP BY e.user_pseudo_id
),

new_user AS (
  SELECT DISTINCT
    event_date,
    user_pseudo_id
  FROM `aidata2025.ads_collart.ads_oper_user_new_collartweb_di`
  WHERE event_date BETWEEN start_date AND end_date
),

classified AS (
  SELECT
    c.user_pseudo_id,
    im.user_id,
    c.first_click_dt,
    c.first_click_time_cn,
    c.last_click_time_cn,
    c.click_cnt,
    IF(r.user_pseudo_id IS NOT NULL, TRUE, FALSE) AS has_real_use,
    r.first_real_use_time_cn,
    IFNULL(r.real_use_cnt, 0) AS real_use_cnt,
    IF(nu.user_pseudo_id IS NOT NULL, 'new', 'old') AS new_old_type,
    fc.country,
    fc.device_category,
    fc.source,
    fc.medium,
    fc.campaign,
    fc.first_page_location
  FROM click_users c
  LEFT JOIN identity_map im
    USING (user_pseudo_id)
  LEFT JOIN real_use r
    USING (user_pseudo_id)
  LEFT JOIN first_context fc
    USING (user_pseudo_id)
  LEFT JOIN new_user nu
    ON c.first_click_dt = nu.event_date
   AND c.user_pseudo_id = nu.user_pseudo_id
  WHERE im.user_id IS NULL
     OR im.user_id NOT IN (SELECT user_id FROM internal_users)
)

SELECT *
FROM classified
ORDER BY has_real_use DESC, first_click_time_cn;
