WITH date_range AS (
  SELECT 
    DATE_SUB(CURRENT_DATE(), INTERVAL 30 DAY) AS start_date,
    CURRENT_DATE() AS end_date
),

base_users AS (
  SELECT
    user_pseudo_id,
    traffic_src_source AS channel,
    traffic_src_type AS channel_type,
    country,
    is_vip,
    is_login
  FROM `aidata2025.ads_collart.ads_oper_user_new_collartweb_di`
  WHERE event_date BETWEEN (SELECT start_date FROM date_range) AND (SELECT end_date FROM date_range)
),

all_events AS (
  SELECT
    user_pseudo_id,
    event_name,
    PARSE_DATE("%Y%m%d", event_date) AS event_dt
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN 
    FORMAT_DATE("%Y%m%d", (SELECT start_date FROM date_range))
    AND FORMAT_DATE("%Y%m%d", (SELECT end_date FROM date_range))
    AND app_info.id IS NULL
),

metrics AS (
  SELECT
    bu.channel,
    bu.channel_type,
    COUNT(DISTINCT bu.user_pseudo_id) AS new_users,
    
    COUNT(DISTINCT CASE WHEN ae.event_name = "first_visit" THEN ae.user_pseudo_id END) AS visit_users,
    COUNT(DISTINCT CASE WHEN ae.event_name = "explore_content_generate_click" THEN ae.user_pseudo_id END) AS click_gen_users,
    COUNT(DISTINCT CASE WHEN ae.event_name = "login_success" THEN ae.user_pseudo_id END) AS login_success_users,
    
    COUNT(DISTINCT CASE WHEN ae.event_name = "img2video_generate_click" THEN ae.user_pseudo_id END) AS img2video_click,
    COUNT(DISTINCT CASE WHEN ae.event_name = "text2video_generate_click" THEN ae.user_pseudo_id END) AS text2video_click,
    COUNT(DISTINCT CASE WHEN ae.event_name = "img2img_generate_click" THEN ae.user_pseudo_id END) AS img2img_click,
    COUNT(DISTINCT CASE WHEN ae.event_name = "text2img_generate_click" THEN ae.user_pseudo_id END) AS text2img_click,
    COUNT(DISTINCT CASE WHEN ae.event_name = "ai_service_success" THEN ae.user_pseudo_id END) AS gen_success_users,
    COUNT(DISTINCT CASE WHEN ae.event_name = "ai_result_download" THEN ae.user_pseudo_id END) AS download_users,
    
    COUNT(DISTINCT CASE WHEN ae.event_name = "vip_show" THEN ae.user_pseudo_id END) AS vip_show_users,
    COUNT(DISTINCT CASE WHEN ae.event_name = "vip_subscribe_succeed" THEN ae.user_pseudo_id END) AS vip_success_users,
    COUNT(DISTINCT CASE WHEN ae.event_name = "credit_pack_purchase" THEN ae.user_pseudo_id END) AS credit_purchase_users
    
  FROM base_users bu
  LEFT JOIN all_events ae ON bu.user_pseudo_id = ae.user_pseudo_id
  GROUP BY bu.channel, bu.channel_type
)

SELECT
  COALESCE(channel, "unknown") AS channel,
  COALESCE(channel_type, "unknown") AS channel_type,
  new_users,
  visit_users,
  login_success_users,
  ROUND(100.0 * login_success_users / NULLIF(visit_users, 0), 2) AS login_conv_rate,
  img2video_click + text2video_click + img2img_click + text2img_click AS total_gen_clicks,
  gen_success_users,
  ROUND(100.0 * gen_success_users / NULLIF(img2video_click + text2video_click + img2img_click + text2img_click, 0), 2) AS gen_success_rate,
  download_users,
  ROUND(100.0 * download_users / NULLIF(gen_success_users, 0), 2) AS download_rate,
  vip_success_users,
  credit_purchase_users,
  ROUND(100.0 * vip_success_users / NULLIF(vip_show_users, 0), 2) AS pay_conv_rate
FROM metrics
ORDER BY new_users DESC
LIMIT 20
