-- Collart Web: 过去一周进入 /soft-allure 且同窗口有成功付费的用户
-- 查询日期: 2026-06-26
-- 统计窗口: 2026-06-19 ~ 2026-06-25（最近 7 个完整自然日，北京时间）
-- 页面口径: GA4 page_view 的 page_location 命中 https://collart.ai/soft-allure（兼容 www、query、hash、尾斜杠）
-- 付费口径: aidata2025.ods.orders 成功状态 status IN ('active','paid','active_ending')

DECLARE start_date DATE DEFAULT DATE '2026-06-19';
DECLARE end_date DATE DEFAULT DATE '2026-06-25';

WITH event_base AS (
  SELECT
    PARSE_DATE('%Y%m%d', event_date) AS event_dt,
    DATETIME(TIMESTAMP_MICROS(event_timestamp), 'Asia/Shanghai') AS event_time_cn,
    user_pseudo_id,
    user_id,
    event_name,
    (
      SELECT ep.value.string_value
      FROM UNNEST(event_params) ep
      WHERE ep.key = 'page_location'
    ) AS page_location
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN FORMAT_DATE('%Y%m%d', start_date)
                          AND FORMAT_DATE('%Y%m%d', end_date)
    AND app_info.id IS NULL
    AND (event_name = 'page_view' OR user_id IS NOT NULL)
),

soft_allure_visits AS (
  SELECT
    user_pseudo_id,
    MIN(event_time_cn) AS first_soft_allure_visit_time_cn,
    MAX(event_time_cn) AS last_soft_allure_visit_time_cn,
    COUNT(*) AS soft_allure_pv,
    ARRAY_AGG(DISTINCT event_dt IGNORE NULLS) AS soft_allure_visit_dates,
    ARRAY_AGG(DISTINCT page_location IGNORE NULLS LIMIT 5) AS sample_page_locations
  FROM event_base
  WHERE event_name = 'page_view'
    AND REGEXP_CONTAINS(
      LOWER(COALESCE(page_location, '')),
      r'(^https?://([^/]+\.)?collart\.ai)?/soft-allure([/?#]|$)'
    )
  GROUP BY user_pseudo_id
),

identity_map AS (
  SELECT
    user_pseudo_id,
    user_id
  FROM event_base
  WHERE user_pseudo_id IS NOT NULL
    AND user_id IS NOT NULL
  GROUP BY ALL
),

soft_allure_users AS (
  SELECT
    im.user_id,
    v.user_pseudo_id,
    v.first_soft_allure_visit_time_cn,
    v.last_soft_allure_visit_time_cn,
    v.soft_allure_pv,
    v.soft_allure_visit_dates,
    v.sample_page_locations
  FROM soft_allure_visits v
  JOIN identity_map im
    USING (user_pseudo_id)
),

soft_allure_user_rollup_base AS (
  SELECT
    user_id,
    ARRAY_AGG(DISTINCT user_pseudo_id IGNORE NULLS LIMIT 20) AS user_pseudo_ids,
    MIN(first_soft_allure_visit_time_cn) AS first_soft_allure_visit_time_cn,
    MAX(last_soft_allure_visit_time_cn) AS last_soft_allure_visit_time_cn,
    SUM(soft_allure_pv) AS soft_allure_pv,
    ARRAY_CONCAT_AGG(soft_allure_visit_dates) AS all_visit_dates,
    ARRAY_CONCAT_AGG(sample_page_locations) AS sample_page_locations
  FROM soft_allure_users
  GROUP BY user_id
),

soft_allure_user_rollup AS (
  SELECT
    user_id,
    user_pseudo_ids,
    first_soft_allure_visit_time_cn,
    last_soft_allure_visit_time_cn,
    soft_allure_pv,
    (SELECT COUNT(DISTINCT visit_date) FROM UNNEST(all_visit_dates) AS visit_date) AS soft_allure_active_days,
    sample_page_locations
  FROM soft_allure_user_rollup_base
),

paid_orders AS (
  SELECT
    o.user_id,
    COUNT(*) AS paid_order_cnt,
    MIN(DATETIME(TIMESTAMP(o.created_at), 'Asia/Shanghai')) AS first_payment_time_cn,
    MAX(DATETIME(TIMESTAMP(o.created_at), 'Asia/Shanghai')) AS last_payment_time_cn,
    ROUND(SUM(SAFE_DIVIDE(o.total_price, er.rate) / 100), 2) AS revenue_usd,
    ARRAY_AGG(DISTINCT o.product_name IGNORE NULLS LIMIT 10) AS product_names,
    ARRAY_AGG(DISTINCT o.status IGNORE NULLS LIMIT 10) AS order_statuses
  FROM `aidata2025.ods.orders` o
  LEFT JOIN `aidata2025.dim.exchange_rate_days` er
    ON DATE(o.created_at) = CAST(er.stats_date AS DATE)
   AND LOWER(o.currency) = LOWER(er.currency)
  WHERE DATE(o.created_at) BETWEEN start_date AND end_date
    AND o.status IN ('active', 'paid', 'active_ending')
  GROUP BY o.user_id
)

SELECT
  sau.user_id,
  sau.user_pseudo_ids,
  sau.first_soft_allure_visit_time_cn,
  sau.last_soft_allure_visit_time_cn,
  sau.soft_allure_pv,
  sau.soft_allure_active_days,
  po.first_payment_time_cn,
  po.last_payment_time_cn,
  po.paid_order_cnt,
  po.revenue_usd,
  po.product_names,
  po.order_statuses,
  sau.sample_page_locations
FROM soft_allure_user_rollup sau
JOIN paid_orders po
  USING (user_id)
ORDER BY po.revenue_usd DESC, po.first_payment_time_cn;
