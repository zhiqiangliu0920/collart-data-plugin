DELETE FROM `aidata2025.ads_collartfashion.ads_oper_user_active_di`

INSERT INTO `aidata2025.ads_collartfashion.ads_oper_user_active_di` (
  package_name, app_name, event_date, user_pseudo_id, user_id, country, app_version,
  is_new, is_churn, is_return, is_remove, is_login, is_active, is_vip, first_open_time,
  traffic_src_name, traffic_src_type, traffic_src_platform,
  identity, geo, device, traffic_src, subscription, revenue, retain, rolling_retain
)

WITH ev AS (
  SELECT *
  FROM `aidata2025.dm.dm_collart_web_user_event_di`
  WHERE event_date BETWEEN source_from_date AND to_date
    AND package_name = 'collart_fashion'
    AND user_pseudo_id IS NOT NULL
),

agg AS (
  SELECT
    event_date,
    user_pseudo_id,
    COUNTIF(event_name = 'first_visit') > 0 AS is_new,
    COUNTIF(event_name IN (
      'page_view', 'user_engagement', 'session_start', 'first_visit'
    )) > 0 AS is_active,
    FALSE AS is_remove,
    LOGICAL_OR(is_vip IN ('1', '2')) AS any_vip,
    LOGICAL_OR(traffic_src_type = 'delivery') AS any_delivery,
    LOGICAL_OR(event_name IN ('vip_subscribe_succeed', 'credit_pack_purchase')) AS is_purchase
  FROM ev
  GROUP BY event_date, user_pseudo_id
),

last_ev AS (
  SELECT * EXCEPT(rn)
  FROM (
    SELECT
      event_date,
      user_pseudo_id,
      user_id,
      country,
      CAST(NULL AS STRING) AS app_version,
      is_login,
      is_vip,
      first_open_time,
      traffic_src_name,
      traffic_src_type,
      CASE
        WHEN NULLIF(traffic_src_platform, '') IS NOT NULL THEN traffic_src_platform
        WHEN traffic_src_type != 'delivery' THEN traffic_src_type
        WHEN LOWER(COALESCE(traffic.source, '')) IN ('x', 'twitter')
          OR REGEXP_CONTAINS(UPPER(COALESCE(traffic_src_name, '')), r'^(X[-_]|CW_X_|TW[-_]|TWITTER)') THEN 'x_delivery'
        WHEN LOWER(COALESCE(traffic.source, '')) LIKE '%tiktok%'
          OR LOWER(COALESCE(traffic.source, '')) LIKE '%bytedance%'
          OR LOWER(REGEXP_EXTRACT(COALESCE(page.location, ''), r'[?&]utm_source=([^&#]+)')) LIKE '%tiktok%'
          OR REGEXP_CONTAINS(UPPER(COALESCE(traffic_src_name, '')), r'^(TT[-]|TIKTOK)') THEN 'tt_delivery'
        WHEN LOWER(COALESCE(traffic.medium, '')) = 'paid_social'
          OR REGEXP_CONTAINS(LOWER(COALESCE(traffic_src_name, '')), r'(^fb[-_]|meta|facebook)')
          OR UPPER(COALESCE(traffic_src_name, '')) LIKE 'IG%'
          OR LOWER(COALESCE(traffic.source, '')) LIKE '%facebook%'
          OR LOWER(COALESCE(traffic.source, '')) IN ('ig', 'instagram') THEN 'fb_delivery'
        WHEN LOWER(COALESCE(traffic.medium, '')) IN ('cpc', 'ppc', 'paid', 'paid_search', 'display', 'cpv', 'cpm')
          OR REGEXP_CONTAINS(LOWER(COALESCE(traffic_src_name, '')), r'(^ga[-_]|google.?ads)') THEN 'ga_delivery'
        WHEN LOWER(COALESCE(traffic.source, '')) LIKE '%applovin%' THEN 'applovin_delivery'
        ELSE 'other_delivery'
      END AS traffic_src_platform,
      identity,
      geo,
      device,
      traffic,
      ROW_NUMBER() OVER (
        PARTITION BY event_date, user_pseudo_id
        ORDER BY event_time DESC
      ) AS rn
    FROM ev
  )
  WHERE rn = 1
),

delivery_ev AS (
  SELECT * EXCEPT(rn)
  FROM (
    SELECT
      event_date,
      user_pseudo_id,
      traffic_src_name,
      traffic_src_type,
      CASE
        WHEN NULLIF(traffic_src_platform, '') IS NOT NULL THEN traffic_src_platform
        WHEN LOWER(COALESCE(traffic.source, '')) IN ('x', 'twitter')
          OR REGEXP_CONTAINS(UPPER(COALESCE(traffic_src_name, '')), r'^(X[-_]|CW_X_|TW[-_]|TWITTER)') THEN 'x_delivery'
        WHEN LOWER(COALESCE(traffic.source, '')) LIKE '%tiktok%'
          OR LOWER(COALESCE(traffic.source, '')) LIKE '%bytedance%'
          OR LOWER(REGEXP_EXTRACT(COALESCE(page.location, ''), r'[?&]utm_source=([^&#]+)')) LIKE '%tiktok%'
          OR REGEXP_CONTAINS(UPPER(COALESCE(traffic_src_name, '')), r'^(TT[-]|TIKTOK)') THEN 'tt_delivery'
        WHEN LOWER(COALESCE(traffic.medium, '')) = 'paid_social'
          OR REGEXP_CONTAINS(LOWER(COALESCE(traffic_src_name, '')), r'(^fb[-_]|meta|facebook)')
          OR UPPER(COALESCE(traffic_src_name, '')) LIKE 'IG%'
          OR LOWER(COALESCE(traffic.source, '')) LIKE '%facebook%'
          OR LOWER(COALESCE(traffic.source, '')) IN ('ig', 'instagram') THEN 'fb_delivery'
        WHEN LOWER(COALESCE(traffic.medium, '')) IN ('cpc', 'ppc', 'paid', 'paid_search', 'display', 'cpv', 'cpm')
          OR REGEXP_CONTAINS(LOWER(COALESCE(traffic_src_name, '')), r'(^ga[-_]|google.?ads)') THEN 'ga_delivery'
        WHEN LOWER(COALESCE(traffic.source, '')) LIKE '%applovin%' THEN 'applovin_delivery'
        ELSE 'other_delivery'
      END AS traffic_src_platform,
      traffic,
      ROW_NUMBER() OVER (
        PARTITION BY event_date, user_pseudo_id
        ORDER BY event_time DESC
      ) AS rn
    FROM ev
    WHERE traffic_src_type = 'delivery'
  )
  WHERE rn = 1
),


active_all AS (
  SELECT
    a.event_date,
    a.user_pseudo_id,
    a.is_new,
    a.is_remove,
    l.user_id,
    'collart_fashion' AS package_name,
    l.country,
    l.app_version,
    l.is_login,
    IF(a.any_vip, '1', COALESCE(l.is_vip, '0')) AS is_vip,
    l.first_open_time,
    IF(a.any_delivery, COALESCE(d.traffic_src_name, l.traffic_src_name), l.traffic_src_name) AS traffic_src_name,
    IF(a.any_delivery, 'delivery', l.traffic_src_type) AS traffic_src_type,
    IF(a.any_delivery, COALESCE(d.traffic_src_platform, l.traffic_src_platform), l.traffic_src_platform) AS traffic_src_platform,
    l.identity,
    l.geo,
    l.device,
    IF(a.any_delivery, COALESCE(d.traffic, l.traffic), l.traffic) AS traffic_src,
    a.is_purchase
  FROM agg AS a
  JOIN last_ev AS l USING (event_date, user_pseudo_id)
  LEFT JOIN delivery_ev AS d USING (event_date, user_pseudo_id)
  WHERE a.is_active
),

active AS (
  SELECT *
  FROM active_all
  WHERE event_date BETWEEN from_date AND to_date
),

revenue_user_day AS (
  SELECT event_date, user_id, user_pseudo_id, revenue_usd
  FROM `aidata2025.ads_collart.ads_oper_user_revenue_collart_fashion_di`
  WHERE event_date BETWEEN from_date AND to_date
    AND package_name = 'collart_fashion'
),

direct_revenue_target AS (
  SELECT r.event_date, r.user_id, a.user_pseudo_id
  FROM revenue_user_day AS r
  JOIN active AS a USING (event_date, user_id)
  QUALIFY ROW_NUMBER() OVER (
    PARTITION BY r.event_date, r.user_id
    ORDER BY a.user_pseudo_id = r.user_pseudo_id DESC, a.user_pseudo_id
  ) = 1
),

user_dim_candidates AS (
  SELECT
    event_date,
    user_pseudo_id,
    user_id,
    country,
    app_version,
    is_login,
    is_vip,
    first_open_time,
    traffic_src_name,
    traffic_src_type,
    traffic_src_platform,
    identity,
    geo,
    device,
    traffic_src
  FROM (
    SELECT
      event_date, user_pseudo_id, user_id, country, app_version, is_login, is_vip, first_open_time,
      traffic_src_name, traffic_src_type, traffic_src_platform, identity, geo, device, traffic_src
    FROM active
    WHERE NULLIF(user_id, '') IS NOT NULL

    UNION ALL

    SELECT
      event_date, user_pseudo_id, user_id, country, app_version, is_login, is_vip, first_open_time,
      traffic_src_name, traffic_src_type, traffic_src_platform, identity, geo,
      STRUCT(
        device.category AS device_category,
        device.mobile_brand_name,
        device.mobile_model_name,
        device.operating_system,
        device.operating_system_version AS device_operating_system_version,
        device.vendor_id AS device_vendor_id,
        device.language AS device_language,
        device.browser,
        device.browser_version
      ) AS device,
      traffic_src
    FROM `aidata2025.ads_collartfashion.ads_oper_user_active_di`
    WHERE event_date BETWEEN DATE '2026-06-01' AND from_date - 1
      AND NULLIF(user_id, '') IS NOT NULL
  )
),

historical_revenue_target AS (
  SELECT
    r.event_date,
    r.user_id,
    h.user_pseudo_id,
    h.country,
    h.app_version,
    h.is_login,
    h.is_vip,
    h.first_open_time,
    h.traffic_src_name,
    h.traffic_src_type,
    h.traffic_src_platform,
    h.identity,
    h.geo,
    h.device,
    h.traffic_src
  FROM revenue_user_day AS r
  JOIN user_dim_candidates AS h
    ON (r.user_pseudo_id = h.user_pseudo_id OR r.user_id = h.user_id)
   AND h.event_date <= DATE_ADD(r.event_date, INTERVAL 1 DAY)
  QUALIFY ROW_NUMBER() OVER (PARTITION BY r.event_date, r.user_id ORDER BY h.event_date DESC, h.user_pseudo_id) = 1
),

revenue_target AS (
  SELECT
    r.event_date,
    r.user_id,
    r.revenue_usd,
    COALESCE(r.user_pseudo_id, d.user_pseudo_id, h.user_pseudo_id) AS target_user_pseudo_id,
    h.country,
    h.app_version,
    h.is_login,
    h.is_vip,
    h.first_open_time,
    h.traffic_src_name,
    h.traffic_src_type,
    h.traffic_src_platform,
    h.identity,
    h.geo,
    h.device,
    h.traffic_src
  FROM revenue_user_day AS r
  LEFT JOIN direct_revenue_target AS d USING (event_date, user_id)
  LEFT JOIN historical_revenue_target AS h USING (event_date, user_id)
),

active_output AS (
  SELECT a.*, TRUE AS is_active
  FROM active AS a

  UNION ALL

  SELECT
    r.event_date,
    r.target_user_pseudo_id AS user_pseudo_id,
    FALSE AS is_new,
    FALSE AS is_remove,
    r.user_id,
    'collart_fashion' AS package_name,
    r.country,
    r.app_version,
    COALESCE(r.is_login, TRUE) AS is_login,
    COALESCE(r.is_vip, '0') AS is_vip,
    r.first_open_time,
    r.traffic_src_name,
    r.traffic_src_type,
    r.traffic_src_platform,
    r.identity,
    r.geo,
    r.device,
    r.traffic_src,
    FALSE AS is_purchase,
    FALSE AS is_active
  FROM revenue_target AS r
  WHERE r.target_user_pseudo_id IS NOT NULL
    AND NOT EXISTS (
      SELECT 1
      FROM active AS a
      WHERE a.event_date = r.event_date
        AND a.user_pseudo_id = r.target_user_pseudo_id
    )
),


retain_flags AS (

    event_date,
    user_pseudo_id,
    LOGICAL_OR(diff = 1) AS r1,
    LOGICAL_OR(diff = 2) AS r2,
    LOGICAL_OR(diff = 3) AS r3,
    LOGICAL_OR(diff = 4) AS r4,
    LOGICAL_OR(diff = 5) AS r5,
    LOGICAL_OR(diff = 6) AS r6,

    LOGICAL_OR(diff <= 2) AS rr2,
    LOGICAL_OR(diff <= 3) AS rr3,
    LOGICAL_OR(diff <= 4) AS rr4,
    LOGICAL_OR(diff <= 5) AS rr5,
    LOGICAL_OR(diff <= 6) AS rr6,

    LOGICAL_OR(diff <= 8) AS rr8,
    LOGICAL_OR(diff <= 9) AS rr9,

    LOGICAL_OR(diff <= 12) AS rr12,

  SELECT

      DATE_ADD(a.event_date, INTERVAL 14 DAY) > to_date,
      NULL,
      NOT EXISTS (
        SELECT 1 FROM active_all AS n
        WHERE n.user_pseudo_id = a.user_pseudo_id
          AND DATE_DIFF(n.event_date, a.event_date, DAY) BETWEEN 1 AND 14
      )
    ) AS is_churn,
    IF(
      a.is_new,
      FALSE,
      IF(
        DATE_SUB(a.event_date, INTERVAL 14 DAY) < c.min_event_date,
        NULL,
        NOT EXISTS (
          SELECT 1 FROM active_all AS p
          WHERE p.user_pseudo_id = a.user_pseudo_id
            AND DATE_DIFF(a.event_date, p.event_date, DAY) BETWEEN 1 AND 14
        )
      )
    ) AS is_return
  FROM active AS a
  CROSS JOIN coverage AS c
)

    f.traffic_src.term,

    0.0 AS credit_revenue,

  ) AS revenue,
  STRUCT(
    IF(DATE_ADD(f.event_date, INTERVAL 1 DAY) > to_date, NULL, COALESCE(rf.r1, FALSE)) AS d2,
