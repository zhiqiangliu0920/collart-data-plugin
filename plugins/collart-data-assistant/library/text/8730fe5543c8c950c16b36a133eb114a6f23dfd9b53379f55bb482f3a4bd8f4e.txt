-- Collart Android：非订阅直接买点数（novip + 行为口径）
-- 凭证 aidata2025；窗口可改

-- 1) novip 日趋势
SELECT
  order_date,
  sku_id,
  COUNT(*) AS orders,
  COUNT(DISTINCT COALESCE(user_pseudo_id, CAST(user_id AS STRING))) AS users,
  ROUND(SUM(revenue), 2) AS revenue
FROM `aidata2025.dwd.dwd_oper_sub_order_revenue_df`
WHERE package_name = 'free.ai.photo.generator.collart.ai'
  AND order_date BETWEEN DATE '2026-07-28' AND DATE '2026-08-19'
  AND (financial_status IS NULL OR financial_status = 'Charged')
  AND LOWER(sku_id) LIKE '%novip%'
GROUP BY 1, 2
ORDER BY 1, 2;

-- 2) 8 月买点路径：订前 / 先点后订 / 只买点
WITH all_o AS (
  SELECT
    COALESCE(user_pseudo_id, CAST(user_id AS STRING)) AS uid,
    order_date,
    revenue,
    order_type = '点数包' OR LOWER(COALESCE(sku_id, '')) LIKE '%credit%' AS is_credit,
    order_type IN ('周包', '年包', '月包') AS is_sub,
    LOWER(COALESCE(sku_id, '')) LIKE '%novip%' AS is_novip
  FROM `aidata2025.dwd.dwd_oper_sub_order_revenue_df`
  WHERE package_name = 'free.ai.photo.generator.collart.ai'
    AND order_date BETWEEN DATE '2025-01-01' AND DATE '2026-08-19'
    AND (financial_status IS NULL OR financial_status = 'Charged')
),
aug_credit_users AS (
  SELECT
    uid,
    MIN(order_date) AS first_c,
    ROUND(SUM(revenue), 2) AS aug_credit_rev,
    MAX(IF(is_novip, 1, 0)) AS bought_novip
  FROM all_o
  WHERE is_credit AND order_date BETWEEN DATE '2026-08-01' AND DATE '2026-08-19'
  GROUP BY 1
),
flag AS (
  SELECT
    a.*,
    MAX(IF(o.is_sub AND o.order_date < a.first_c, 1, 0)) AS had_sub_before,
    MAX(IF(o.is_sub AND o.order_date >= a.first_c, 1, 0)) AS sub_on_or_after
  FROM aug_credit_users a
  LEFT JOIN all_o o ON a.uid = o.uid
  GROUP BY ALL
)
SELECT
  CASE
    WHEN had_sub_before = 1 THEN 'had_sub_before'
    WHEN sub_on_or_after = 1 THEN 'credit_then_sub'
    ELSE 'credit_only_no_sub'
  END AS path,
  bought_novip,
  COUNT(*) AS users,
  ROUND(SUM(aug_credit_rev), 2) AS aug_credit_rev
FROM flag
GROUP BY 1, 2
ORDER BY 1, 2;
