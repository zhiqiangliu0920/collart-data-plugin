-- I. 基础运营指标 (每日趋势)
-- =============================================
-- 1. 每日 DAU + 新用户 (上线以来)
SELECT
  event_date,
  COUNT(DISTINCT CASE WHEN event_name = 'session_start' THEN user_pseudo_id END) AS dau,
  COUNT(DISTINCT CASE WHEN event_name = 'first_open' THEN user_pseudo_id END) AS new_users,
  COUNT(DISTINCT user_pseudo_id) AS active_users,
  COUNT(*) AS total_events
FROM `vibedance-540a0.analytics_533044528.events_*`
WHERE _TABLE_SUFFIX >= '20260101'
  -- 如果 Android/iOS 分开，加过滤: AND platform = 'ANDROID'
GROUP BY event_date
ORDER BY event_date;

-- 2. 新用户增长周环比 (每周一对比)
WITH weekly AS (
  SELECT
    FORMAT_DATE('%Y-%W', PARSE_DATE('%Y%m%d', _TABLE_SUFFIX)) AS week,
    COUNT(DISTINCT CASE WHEN event_name = 'first_open' THEN user_pseudo_id END) AS new_users
  FROM `vibedance-540a0.analytics_533044528.events_*`
  WHERE _TABLE_SUFFIX >= '20260101'
  GROUP BY week
)
SELECT
  week,
  new_users,
  LAG(new_users) OVER (ORDER BY week) AS prev_week_new,
  ROUND((new_users - LAG(new_users) OVER (ORDER BY week)) * 100.0
    / NULLIF(LAG(new_users) OVER (ORDER BY week), 0), 1) AS wow_change_pct
FROM weekly
ORDER BY week;

-- =============================================
-- II. 留存分析 (核心健康度指标)
-- =============================================
-- 3. D1 留存率 (按日 cohort)
-- 注意: 需要产品上线>=2天才能算
WITH new_users AS (
  SELECT DISTINCT
    event_date AS install_date,
    user_pseudo_id
  FROM `vibedance-540a0.analytics_533044528.events_*`
  WHERE event_date >= DATE_SUB(CURRENT_DATE(), INTERVAL 30 DAY)
    AND event_date < DATE_SUB(CURRENT_DATE(), INTERVAL 1 DAY)
    AND event_name = 'first_open'
),
d1_active AS (
  SELECT DISTINCT
    DATE_SUB(event_date, INTERVAL 1 DAY) AS install_date,
    user_pseudo_id
  FROM `vibedance-540a0.analytics_533044528.events_*`
  WHERE event_date >= DATE_SUB(CURRENT_DATE(), INTERVAL 29 DAY)
    AND event_date < CURRENT_DATE()
    AND event_name = 'session_start'
)
SELECT
  n.install_date,
  COUNT(DISTINCT n.user_pseudo_id) AS new_users,
  COUNT(DISTINCT d1.user_pseudo_id) AS d1_returned,
  ROUND(COUNT(DISTINCT d1.user_pseudo_id) * 100.0 / COUNT(DISTINCT n.user_pseudo_id), 1) AS d1_retention_pct
FROM new_users n
LEFT JOIN d1_active d1 ON n.install_date = d1.install_date AND n.user_pseudo_id = d1.user_pseudo_id
GROUP BY n.install_date
ORDER BY n.install_date;

-- 4. D7 / D14 / D30 留存率 (完整留存曲线)
-- 适用条件: 产品上线>=31天
WITH cohorts AS (
  SELECT DISTINCT
    event_date AS install_date,
    user_pseudo_id
  FROM `vibedance-540a0.analytics_533044528.events_*`
  WHERE event_date >= DATE_SUB(CURRENT_DATE(), INTERVAL 60 DAY)
    AND event_date < DATE_SUB(CURRENT_DATE(), INTERVAL 30 DAY)
    AND event_name = 'first_open'
),
retention AS (
  SELECT
    c.install_date,
    DATE_DIFF(d.event_date, c.install_date, DAY) AS day_diff,
    COUNT(DISTINCT d.user_pseudo_id) AS retained_users
  FROM cohorts c
  INNER JOIN `vibedance-540a0.analytics_533044528.events_*` d
    ON c.user_pseudo_id = d.user_pseudo_id
  WHERE d.event_date >= c.install_date
    AND d.event_date <= DATE_ADD(c.install_date, INTERVAL 30 DAY)
    AND d.event_name = 'session_start'
  GROUP BY c.install_date, day_diff
),
cohort_size AS (
  SELECT install_date, COUNT(*) AS total_users FROM cohorts GROUP BY install_date
)
SELECT
  r.day_diff,
  ROUND(AVG(r.retained_users * 100.0 / cs.total_users), 1) AS avg_retention_pct
FROM retention r
JOIN cohort_size cs ON r.install_date = cs.install_date
GROUP BY r.day_diff
HAVING r.day_diff IN (0, 1, 3, 7, 14, 30)
ORDER BY r.day_diff;

-- =============================================
-- IV. 用户质量与分布
-- =============================================
-- 6. 国家分布 TOP15 (看看有没有意外市场)
SELECT
  COALESCE(geo.country, 'Unknown') AS country,
  COUNT(DISTINCT user_pseudo_id) AS users,
  COUNT(*) AS events,
  ROUND(COUNT(*) / COUNT(DISTINCT user_pseudo_id), 1) AS events_per_user
FROM `vibedance-540a0.analytics_533044528.events_*`
WHERE _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 7 DAY))
GROUP BY geo.country
ORDER BY users DESC
LIMIT 15;

-- 7. 平台分布 (Android vs iOS 占比)
SELECT
  device.operating_system AS os,
  COUNT(DISTINCT user_pseudo_id) AS users,
  COUNT(*) AS events
FROM `vibedance-540a0.analytics_533044528.events_*`
WHERE _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 7 DAY))
GROUP BY device.operating_system
ORDER BY users DESC;

-- 8. 版本分布 (是否有版本差异导致的问题)
SELECT
  app_info.version AS app_version,
  COUNT(DISTINCT user_pseudo_id) AS users,
  COUNT(*) AS events
FROM `vibedance-540a0.analytics_533044528.events_*`
WHERE _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 7 DAY))
GROUP BY app_info.version
ORDER BY users DESC
LIMIT 10;

-- =============================================
-- V. 异常监控
-- =============================================
-- 9. 崩溃/错误趋势 (如果有 firebase_crashlytics 或自定义 error 事件)
SELECT
  event_date,
  COUNT(DISTINCT CASE WHEN event_name = 'app_exception' THEN user_pseudo_id END) AS crash_users,
  COUNT(CASE WHEN event_name = 'app_exception' THEN 1 END) AS crash_count
FROM `vibedance-540a0.analytics_533044528.events_*`
WHERE _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 14 DAY))
GROUP BY event_date
ORDER BY event_date;
