"""Read-only BigQuery query runner; credentials supplied by environment, no secret output."""
import argparse
import datetime
import json
import os
from pathlib import Path
from google.cloud import bigquery

ROOT = Path(__file__).resolve().parent
TABLES = {
 'daily':'aidata2025.ads_collartweb.ads_oper_basic_indicator_daily_di',
 'country':'aidata2025.ads_collartweb.ads_oper_basic_indicator_country_di',
 'attr':'aidata2025.ads_collartweb.ads_oper_basic_indicator_attr_di',
 'active':'aidata2025.ads_collartweb.ads_oper_user_active_di',
 'profile':'aidata2025.ads_collartweb.ads_oper_user_profile_df',
 'stripe':'pubdata2025.dwd.dwd_cdct_revenue_stripe_di',
}

def flatten(fields, prefix=''):
    out=[]
    for f in fields:
        name=prefix+f.name
        out.append(dict(name=name,type=f.field_type,mode=f.mode))
        out.extend(flatten(f.fields,name+'.'))
    return out

def save(name,value):
    (ROOT/name).write_text(json.dumps(value,ensure_ascii=False,indent=2,default=str)+'\n',encoding='utf-8')

def main():
    parser=argparse.ArgumentParser()
    parser.add_argument('name')
    args=parser.parse_args()
    client=bigquery.Client(project=os.environ.get('BQ_EXECUTION_PROJECT','aidata2025'))
    if args.name=='schema':
        result={}
        for name,table in TABLES.items():
            t=client.get_table(table,timeout=30)
            result[name]=dict(table=table,rows=t.num_rows,modified=t.modified,partition=t.time_partitioning.to_api_repr() if t.time_partitioning else None,fields=flatten(t.schema))
        save('schemas.json',result)
        print(json.dumps({n:dict(rows=v['rows'],modified=str(v['modified']),fields=[f['name'] for f in v['fields']]) for n,v in result.items()},ensure_ascii=False))
        return
    sql=(ROOT/(args.name+'.sql')).read_text(encoding='utf-8')
    params=[]
    if '@internal_user_ids' in sql:
        cfg=json.loads((ROOT/'internal-exclusions.json').read_text(encoding='utf-8'))
        params.append(bigquery.ArrayQueryParameter('internal_user_ids','STRING',cfg['user_ids']))
    dry=client.query(sql,job_config=bigquery.QueryJobConfig(dry_run=True,use_query_cache=False,query_parameters=params),timeout=30)
    if dry.total_bytes_processed and dry.total_bytes_processed>20_000_000_000:
        raise ValueError('Query exceeds 20 GB review threshold')
    job=client.query(sql,job_config=bigquery.QueryJobConfig(maximum_bytes_billed=20_000_000_000,query_parameters=params),timeout=30)
    rows=[dict(r) for r in job.result(timeout=160)]
    save(args.name+'.json',rows)
    receipt=dict(name=args.name,job_id=job.job_id,location=job.location,rows=len(rows),dry_run_bytes=dry.total_bytes_processed,bytes_processed=job.total_bytes_processed,cache_hit=job.cache_hit,completed_at=datetime.datetime.now(datetime.timezone.utc).isoformat())
    save(args.name+'_receipt.json',receipt)
    print(json.dumps(dict(receipt=receipt,data=rows if len(rows)<=35 else rows[:8],preview_truncated=len(rows)>35),ensure_ascii=False,default=str))

if __name__=='__main__': main()
