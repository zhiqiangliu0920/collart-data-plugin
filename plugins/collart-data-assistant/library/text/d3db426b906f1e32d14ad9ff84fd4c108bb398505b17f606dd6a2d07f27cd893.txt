-- Collart Web: 韩国新增用户增长来源 + 行为 + 付费转化
-- 查询日期: 2026-07-15
-- 新增口径: ads_oper_user_new_collartweb_di
-- 对比窗口: 近7日(2026-07-08~2026-07-14) vs 前7日(2026-07-01~2026-07-07)
-- 趋势窗口: 近30日 2026-06-15~2026-07-14

DECLARE trend_start DATE DEFAULT DATE '2026-06-15';
DECLARE prior_start DATE DEFAULT DATE '2026-07-01';
DECLARE prior_end DATE DEFAULT DATE '2026-07-07';
DECLARE recent_start DATE DEFAULT DATE '2026-07-08';
DECLARE recent_end DATE DEFAULT DATE '2026-07-14';

WITH korea_new AS (
  SELECT
    event_date,
    user_pseudo_id,
    user_id,
    country,
    device_category,
    operating_system,
    browser,
    traffic_src_source,
    traffic_src_name,
    traffic_src_type
  FROM `aidata2025.ads_collart.ads_oper_user_new_collartweb_di`
  WHERE event_date BETWEEN trend_start AND recent_end
    AND (
      LOWER(country) LIKE '%korea%'
      OR country IN ('KR', 'South Korea', 'Korea, Republic of')
    )
),

daily_trend AS (
  SELECT
    event_date,
    COUNT(DISTINCT user_pseudo_id) AS new_users,
    COUNTIF(LOWER(COALESCE(traffic_src_type, '')) = 'delivery') AS delivery_users,
    COUNTIF(LOWER(COALESCE(traffic_src_type, '')) = 'organic') AS organic_users,
    COUNTIF(LOWER(COALESCE(traffic_src_type, '')) = 'inhouse') AS inhouse_users
  FROM korea_new
  GROUP BY event_date
),

period_source AS (
  SELECT
    CASE
      WHEN event_date BETWEEN recent_start AND recent_end THEN 'recent_7d'
      WHEN event_date BETWEEN prior_start AND prior_end THEN 'prior_7d'
    END AS period,
    COALESCE(traffic_src_type, '(null)') AS traffic_src_type,
    COALESCE(traffic_src_source, '(null)') AS traffic_src_source,
    COALESCE(traffic_src_name, '(null)') AS traffic_src_name,
    COUNT(DISTINCT user_pseudo_id) AS new_users
  FROM korea_new
  WHERE event_date BETWEEN prior_start AND recent_end
  GROUP BY ALL
),

recent_cohort AS (
  SELECT DISTINCT
    user_pseudo_id,
    user_id,
    event_date AS new_date,
    traffic_src_type,
    traffic_src_source,
    traffic_src_name,
    device_category,
    operating_system
  FROM korea_new
  WHERE event_date BETWEEN recent_start AND recent_end
),

event_base AS (
  SELECT
    e.user_pseudo_id,
    e.user_id,
    e.event_name,
    e.event_timestamp,
    (
      SELECT ep.value.string_value
      FROM UNNEST(e.event_params) ep
      WHERE ep.key = 'page_location'
    ) AS page_location,
    (
      SELECT ep.value.string_value
      FROM UNNEST(e.event_params) ep
      WHERE ep.key = 'service_type'
    ) AS service_type
  FROM `storytemplate-10a27.analytics_232977577.events_*` e
  INNER JOIN recent_cohort c
    ON e.user_pseudo_id = c.user_pseudo_id
  WHERE e._TABLE_SUFFIX BETWEEN FORMAT_DATE('%Y%m%d', recent_start)
                            AND FORMAT_DATE('%Y%m%d', recent_end)
    AND e.app_info.id IS NULL
),

behavior AS (
  SELECT
    COUNT(DISTINCT user_pseudo_id) AS cohort_uv,
    COUNT(DISTINCT IF(event_name = 'session_start', user_pseudo_id, NULL)) AS session_uv,
    COUNT(DISTINCT IF(event_name = 'page_view', user_pseudo_id, NULL)) AS page_view_uv,
    COUNT(DISTINCT IF(event_name IN ('login_success', 'login_account_success'), user_pseudo_id, NULL)) AS login_uv,
    COUNT(DISTINCT IF(event_name LIKE 'fashion_%' OR LOWER(COALESCE(page_location, '')) LIKE '%ai-fashion-video%' OR LOWER(COALESCE(page_location, '')) LIKE '%soft-allure%', user_pseudo_id, NULL)) AS fashion_related_uv,
    COUNT(DISTINCT IF(LOWER(COALESCE(page_location, '')) LIKE '%ai-fashion-video%', user_pseudo_id, NULL)) AS ai_fashion_page_uv,
    COUNT(DISTINCT IF(LOWER(COALESCE(page_location, '')) LIKE '%soft-allure%', user_pseudo_id, NULL)) AS soft_allure_page_uv,
    COUNT(DISTINCT IF(event_name = 'fashion_video_generate_click', user_pseudo_id, NULL)) AS fashion_generate_click_uv,
    COUNT(DISTINCT IF(event_name = 'ai_service_start', user_pseudo_id, NULL)) AS ai_start_uv,
    COUNT(DISTINCT IF(event_name = 'ai_service_success', user_pseudo_id, NULL)) AS ai_success_uv,
    COUNT(DISTINCT IF(event_name = 'ai_service_start' AND LOWER(COALESCE(service_type, '')) = 'fashion_video', user_pseudo_id, NULL)) AS fashion_real_use_uv,
    COUNT(DISTINCT IF(event_name = 'vip_show', user_pseudo_id, NULL)) AS vip_show_uv,
    COUNT(DISTINCT IF(event_name = 'vip_subscribe', user_pseudo_id, NULL)) AS vip_subscribe_click_uv,
    COUNT(DISTINCT IF(event_name = 'vip_subscribe_succeed', user_pseudo_id, NULL)) AS vip_subscribe_succeed_uv,
    COUNT(DISTINCT IF(event_name = 'credit_pack_purchase', user_pseudo_id, NULL)) AS credit_pack_purchase_uv
  FROM event_base
),

landing_pages AS (
  SELECT
    REGEXP_EXTRACT(LOWER(page_location), r'collart\.ai[^?#]*') AS page_path,
    COUNT(DISTINCT user_pseudo_id) AS uv
  FROM (
    SELECT
      user_pseudo_id,
      ARRAY_AGG(page_location IGNORE NULLS ORDER BY event_timestamp LIMIT 1)[SAFE_OFFSET(0)] AS page_location
    FROM event_base
    WHERE page_location IS NOT NULL
    GROUP BY user_pseudo_id
  )
  GROUP BY page_path
  ORDER BY uv DESC
  LIMIT 15
),

top_events AS (
  SELECT
    event_name,
    COUNT(DISTINCT user_pseudo_id) AS uv,
    COUNT(*) AS pv
  FROM event_base
  GROUP BY event_name
  ORDER BY uv DESC
  LIMIT 25
),

identity_map AS (
  SELECT
    c.user_pseudo_id,
    COALESCE(
      NULLIF(c.user_id, ''),
      ARRAY_AGG(e.user_id IGNORE NULLS ORDER BY e.event_timestamp DESC LIMIT 1)[SAFE_OFFSET(0)]
    ) AS resolved_user_id
  FROM recent_cohort c
  LEFT JOIN event_base e
    ON c.user_pseudo_id = e.user_pseudo_id
  GROUP BY c.user_pseudo_id, c.user_id
),

paid AS (
  SELECT
    o.user_id,
    COUNT(*) AS paid_order_cnt,
    ROUND(SUM(SAFE_DIVIDE(o.total_price, er.rate) / 100), 2) AS revenue_usd,
    MIN(DATETIME(TIMESTAMP(o.created_at), 'Asia/Shanghai')) AS first_paid_time_cn,
    ARRAY_AGG(DISTINCT o.product_name IGNORE NULLS LIMIT 10) AS products
  FROM `aidata2025.ods.orders` o
  LEFT JOIN `aidata2025.dim.exchange_rate_days` er
    ON DATE(o.created_at) = CAST(er.stats_date AS DATE)
   AND LOWER(o.currency) = LOWER(er.currency)
  WHERE DATE(TIMESTAMP(o.created_at), 'Asia/Shanghai') BETWEEN recent_start AND recent_end
    AND o.status IN ('active', 'paid', 'active_ending')
    AND o.user_id IN (
      SELECT resolved_user_id FROM identity_map WHERE resolved_user_id IS NOT NULL
    )
  GROUP BY o.user_id
),

pay_summary AS (
  SELECT
    COUNT(DISTINCT im.user_pseudo_id) AS cohort_uv,
    COUNT(DISTINCT im.resolved_user_id) AS logged_in_uv,
    COUNT(DISTINCT p.user_id) AS paid_users,
    IFNULL(SUM(p.paid_order_cnt), 0) AS paid_orders,
    ROUND(IFNULL(SUM(p.revenue_usd), 0), 2) AS revenue_usd
  FROM identity_map im
  LEFT JOIN paid p
    ON im.resolved_user_id = p.user_id
),

paid_detail AS (
  SELECT
    im.user_pseudo_id,
    p.user_id,
    c.new_date,
    c.traffic_src_type,
    c.traffic_src_source,
    c.traffic_src_name,
    p.paid_order_cnt,
    p.revenue_usd,
    p.first_paid_time_cn,
    p.products
  FROM paid p
  JOIN identity_map im
    ON p.user_id = im.resolved_user_id
  JOIN recent_cohort c
    ON im.user_pseudo_id = c.user_pseudo_id
)

SELECT 'daily_trend' AS section, TO_JSON_STRING(ARRAY_AGG(STRUCT(event_date, new_users, delivery_users, organic_users, inhouse_users) ORDER BY event_date)) AS payload
FROM daily_trend

UNION ALL
SELECT 'period_source', TO_JSON_STRING(ARRAY_AGG(STRUCT(period, traffic_src_type, traffic_src_source, traffic_src_name, new_users) ORDER BY period, new_users DESC))
FROM period_source

UNION ALL
SELECT 'behavior', TO_JSON_STRING(ANY_VALUE(behavior))
FROM behavior

UNION ALL
SELECT 'landing_pages', TO_JSON_STRING(ARRAY_AGG(STRUCT(page_path, uv) ORDER BY uv DESC))
FROM landing_pages

UNION ALL
SELECT 'top_events', TO_JSON_STRING(ARRAY_AGG(STRUCT(event_name, uv, pv) ORDER BY uv DESC))
FROM top_events

UNION ALL
SELECT 'pay_summary', TO_JSON_STRING(ANY_VALUE(pay_summary))
FROM pay_summary

UNION ALL
SELECT 'paid_detail', TO_JSON_STRING(ARRAY_AGG(STRUCT(user_pseudo_id, user_id, new_date, traffic_src_type, traffic_src_source, traffic_src_name, paid_order_cnt, revenue_usd, first_paid_time_cn, products) ORDER BY revenue_usd DESC))
FROM paid_detail;
