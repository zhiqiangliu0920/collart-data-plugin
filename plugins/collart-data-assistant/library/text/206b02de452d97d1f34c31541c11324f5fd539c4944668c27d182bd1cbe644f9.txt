-- Collart Web: 周末穿搭带货场景生成用户新老判定
-- 查询日期: 2026-06-29
-- 统计窗口: 2026-06-27 ~ 2026-06-28（周六/周日，北京时间自然日）
-- 场景口径: ai-fashion-video 页面 / fashion_% 事件；优先识别 20260624 SEA CollartFashionAI 投放 campaign
-- 新用户口径: aidata2025.ads_collart.ads_oper_user_new_collartweb_di 命中生成当天 user_pseudo_id

DECLARE start_date DATE DEFAULT DATE '2026-06-27';
DECLARE end_date DATE DEFAULT DATE '2026-06-28';
DECLARE campaign_name STRING DEFAULT '20260624_sea_collartfashionai_sales_webregistration';

WITH event_base AS (
  SELECT
    PARSE_DATE('%Y%m%d', event_date) AS event_dt,
    DATETIME(TIMESTAMP_MICROS(event_timestamp), 'Asia/Shanghai') AS event_time_cn,
    event_timestamp,
    user_pseudo_id,
    user_id,
    event_name,
    geo.country AS country,
    device.category AS device_category,
    traffic_source.source AS source,
    traffic_source.medium AS medium,
    traffic_source.name AS campaign,
    (
      SELECT ep.value.string_value
      FROM UNNEST(event_params) ep
      WHERE ep.key = 'page_location'
    ) AS page_location
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN FORMAT_DATE('%Y%m%d', start_date)
                          AND FORMAT_DATE('%Y%m%d', end_date)
    AND app_info.id IS NULL
),

fashion_cohort AS (
  SELECT DISTINCT
    user_pseudo_id
  FROM event_base
  WHERE event_name LIKE 'fashion_%'
     OR LOWER(COALESCE(page_location, '')) LIKE '%ai-fashion-video%'
     OR LOWER(COALESCE(campaign, '')) = campaign_name
),

generate_events AS (
  SELECT
    e.*
  FROM event_base e
  JOIN fashion_cohort c
    USING (user_pseudo_id)
  WHERE e.event_name = 'fashion_video_generate_click'
),

first_landing AS (
  SELECT
    user_pseudo_id,
    ARRAY_AGG(page_location IGNORE NULLS ORDER BY event_timestamp LIMIT 1)[SAFE_OFFSET(0)] AS first_page_location,
    ARRAY_AGG(source IGNORE NULLS ORDER BY event_timestamp LIMIT 1)[SAFE_OFFSET(0)] AS first_source,
    ARRAY_AGG(medium IGNORE NULLS ORDER BY event_timestamp LIMIT 1)[SAFE_OFFSET(0)] AS first_medium,
    ARRAY_AGG(campaign IGNORE NULLS ORDER BY event_timestamp LIMIT 1)[SAFE_OFFSET(0)] AS first_campaign
  FROM event_base
  GROUP BY user_pseudo_id
),

new_user AS (
  SELECT DISTINCT
    event_date,
    user_pseudo_id
  FROM `aidata2025.ads_collart.ads_oper_user_new_collartweb_di`
  WHERE event_date BETWEEN start_date AND end_date
),

active_user AS (
  SELECT
    event_date,
    user_pseudo_id,
    ANY_VALUE(is_new) AS active_table_is_new,
    ANY_VALUE(user_id) AS active_user_id,
    ANY_VALUE(traffic_src_name) AS traffic_src_name,
    ANY_VALUE(traffic_src_source) AS traffic_src_source,
    ANY_VALUE(traffic_src_type) AS traffic_src_type,
    ANY_VALUE(country) AS active_country,
    ANY_VALUE(device_category) AS active_device_category
  FROM `aidata2025.dwd.dwd_oper_user_collartweb_di`
  WHERE event_date BETWEEN start_date AND end_date
  GROUP BY event_date, user_pseudo_id
)

SELECT
  g.event_dt,
  g.event_time_cn,
  g.user_pseudo_id,
  COALESCE(g.user_id, au.active_user_id) AS user_id,
  IF(nu.user_pseudo_id IS NOT NULL, '新用户', '老用户') AS new_old_type,
  au.active_table_is_new,
  COALESCE(au.active_country, g.country) AS country,
  COALESCE(au.active_device_category, g.device_category) AS device_category,
  COALESCE(au.traffic_src_source, fl.first_source, g.source) AS source,
  COALESCE(fl.first_medium, g.medium) AS medium,
  COALESCE(au.traffic_src_name, fl.first_campaign, g.campaign) AS campaign,
  fl.first_page_location,
  COUNT(*) OVER (PARTITION BY g.user_pseudo_id) AS generate_click_cnt_by_pseudo
FROM generate_events g
LEFT JOIN new_user nu
  ON g.event_dt = nu.event_date
 AND g.user_pseudo_id = nu.user_pseudo_id
LEFT JOIN active_user au
  ON g.event_dt = au.event_date
 AND g.user_pseudo_id = au.user_pseudo_id
LEFT JOIN first_landing fl
  ON g.user_pseudo_id = fl.user_pseudo_id
ORDER BY g.event_time_cn;
