-- Web留存率计算
WITH first_visit AS (
  SELECT 
    user_pseudo_id, 
    MIN(DATE(TIMESTAMP_MICROS(event_timestamp))) as first_date
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX >= '20260301' AND _TABLE_SUFFIX <= '20260331'
    AND app_info.id IS NULL
    AND event_name = 'session_start'
  GROUP BY user_pseudo_id
),
retention AS (
  SELECT 
    f.first_date,
    DATE_DIFF(DATE(TIMESTAMP_MICROS(e.event_timestamp)), f.first_date, DAY) as day_diff,
    COUNT(DISTINCT e.user_pseudo_id) as retained_users
  FROM first_visit f
  JOIN `storytemplate-10a27.analytics_232977577.events_*` e 
    ON f.user_pseudo_id = e.user_pseudo_id
  WHERE e.app_info.id IS NULL
    AND e.event_name = 'session_start'
    AND e._TABLE_SUFFIX >= '20260301' AND e._TABLE_SUFFIX <= '20260331'
  GROUP BY f.first_date, day_diff
),
first_day_users AS (
  SELECT first_date, COUNT(*) as total_users
  FROM first_visit
  GROUP BY first_date
)
SELECT 
  r.first_date,
  r.day_diff,
  r.retained_users,
  f.total_users,
  ROUND(r.retained_users * 100.0 / f.total_users, 2) as retention_rate
FROM retention r
JOIN first_day_users f ON r.first_date = f.first_date
WHERE r.day_diff IN (1, 3, 7, 14, 30)
ORDER BY r.first_date, r.day_diff;
