"""Build a Chinese, source-backed revenue readout from the saved aggregate query results."""
import csv
import datetime as dt
import json
from pathlib import Path

ROOT=Path(__file__).resolve().parent
def read(name): return json.loads((ROOT/name).read_text(encoding='utf-8'))
rows=read('stripe_analysis.json')
def get(dim,period): return {r['value']:r for r in rows if r['dimension']==dim and r['period']==period}
def money(x): return ('-$' if x<0 else '$')+f'{abs(x):,.2f}'
def pct(x): return f'{x:+.2%}'
totals={p:get('total',p)['all'] for p in ['previous','current']}
old,new=totals['previous'],totals['current']
delta=new['net_revenue']-old['net_revenue']
qa={r['section']:json.loads(r['detail']) for r in read('qa.json') if r['section']!='status_mix'}
assert qa['stripe_quality']['raw_rows']==qa['stripe_quality']['unique_transactions']==1197
assert qa['stripe_quality']['missing_transaction']==0
assert qa['stripe_quality']['beijing_date_mismatch']==0
assert qa['duplicates']['conflicting_keys']==0
assert qa['internal_orders']['transactions']==0
ads={r['event_date']:r for r in read('ads_daily.json')}
country={r['event_date']:r for r in read('freshness.json') if r['layer']=='country' and r['package_name']=='collart_web'}
day_rows=sorted([r for r in rows if r['dimension']=='date'],key=lambda r:r['value'])
reconcile=[]
for r in day_rows:
    error=r['net_revenue']-ads[r['value']]['purchase_revenue']
    assert abs(error)<0.000001
    assert abs(r['net_revenue']-country[r['value']]['revenue'])<0.000001
    reconcile.append(dict(date=r['value'],stripe=r['net_revenue'],ads=ads[r['value']]['purchase_revenue'],difference=error))
for p in totals:
    for dimension in ['date','type','country','product']:
        assert abs(sum(r['net_revenue'] for r in get(dimension,p).values())-totals[p]['net_revenue'])<1e-6
    assert len(get('date',p))==7

type_labels={'credit':'点数包','new_subscription':'首次订阅','renewal':'续订','other:subscription':'未拆分订阅（subscription）'}
type_changes=[]
for k in type_labels:
    a,b=get('type','previous')[k],get('type','current')[k]
    type_changes.append(dict(type=type_labels[k],previous=a['net_revenue'],current=b['net_revenue'],delta=b['net_revenue']-a['net_revenue'],change=b['net_revenue']/a['net_revenue']-1,contribution=(b['net_revenue']-a['net_revenue'])/delta,previous_transactions=a['transactions'],current_transactions=b['transactions']))

def changes(dim):
    a,b=get(dim,'previous'),get(dim,'current')
    out=[]
    for k in a.keys()|b.keys():
        before=a.get(k,{}).get('net_revenue',0);after=b.get(k,{}).get('net_revenue',0)
        out.append(dict(name=k,previous=before,current=after,delta=after-before))
    return sorted(out,key=lambda r:r['delta'],reverse=True)

def export(name,records):
    with (ROOT/name).open('w',encoding='utf-8-sig',newline='') as f:
        w=csv.DictWriter(f,fieldnames=list(records[0]));w.writeheader();w.writerows(records)

export('daily_revenue.csv',day_rows)
export('revenue_type_changes.csv',type_changes)
export('country_changes.csv',changes('country'))
export('product_changes.csv',changes('product'))
export('reconciliation.csv',reconcile)
summary=dict(window=dict(current=['2026-09-04','2026-09-10'],previous=['2026-08-28','2026-09-03'],timezone='Asia/Shanghai',requested_calendar_window=['2026-09-05','2026-09-11'],unavailable_date='2026-09-11'),totals=totals,revenue_delta=delta,revenue_growth=delta/old['net_revenue'],type_changes=type_changes,qa=qa)
(ROOT/'summary.json').write_text(json.dumps(summary,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
