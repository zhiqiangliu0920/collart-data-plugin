import os
from datetime import datetime
from google.cloud import bigquery
from google.oauth2 import service_account

os.environ['PYTHONIOENCODING'] = 'utf-8'

KEY_PATH = r"{LOCAL_PATH}"
PROJECT_ID = "vidart-8b8ca"

def get_client():
    creds = service_account.Credentials.from_service_account_file(
        KEY_PATH, scopes=["https://www.googleapis.com/auth/cloud-platform"],
    )
    return bigquery.Client(credentials=creds, project=PROJECT_ID)

def fmt(val):
    if val is None:
        return "NULL"
    if isinstance(val, float):
        return f"{val:.2f}"
    return str(val)

def print_table(rows, headers, title):
    print(f"\n{'='*80}")
    print(f"[{title}]")
    print('='*80)
    if not rows:
        print("  (no data)")
        return
    widths = [len(h) for h in headers]
    for row in rows:
        for i, v in enumerate(row):
            widths[i] = max(widths[i], len(fmt(v)))
    sep = "+-" + "-+-".join("-"*w for w in widths) + "-+"
    print(sep)
    print("| " + " | ".join(h.ljust(w) for h, w in zip(headers, widths)) + " |")
    print(sep)
    for row in rows:
        print("| " + " | ".join(fmt(v).ljust(w) for v, w in zip(row, widths)) + " |")
    print(sep)
    print(f"  Total rows: {len(rows)}")

def run_query(client, sql, title, headers):
    try:
        rows = list(client.query(sql).result())
        data = [tuple(row.values()) for row in rows]
        print_table(data, headers, title)
        return data
    except Exception as e:
        print(f"\n{'='*80}")
        print(f"[{title}]")
        print('='*80)
        print(f"  Error: {e}")
        return []

# ALL event names in the entire dataset (to understand available events)

FROM `vidart-8b8ca.analytics_528583115.events_*`

# 1. Event types distribution for VIP users vs all users
SQL_EVENT_DIST = """
WITH vip_users AS (
  SELECT DISTINCT user_pseudo_id
  FROM `vidart-8b8ca.analytics_528583115.events_*`
  WHERE event_name = 'vip_subscribe_succeed'
    AND app_info.install_source <> 'manual_install'
    AND _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 90 DAY))
)
SELECT
  e.event_name,
  COUNT(DISTINCT e.user_pseudo_id) as vip_users,
  COUNT(*) as vip_events,
  ROUND(COUNT(*) * 100.0 / SUM(COUNT(*)) OVER(), 2) as pct_of_vip_events
FROM `vidart-8b8ca.analytics_528583115.events_*` e
JOIN vip_users v ON e.user_pseudo_id = v.user_pseudo_id
WHERE e._TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 90 DAY))
  AND e.app_info.install_source <> 'manual_install'
GROUP BY e.event_name
ORDER BY vip_events DESC
LIMIT 30
"""

# 2. Template-related events for VIP users (check event params)
SQL_TEMPLATE_USAGE = """
WITH vip_users AS (
  SELECT DISTINCT user_pseudo_id
  FROM `vidart-8b8ca.analytics_528583115.events_*`
  WHERE event_name = 'vip_subscribe_succeed'
    AND app_info.install_source <> 'manual_install'
    AND _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 90 DAY))
)
SELECT
  e.event_name,
  (SELECT value.string_value FROM UNNEST(e.event_params) WHERE key = 'template_id') as template_id,
  (SELECT value.string_value FROM UNNEST(e.event_params) WHERE key = 'template_name') as template_name,
  COUNT(DISTINCT e.user_pseudo_id) as users,
  COUNT(*) as events
FROM `vidart-8b8ca.analytics_528583115.events_*` e
JOIN vip_users v ON e.user_pseudo_id = v.user_pseudo_id
WHERE e._TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 90 DAY))
  AND e.app_info.install_source <> 'manual_install'
  AND (
    e.event_name LIKE '%template%'
    OR e.event_name LIKE '%temp%'
    OR e.event_name LIKE '%select%'
    OR e.event_name LIKE '%choose%'
    OR e.event_name LIKE '%use%'
    OR e.event_name LIKE '%apply%'
    OR (SELECT value.string_value FROM UNNEST(e.event_params) WHERE key = 'template_id') IS NOT NULL
    OR (SELECT value.string_value FROM UNNEST(e.event_params) WHERE key = 'template_name') IS NOT NULL
  )
GROUP BY e.event_name, template_id, template_name
ORDER BY events DESC
LIMIT 50
"""

# 3. Feature usage params keys that VIP users have
SQL_FEATURE_PARAMS = """
WITH vip_users AS (
  SELECT DISTINCT user_pseudo_id
  FROM `vidart-8b8ca.analytics_528583115.events_*`
  WHERE event_name = 'vip_subscribe_succeed'

# 4. Event sequence before purchase for each VIP user
SQL_BEFORE_PURCHASE = """
WITH vip_events AS (
  SELECT
    user_pseudo_id,
    event_name,
    event_timestamp,
    event_date,
    (SELECT value.string_value FROM UNNEST(event_params) WHERE key = 'template_id') as template_id,
    (SELECT value.string_value FROM UNNEST(event_params) WHERE key = 'template_name') as template_name,
    ROW_NUMBER() OVER (PARTITION BY user_pseudo_id ORDER BY event_timestamp DESC) as rn
  FROM `vidart-8b8ca.analytics_528583115.events_*`
  WHERE user_pseudo_id IN (
    SELECT DISTINCT user_pseudo_id
    FROM `vidart-8b8ca.analytics_528583115.events_*`
    WHERE event_name = 'vip_subscribe_succeed'
      AND app_info.install_source <> 'manual_install'
      AND _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 90 DAY))
  )
    AND app_info.install_source <> 'manual_install'
    AND _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 90 DAY))
),
purchase_time AS (
  SELECT user_pseudo_id, MIN(event_timestamp) as purchase_ts
  FROM `vidart-8b8ca.analytics_528583115.events_*`
  WHERE event_name = 'vip_subscribe_succeed'
    AND app_info.install_source <> 'manual_install'
    AND _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 90 DAY))
  GROUP BY user_pseudo_id
)
SELECT
  v.user_pseudo_id,
  v.event_name,
  v.event_date,
  v.template_id,
  ROUND((v.event_timestamp - p.purchase_ts) / 1000000.0 / 60, 1) as mins_before_purchase
FROM vip_events v
JOIN purchase_time p ON v.user_pseudo_id = p.user_pseudo_id
WHERE v.event_timestamp <= p.purchase_ts
  AND v.event_name <> 'vip_subscribe_succeed'
QUALIFY ROW_NUMBER() OVER (PARTITION BY v.user_pseudo_id ORDER BY v.event_timestamp DESC) <= 10
ORDER BY v.user_pseudo_id, v.event_timestamp DESC
"""

# 5. What features did VIP users use AFTER purchase?
SQL_AFTER_PURCHASE = """
WITH vip_events AS (
  SELECT
    user_pseudo_id,
    event_name,
    event_timestamp,
    event_date,
    (SELECT value.string_value FROM UNNEST(event_params) WHERE key = 'template_id') as template_id,
    (SELECT value.string_value FROM UNNEST(event_params) WHERE key = 'template_name') as template_name
  FROM `vidart-8b8ca.analytics_528583115.events_*`
  WHERE user_pseudo_id IN (
    SELECT DISTINCT user_pseudo_id
    FROM `vidart-8b8ca.analytics_528583115.events_*`
    WHERE event_name = 'vip_subscribe_succeed'
      AND app_info.install_source <> 'manual_install'
      AND _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 90 DAY))
  )
    AND app_info.install_source <> 'manual_install'
    AND _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 90 DAY))
),
purchase_time AS (
  SELECT user_pseudo_id, MAX(event_timestamp) as purchase_ts
  FROM `vidart-8b8ca.analytics_528583115.events_*`
  WHERE event_name = 'vip_subscribe_succeed'
    AND app_info.install_source <> 'manual_install'
    AND _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 90 DAY))

SELECT
  v.event_name,
  COUNT(DISTINCT v.user_pseudo_id) as users,
  COUNT(*) as events
FROM vip_events v
JOIN purchase_time p ON v.user_pseudo_id = p.user_pseudo_id
WHERE v.event_timestamp > p.purchase_ts
  AND v.event_name <> 'vip_subscribe_succeed'

LIMIT 30
"""

# 6. Session count and engagement for VIP users
SQL_ENGAGEMENT = """
WITH vip_users AS (
  SELECT DISTINCT user_pseudo_id
  FROM `vidart-8b8ca.analytics_528583115.events_*`
  WHERE event_name = 'vip_subscribe_succeed'
    AND app_info.install_source <> 'manual_install'
    AND _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 90 DAY))
)
SELECT
  e.user_pseudo_id,
  COUNT(DISTINCT e.event_name) as unique_events,
  COUNT(*) as total_events,
  COUNT(DISTINCT e.event_date) as active_days,
  COUNT(DISTINCT e.ga_session_id) as sessions,
  MIN(PARSE_DATE('%Y%m%d', e.event_date)) as first_seen,
  MAX(PARSE_DATE('%Y%m%d', e.event_date)) as last_seen
FROM `vidart-8b8ca.analytics_528583115.events_*` e
JOIN vip_users v ON e.user_pseudo_id = v.user_pseudo_id
WHERE e._TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 90 DAY))
  AND e.app_info.install_source <> 'manual_install'
GROUP BY e.user_pseudo_id
ORDER BY total_events DESC
"""

# 7. Key feature params values (e.g. what tool, what effect, what ai_feature was used)
SQL_FEATURE_DETAILS = """
WITH vip_users AS (
  SELECT DISTINCT user_pseudo_id
  FROM `vidart-8b8ca.analytics_528583115.events_*`
  WHERE event_name = 'vip_subscribe_succeed'
    AND app_info.install_source <> 'manual_install'
    AND _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 90 DAY))
)
SELECT
  e.event_name,
  p.key,
  CASE
    WHEN p.value.string_value IS NOT NULL THEN p.value.string_value
    WHEN p.value.int_value IS NOT NULL THEN CAST(p.value.int_value AS STRING)
    WHEN p.value.double_value IS NOT NULL THEN CAST(p.value.double_value AS STRING)
    WHEN p.value.float_value IS NOT NULL THEN CAST(p.value.float_value AS STRING)
    ELSE 'other'
  END as param_value,
  COUNT(DISTINCT e.user_pseudo_id) as users,
  COUNT(*) as events
FROM `vidart-8b8ca.analytics_528583115.events_*` e,
UNNEST(e.event_params) p
JOIN vip_users v ON e.user_pseudo_id = v.user_pseudo_id
WHERE e._TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 90 DAY))
  AND e.app_info.install_source <> 'manual_install'
  AND p.key IN ('feature_name', 'tool_name', 'effect_name', 'ai_feature', 'mode',
                'filter_name', 'style', 'category', 'action', 'content_type',
                'screen_name', 'page_name', 'source_screen', 'destination_screen',
                'template_category', 'template_style', 'ai_type', 'generation_type')
GROUP BY e.event_name, p.key, param_value
ORDER BY events DESC
LIMIT 50
"""
