  Collart Fashion 第五张表：日 × 默认维度基础指标
  Dataform: definitions/ads/collart/collart_fashion/ads_oper_basic_indicator_attr_di.sqlx
  目标表  : aidata2025.ads_collartfashion.ads_oper_basic_indicator_attr_di
  上游    : ads_collartfashion.ads_oper_user_active_di

  口径（对齐 Android 表 5 + iOS active）:
  - 维度：is_new / is_vip / country / traffic_src_type / traffic_src_platform / traffic_src_name / app_version
  - app_version：当日 Top5，其余折叠为 'other'；不加 city / package_name；不做 MAU
  - retain / rolling_retain：STRUCT d2-d30（与 active 1:1，d2=次日）；COUNTIF 分子，NULL 不计
  - 当日订阅/购点：active.subscription.is_subscribe / is_credit_pack（源=Apple IAP）
  - 7 日内订阅/购点：含当日，date_diff ∈ [0,6]
  - DAU：仅 COUNT DISTINCT is_active=TRUE；收入补充行不计活跃
  - 金额：SUM(active.revenue.*)（Fashion Stripe 用户日收入）
  - 流失/回流：COUNTIF(is_churn/is_return)；观察窗未满 NULL 不计
  - 刷新窗口 [biz_date-30, biz_date]；源表额外前读 6 日供 7 日商业窗口
*/

WITH active_raw AS (
  SELECT
    event_date,
    user_pseudo_id,
    is_new,
    (is_vip IN ('1', '2')) AS is_vip,
    country,
    traffic_src_type,
    traffic_src_platform,
    traffic_src_name,
    app_version,
    is_active,
    is_churn,
    is_return,
    retain,
    rolling_retain,
    subscription,
    revenue
  FROM `aidata2025.ads_collartfashion.ads_oper_user_active_di`
  WHERE event_date BETWEEN source_from_date AND to_date
    AND user_pseudo_id IS NOT NULL
),

purch_day AS (
  SELECT
    event_date,
    user_pseudo_id,
    LOGICAL_OR(subscription.is_subscribe) AS is_subscribe,
    LOGICAL_OR(subscription.is_credit_pack) AS is_credit_pack
  FROM active_raw
  GROUP BY event_date, user_pseudo_id
),

win7 AS (
  SELECT
    a.event_date,
    a.user_pseudo_id,
    LOGICAL_OR(p.is_subscribe) AS subscribe_in_7d,
    LOGICAL_OR(p.is_credit_pack) AS credit_pack_in_7d
  FROM active_raw AS a
  LEFT JOIN purch_day AS p
    ON a.user_pseudo_id = p.user_pseudo_id
   AND DATE_DIFF(a.event_date, p.event_date, DAY) BETWEEN 0 AND 6
  WHERE a.event_date BETWEEN from_date AND to_date
  GROUP BY a.event_date, a.user_pseudo_id
),

active_base AS (
  SELECT
    a.*,
    IF(DATE_ADD(a.event_date, INTERVAL 6 DAY) > to_date, NULL, COALESCE(w.subscribe_in_7d, FALSE)) AS subscribe_in_7d,
    IF(DATE_ADD(a.event_date, INTERVAL 6 DAY) > to_date, NULL, COALESCE(w.credit_pack_in_7d, FALSE)) AS credit_pack_in_7d
  FROM active_raw AS a
  LEFT JOIN win7 AS w
    USING (event_date, user_pseudo_id)
  WHERE a.event_date BETWEEN from_date AND to_date
),

top_version AS (
  SELECT
    event_date,
    app_version
  FROM (
    SELECT
      event_date,
      app_version,
      COUNT(DISTINCT user_pseudo_id) AS dau,
      ROW_NUMBER() OVER (
        PARTITION BY event_date
        ORDER BY COUNT(DISTINCT user_pseudo_id) DESC, app_version
      ) AS rn
    FROM active_base
    WHERE is_active
    GROUP BY event_date, app_version
  )
  WHERE rn <= 5
)
