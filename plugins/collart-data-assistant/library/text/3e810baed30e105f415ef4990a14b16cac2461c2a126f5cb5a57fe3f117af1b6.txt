-- MotionControl 不同 model 的 PV/UV 统计
-- 时间范围: 2026-04-01 起
-- 提取者: 强强
-- 日期: 2026-04-07

SELECT
  model_value,
  COUNT(*) AS pv,
  COUNT(DISTINCT user_pseudo_id) AS uv
FROM (
  SELECT
    user_pseudo_id,
    (SELECT value.string_value FROM UNNEST(event_params) WHERE key = 'model') AS model_value,
    (SELECT value.string_value FROM UNNEST(event_params) WHERE key = 'function_type') AS function_type
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX >= '20260401'
    AND app_info.id = 'free.ai.photo.generator.collart.ai'
    AND event_name LIKE '%ai_service%'
)
WHERE function_type = 'MotionControl'
  AND model_value IS NOT NULL
GROUP BY model_value
ORDER BY pv DESC
