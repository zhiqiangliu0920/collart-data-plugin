    dependencies: ["h_dws_oper_asa_delivery_di"]

DECLARE
  from_date DATE DEFAULT ${dataform.projectConfig.vars.biz_date}-30; --date('2025-01-01');
DECLARE
  to_date DATE DEFAULT ${dataform.projectConfig.vars.biz_date};
BEGIN TRANSACTION;
  -- 这里放置一系列 DML 语句
  --删除分区数据
DELETE
FROM
  `dwd.dwd_cdct_delivery_cost_di`
WHERE
  event_date BETWEEN from_date
  AND to_date;
INSERT INTO
  dwd.dwd_cdct_delivery_cost_di

WITH
--  Google Ads 投放
ga_data AS (
  SELECT
    event_date,
    'ga-delivery' AS delivery_type,
    app_name,
    CASE
        WHEN app_name='FBD2' THEN 'facebook.video.downloader.savefrom.fb.saver.fast'
        WHEN app_name='INS3' THEN 'instagram.video.downloader.story.saver.ig'
        WHEN app_name='INS3Lite' THEN 'instagram.video.downloader.story.saver.ig.insaver'
        WHEN app_name='PTD' THEN 'pin.pinterest.video.downloader.forpinterest.pinsaver'
        WHEN app_name='ShotCut' THEN 'video.editor.videomaker.effects.fx'
        WHEN app_name='TTD1' THEN 'tiktok.video.downloader.nowatermark.tiktokdownload'
        WHEN app_name='TTD2' THEN 'tiktok.video.downloader.nowatermark.tiktokdownload.snaptik'
        WHEN app_name='TWD' THEN 'twittervideodownloader.twitter.videoindir.savegif.twdown'
        WHEN app_name='WAD' THEN 'statussaver.videodownloader.downloadstatus.savestatus.storysaver'
        WHEN app_name='COLLART' THEN 'free.ai.photo.generator.collart.ai'
        WHEN app_name='DRAMA' THEN 'com.my.drama.box.reel.short.stream.tv.app'
        when app_name='REMOVE' THEN 'com.photo.retouch.remove.objects.magic.eraser.app'
        when app_name='PICFIX' THEN 'ai.photo.editor.fix.retake.face.app'
        WHEN app_name='COLLART-WEB' THEN 'collart_web'
        when app_name='MUSELAB' THEN 'ai.photo.generator.editor.fotos.ai.image.app.free'
    END AS package_name,
    country,
    CAST(NULL AS STRING) AS campaign_id,
    campaign_name,
    CAST(NULL AS STRING) AS adgroup_id,
    CAST(NULL AS STRING) AS adgroup_name,
    CAST(NULL AS STRING) AS ad_id,
    CAST(NULL AS STRING) AS ad_name,
    CAST(NULL AS STRING) AS keyword,
    CAST(impressions AS INT64) AS impressions,
    CAST(clicks AS INT64) AS clicks,
    CAST(IFNULL(cost*hk_us_exchange_rate,0)+IFNULL(gst_cost*hk_us_exchange_rate,0) AS FLOAT64) AS cost,
    CAST(installs AS INT64) AS installs,
    CAST(conversions AS INT64) AS conversions,
    ARRAY<STRUCT<action_type STRING, action_count INT64>>[] AS actions
  FROM `pubdata2025.dws.dws_cdct_cost_google_1h`
  WHERE event_date BETWEEN from_date
  AND to_date
),

-- --  TikTok 投放

--  FB 投放
fb_data AS (
  SELECT
    event_date,
    'fb-delivery' AS delivery_type,
     case
      when package_name = 'com.SpringTech.NewSC' then 'COLLART-IOS'
      when package_name = 'free.ai.photo.generator.collart.ai' then 'COLLART-And'
      when package_name = 'ai.home.design.interior.room.decor.remodel.planner.free' then 'AIHOME'
      when package_name = 'ai.photo.editor.fix.retake.face.app' then 'PICFIX'
      WHEN package_name = 'collart-web' THEN 'collart_web'
      when package_name = 'ai.photo.generator.editor.fotos.ai.image.app.free' THEN 'muselab'
      when package_name = 'danceai.ai.video.generator.micmic.dance' THEN 'vibdance_ios'
      when package_name = 'ai.photo.generator.fotos.ai.image.editor.app.free' THEN 'looksart'
      else package_name
    end as app_name,
    case WHEN package_name = 'collart-web' THEN 'collart_web' else package_name end as package_name,
    country,
    campaign as campaign_id,
    campaign_name,
    CAST(NULL AS STRING) AS adgroup_id,
    CAST(NULL AS STRING) AS adgroup_name,
    CAST(NULL AS STRING) AS ad_id,
    CAST(NULL AS STRING) AS ad_name,
    CAST(NULL AS STRING) AS keyword,
    CAST(impressions AS INT64) AS impressions,
    CAST(clicks AS INT64) AS clicks,
    CAST(spend AS FLOAT64) AS cost,
    IFNULL((SELECT SUM(action_count) FROM UNNEST(actions) WHERE action_type in ('mobile_app_install','omni_app_install')),0) AS installs,
    CAST(NULL AS INT64) AS conversions,
    actions
  FROM `pubdata2025.dws.dws_cdct_cost_delivery_fb_1d`
  WHERE event_date BETWEEN from_date
  AND to_date
),

-- ASA 投放：campaign 为完整账单粒度；keyword 仅覆盖关键词定向子集，不能用于总成本。
asa_data AS (
  SELECT
    event_date,
    'asa-delivery' AS delivery_type,
    app_name,
    package_name,
    country,
    campaign_id,
    campaign_name,
    adgroup_id,
    adgroup_name,
    CAST(NULL AS STRING) AS ad_id,
    CAST(NULL AS STRING) AS ad_name,
    keyword,
    impressions,
    clicks,
    cost_usd AS cost,
    installs,
    CAST(NULL AS INT64) AS conversions,
    ARRAY<STRUCT<action_type STRING, action_count INT64>>[] AS actions
  FROM `dws.dws_oper_asa_delivery_di`
  WHERE event_date BETWEEN from_date AND to_date
)

-- 最终数据
select
  a.*,
  b.country_name
from(
  SELECT *
  FROM ga_data
  -- UNION ALL
  -- SELECT *
  -- FROM tt_data
  UNION ALL
  SELECT *
  FROM fb_data
  UNION ALL
  SELECT *
  FROM asa_data
) a 
left join aidata2025.dim.dim_country_info b
on a.country = b.country_name_3
;

COMMIT;
