-- 阶段一 1.1：流量概览
-- 目的：统计过去7天每日UV、新用户占比、流量来源分布
-- 日期范围: 2026-03-04 至 2026-03-10
-- 创建时间: 2026-03-11

WITH daily_traffic AS (
  -- 每日访问UV（基于events表page_view）
  SELECT
    PARSE_DATE('%Y%m%d', event_date) AS event_date,
    COUNT(DISTINCT user_pseudo_id) AS uv
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN '20260304' AND '20260310'
    AND app_info.id IS NULL
    AND event_name = 'page_view'
  GROUP BY event_date
),

daily_new_users AS (
  -- 每日新增用户
  SELECT
    event_date,
    COUNT(DISTINCT user_pseudo_id) AS new_users,
    SUM(CASE WHEN traffic_src_type = 'organic' THEN 1 ELSE 0 END) AS organic_new,
    SUM(CASE WHEN traffic_src_type = 'delivery' THEN 1 ELSE 0 END) AS delivery_new,
    SUM(CASE WHEN traffic_src_type = 'inhouse' THEN 1 ELSE 0 END) AS inhouse_new
  FROM aidata2025.ads_collart.ads_oper_user_new_collartweb_di
  WHERE event_date BETWEEN '2026-03-04' AND '2026-03-10'
  GROUP BY event_date
),

daily_dau AS (
  -- 每日DAU
  SELECT
    event_date,
    SUM(dau) AS dau
  FROM aidata2025.ads_collart.ads_oper_basic_indicator_collartweb_di
  WHERE event_date BETWEEN '2026-03-04' AND '2026-03-10'
  GROUP BY event_date
)

SELECT
  t.event_date,
  t.uv,
  d.dau,
  n.new_users,
  ROUND(n.new_users / t.uv * 100, 2) AS new_user_rate,
  n.organic_new,
  n.delivery_new,
  n.inhouse_new,
  ROUND(n.organic_new / n.new_users * 100, 2) AS organic_pct,
  ROUND(n.delivery_new / n.new_users * 100, 2) AS delivery_pct
FROM daily_traffic t
LEFT JOIN daily_new_users n ON t.event_date = n.event_date
LEFT JOIN daily_dau d ON t.event_date = d.event_date
ORDER BY t.event_date;
