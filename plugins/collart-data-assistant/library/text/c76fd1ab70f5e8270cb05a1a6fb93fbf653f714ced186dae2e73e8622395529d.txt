-- Collart Web: 周末 SEA 穿搭带货真实生成用户新老判定
-- 查询日期: 2026-06-29
-- 统计窗口: 2026-06-27 ~ 2026-06-28（周六/周日，北京时间自然日）
-- 投放口径: 20260624_SEA_CollartFashionAI_Sales_WebRegistration
-- 行粒度: user_pseudo_id
-- 新用户口径: 生成当天命中 aidata2025.ads_collart.ads_oper_user_new_collartweb_di

DECLARE start_date DATE DEFAULT DATE '2026-06-27';
DECLARE end_date DATE DEFAULT DATE '2026-06-28';
DECLARE campaign_name STRING DEFAULT '20260624_sea_collartfashionai_sales_webregistration';

WITH event_base AS (
  SELECT
    PARSE_DATE('%Y%m%d', event_date) AS event_dt,
    DATETIME(TIMESTAMP_MICROS(event_timestamp), 'Asia/Shanghai') AS event_time_cn,
    event_timestamp,
    user_pseudo_id,
    user_id,
    event_name,
    geo.country AS country,
    device.category AS device_category,
    traffic_source.source AS source,
    traffic_source.medium AS medium,
    traffic_source.name AS campaign,
    (
      SELECT ep.value.string_value
      FROM UNNEST(event_params) ep
      WHERE ep.key = 'page_location'
    ) AS page_location,
    (
      SELECT ep.value.string_value
      FROM UNNEST(event_params) ep
      WHERE ep.key = 'service_type'
    ) AS service_type
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN FORMAT_DATE('%Y%m%d', start_date)
                          AND FORMAT_DATE('%Y%m%d', end_date)
    AND app_info.id IS NULL
),

sea_fashion_users AS (
  SELECT DISTINCT user_pseudo_id
  FROM event_base
  WHERE LOWER(COALESCE(campaign, '')) = campaign_name
     OR LOWER(COALESCE(page_location, '')) LIKE CONCAT('%utm_campaign=', campaign_name, '%')
),

new_user AS (
  SELECT DISTINCT
    event_date,
    user_pseudo_id
  FROM `aidata2025.ads_collart.ads_oper_user_new_collartweb_di`
  WHERE event_date BETWEEN start_date AND end_date
),

active_user AS (
  SELECT
    event_date,
    user_pseudo_id,
    ANY_VALUE(is_new) AS active_table_is_new,
    ANY_VALUE(user_id) AS active_user_id,
    ANY_VALUE(traffic_src_source) AS traffic_src_source,
    ANY_VALUE(traffic_src_type) AS traffic_src_type,
    ANY_VALUE(country) AS active_country,
    ANY_VALUE(device_category) AS active_device_category
  FROM `aidata2025.dwd.dwd_oper_user_collartweb_di`
  WHERE event_date BETWEEN start_date AND end_date
  GROUP BY event_date, user_pseudo_id
),

user_rollup AS (
  SELECT
    e.user_pseudo_id,
    ARRAY_AGG(e.user_id IGNORE NULLS ORDER BY e.event_timestamp DESC LIMIT 1)[SAFE_OFFSET(0)] AS user_id,
    MIN(e.event_dt) AS first_event_dt,
    MIN(e.event_time_cn) AS first_event_time_cn,
    MAX(e.event_time_cn) AS last_event_time_cn,
    ARRAY_AGG(e.country IGNORE NULLS ORDER BY e.event_timestamp LIMIT 1)[SAFE_OFFSET(0)] AS country,
    ARRAY_AGG(e.device_category IGNORE NULLS ORDER BY e.event_timestamp LIMIT 1)[SAFE_OFFSET(0)] AS device_category,
    ARRAY_AGG(e.source IGNORE NULLS ORDER BY e.event_timestamp LIMIT 1)[SAFE_OFFSET(0)] AS source,
    ARRAY_AGG(e.medium IGNORE NULLS ORDER BY e.event_timestamp LIMIT 1)[SAFE_OFFSET(0)] AS medium,
    ARRAY_AGG(e.campaign IGNORE NULLS ORDER BY e.event_timestamp LIMIT 1)[SAFE_OFFSET(0)] AS campaign,
    ARRAY_AGG(e.page_location IGNORE NULLS ORDER BY e.event_timestamp LIMIT 1)[SAFE_OFFSET(0)] AS first_page_location,
    COUNTIF(e.event_name = 'fashion_video_generate_click') AS generate_click_cnt,
    COUNTIF(e.event_name = 'ai_service_start') AS ai_service_start_cnt,
    COUNTIF(e.event_name = 'ai_service_success') AS ai_service_success_cnt,
    ARRAY_AGG(DISTINCT e.service_type IGNORE NULLS LIMIT 10) AS service_types
  FROM event_base e
  JOIN sea_fashion_users s
    USING (user_pseudo_id)
  GROUP BY e.user_pseudo_id
),

classified AS (
  SELECT
    u.*,
    COALESCE(au.active_user_id, u.user_id) AS resolved_user_id,
    IF(nu.user_pseudo_id IS NOT NULL, TRUE, FALSE) AS is_new_by_new_user_table,
    au.active_table_is_new,
    COALESCE(au.active_country, u.country) AS resolved_country,
    COALESCE(au.active_device_category, u.device_category) AS resolved_device_category,
    COALESCE(au.traffic_src_source, u.source) AS resolved_source,
    au.traffic_src_type
  FROM user_rollup u
  LEFT JOIN new_user nu
    ON u.first_event_dt = nu.event_date
   AND u.user_pseudo_id = nu.user_pseudo_id
  LEFT JOIN active_user au
    ON u.first_event_dt = au.event_date
   AND u.user_pseudo_id = au.user_pseudo_id
)

SELECT
  user_pseudo_id,
  resolved_user_id AS user_id,
  first_event_dt,
  first_event_time_cn,
  last_event_time_cn,
  IF(is_new_by_new_user_table, 'new', 'old') AS new_old_type,
  active_table_is_new,
  resolved_country AS country,
  resolved_device_category AS device_category,
  resolved_source AS source,
  medium,
  campaign,
  traffic_src_type,
  generate_click_cnt,
  ai_service_start_cnt,
  ai_service_success_cnt,
  service_types,
  first_page_location
FROM classified
WHERE generate_click_cnt > 0
   OR ai_service_start_cnt > 0
   OR ai_service_success_cnt > 0
ORDER BY
  ai_service_start_cnt DESC,
  ai_service_success_cnt DESC,
  generate_click_cnt DESC,
  first_event_time_cn;
