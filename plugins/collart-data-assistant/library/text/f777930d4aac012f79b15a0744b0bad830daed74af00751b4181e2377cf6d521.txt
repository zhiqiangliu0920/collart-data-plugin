WITH s AS (
  SELECT * FROM `pubdata2025.dwd.dwd_cdct_revenue_stripe_di`
  WHERE event_date BETWEEN '2026-08-28' AND '2026-09-10' AND app_name='collart_web'
), duplicates AS (
  SELECT transaction_id,COUNT(*) AS n,COUNT(DISTINCT TO_JSON_STRING(STRUCT(event_date,event_timestamp,revneue,user_id,description_type))) AS versions
  FROM s GROUP BY transaction_id
)
SELECT 'stripe_quality' AS section, TO_JSON_STRING(STRUCT(
  COUNT(*) AS raw_rows, COUNT(DISTINCT transaction_id) AS unique_transactions,
  COUNTIF(transaction_id IS NULL) AS missing_transaction,
  COUNTIF(user_id IS NULL) AS missing_user,
  COUNTIF(event_date!=DATE(TIMESTAMP_SECONDS(event_timestamp))) AS utc_date_mismatch,
  COUNTIF(event_date!=DATE(TIMESTAMP_SECONDS(event_timestamp),'Asia/Shanghai')) AS beijing_date_mismatch,
  COUNTIF(revneue<=0) AS nonpositive,
  MIN(event_date) AS min_date,MAX(event_date) AS max_date,
  MAX(TIMESTAMP_SECONDS(event_timestamp)) AS last_transaction_utc
)) AS detail FROM s
UNION ALL
SELECT 'duplicates',TO_JSON_STRING(STRUCT(COUNTIF(n>1) AS repeated_keys,COUNTIF(versions>1) AS conflicting_keys)) FROM duplicates
UNION ALL
SELECT 'internal_orders',TO_JSON_STRING(STRUCT(COUNT(*) AS transactions,CAST(SUM(revneue) AS FLOAT64) AS net_revenue))
FROM s WHERE EXISTS(SELECT 1 FROM UNNEST(@internal_user_ids) i WHERE UPPER(i)=UPPER(s.user_id))
UNION ALL
SELECT 'status_mix',TO_JSON_STRING(STRUCT(charge_type,order_status,COUNT(*) AS transactions,CAST(SUM(revneue) AS FLOAT64) AS net_revenue))
FROM s GROUP BY charge_type,order_status;
