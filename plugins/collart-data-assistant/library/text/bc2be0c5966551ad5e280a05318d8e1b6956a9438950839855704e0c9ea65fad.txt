BEGIN TRANSACTION;

DELETE FROM `aidata2025.ads_collartfashion.ads_oper_basic_indicator_daily_di`
WHERE event_date BETWEEN from_date AND to_date
  AND package_name = 'collart_fashion';

INSERT INTO `aidata2025.ads_collartfashion.ads_oper_basic_indicator_daily_di`
(
  event_date,
  package_name,
  dau,
  vip_dau,
  churn_uv,
  return_uv,
  dnu,
  dnu_vip,
  dnu_free,
  dnu_by_source,
  retain,
  retain2_vip,
  retain2_free,
  retain2_by_source,
  rolling_retain,
  subscription,
  revenue,
  delivery,
  welcome_show,
  page_view,

  video_generate_click,
  generate_start,

  ai_execute_start,
  ai_execute_success,
  fashion_ai_service_start,
  fashion_ai_service_success,
  vip_subscribe_succeed

WITH basic AS (
  SELECT
    event_date,
    SUM(dau) AS dau,
    SUM(vip_dau) AS vip_dau,
    SUM(churn_uv) AS churn_uv,
    SUM(return_uv) AS return_uv,
    SUM(dnu) AS dnu,
    SUM(dnu_vip) AS dnu_vip,
    SUM(dnu_free) AS dnu_free,
    STRUCT(
      COALESCE(SUM(dnu_by_source.organic), 0) AS organic,
      COALESCE(SUM(dnu_by_source.delivery), 0) AS delivery,
      COALESCE(SUM(dnu_by_source.kol), 0) AS kol,
      COALESCE(SUM(dnu_by_source.inhouse), 0) AS inhouse
    ) AS dnu_by_source,
  STRUCT(
    COALESCE(SUM(b.retain.d2), 0) AS d2,
    COALESCE(SUM(b.retain.d3), 0) AS d3,
    COALESCE(SUM(b.retain.d4), 0) AS d4,
    COALESCE(SUM(b.retain.d5), 0) AS d5,
    COALESCE(SUM(b.retain.d6), 0) AS d6,
    COALESCE(SUM(b.retain.d7), 0) AS d7,
    COALESCE(SUM(b.retain.d8), 0) AS d8,
    COALESCE(SUM(b.retain.d9), 0) AS d9,
    COALESCE(SUM(b.retain.d10), 0) AS d10,
    COALESCE(SUM(b.retain.d11), 0) AS d11,
    COALESCE(SUM(b.retain.d12), 0) AS d12,
    COALESCE(SUM(b.retain.d13), 0) AS d13,
    COALESCE(SUM(b.retain.d14), 0) AS d14,
    COALESCE(SUM(b.retain.d15), 0) AS d15,
    COALESCE(SUM(b.retain.d16), 0) AS d16,
    COALESCE(SUM(b.retain.d17), 0) AS d17,
    COALESCE(SUM(b.retain.d18), 0) AS d18,
    COALESCE(SUM(b.retain.d19), 0) AS d19,
    COALESCE(SUM(b.retain.d20), 0) AS d20,
    COALESCE(SUM(b.retain.d21), 0) AS d21,
    COALESCE(SUM(b.retain.d22), 0) AS d22,
    COALESCE(SUM(b.retain.d23), 0) AS d23,
    COALESCE(SUM(b.retain.d24), 0) AS d24,
    COALESCE(SUM(b.retain.d25), 0) AS d25,
    COALESCE(SUM(b.retain.d26), 0) AS d26,
    COALESCE(SUM(b.retain.d27), 0) AS d27,
    COALESCE(SUM(b.retain.d28), 0) AS d28,
    COALESCE(SUM(b.retain.d29), 0) AS d29,
    COALESCE(SUM(b.retain.d30), 0) AS d30
  ) AS retain,
    COALESCE(SUM(retain2_vip), 0) AS retain2_vip,
    COALESCE(SUM(retain2_free), 0) AS retain2_free,
    STRUCT(
      COALESCE(SUM(retain2_by_source.organic), 0) AS organic,
      COALESCE(SUM(retain2_by_source.delivery), 0) AS delivery,
      COALESCE(SUM(retain2_by_source.kol), 0) AS kol,
      COALESCE(SUM(retain2_by_source.inhouse), 0) AS inhouse
    ) AS retain2_by_source,
  STRUCT(
    COALESCE(SUM(b.rolling_retain.d2), 0) AS d2,
    COALESCE(SUM(b.rolling_retain.d3), 0) AS d3,
    COALESCE(SUM(b.rolling_retain.d4), 0) AS d4,
    COALESCE(SUM(b.rolling_retain.d5), 0) AS d5,
    COALESCE(SUM(b.rolling_retain.d6), 0) AS d6,
    COALESCE(SUM(b.rolling_retain.d7), 0) AS d7,
    COALESCE(SUM(b.rolling_retain.d8), 0) AS d8,
    COALESCE(SUM(b.rolling_retain.d9), 0) AS d9,
    COALESCE(SUM(b.rolling_retain.d10), 0) AS d10,
    COALESCE(SUM(b.rolling_retain.d11), 0) AS d11,
    COALESCE(SUM(b.rolling_retain.d12), 0) AS d12,
    COALESCE(SUM(b.rolling_retain.d13), 0) AS d13,
    COALESCE(SUM(b.rolling_retain.d14), 0) AS d14,
    COALESCE(SUM(b.rolling_retain.d15), 0) AS d15,
    COALESCE(SUM(b.rolling_retain.d16), 0) AS d16,
    COALESCE(SUM(b.rolling_retain.d17), 0) AS d17,
    COALESCE(SUM(b.rolling_retain.d18), 0) AS d18,
    COALESCE(SUM(b.rolling_retain.d19), 0) AS d19,
    COALESCE(SUM(b.rolling_retain.d20), 0) AS d20,
    COALESCE(SUM(b.rolling_retain.d21), 0) AS d21,
    COALESCE(SUM(b.rolling_retain.d22), 0) AS d22,
    COALESCE(SUM(b.rolling_retain.d23), 0) AS d23,
    COALESCE(SUM(b.rolling_retain.d24), 0) AS d24,
    COALESCE(SUM(b.rolling_retain.d25), 0) AS d25,
    COALESCE(SUM(b.rolling_retain.d26), 0) AS d26,
    COALESCE(SUM(b.rolling_retain.d27), 0) AS d27,
    COALESCE(SUM(b.rolling_retain.d28), 0) AS d28,
    COALESCE(SUM(b.rolling_retain.d29), 0) AS d29,
    COALESCE(SUM(b.rolling_retain.d30), 0) AS d30
  ) AS rolling_retain,
    STRUCT(
      COALESCE(SUM(subscription.subscribe_uv), 0) AS subscribe_uv,
      COALESCE(SUM(subscription.subscribe_uv_1d), 0) AS subscribe_uv_1d,
      COALESCE(SUM(subscription.subscribe_uv_old), 0) AS subscribe_uv_old,
      COALESCE(SUM(subscription.subscribe_uv_7d), 0) AS subscribe_uv_7d,
      COALESCE(SUM(subscription.credit_pack_uv), 0) AS credit_pack_uv,
      COALESCE(SUM(subscription.credit_pack_uv_1d), 0) AS credit_pack_uv_1d,
      COALESCE(SUM(subscription.credit_pack_uv_old), 0) AS credit_pack_uv_old,
      COALESCE(SUM(subscription.credit_pack_uv_7d), 0) AS credit_pack_uv_7d,
      COALESCE(SUM(subscription.credit_pack_cnt), 0) AS credit_pack_cnt,
      COALESCE(SUM(subscription.credit_amount), 0) AS credit_amount,
      COALESCE(SUM(subscription.purchase_uv), 0) AS purchase_uv,
      COALESCE(SUM(subscription.purchase_uv_1d), 0) AS purchase_uv_1d
    ) AS subscription,
    STRUCT(
      COALESCE(SUM(revenue.purchase_revenue), 0.0) AS purchase_revenue,
      SUM(revenue.resub_revenue) AS resub_revenue,
      SUM(revenue.new_revenue) AS new_revenue,
      COALESCE(SUM(revenue.credit_revenue), 0.0) AS credit_revenue,
      COALESCE(SUM(revenue.ad_revenue), 0.0) AS ad_revenue,
      COALESCE(SUM(revenue.new_subscribe_uv), 0) AS new_subscribe_uv,
      COALESCE(SUM(revenue.renew_subscribe_uv), 0) AS renew_subscribe_uv,
      COALESCE(SUM(revenue.credit_pack_uv), 0) AS credit_pack_uv,
      COALESCE(SUM(revenue.ad_uv), 0) AS ad_uv,
      COALESCE(SUM(revenue.revenue), 0.0) AS revenue
    ) AS revenue,
    STRUCT(
      COALESCE(SUM(delivery.cost), 0.0) AS cost,
      COALESCE(SUM(delivery.impressions), 0) AS impressions,
      COALESCE(SUM(delivery.clicks), 0) AS clicks,
      COALESCE(SUM(delivery.installs), 0) AS installs
    ) AS delivery
  FROM `aidata2025.ads_collartfashion.ads_oper_basic_indicator_country_di` AS b
  WHERE event_date BETWEEN from_date AND to_date
  GROUP BY event_date
),

evt AS (
  SELECT
    event_date,
  STRUCT(
    COALESCE(SUM(welcome_show.uv), 0) AS uv,
    COALESCE(SUM(welcome_show.pv), 0) AS pv
  ) AS welcome_show,  -- 欢迎页曝光
  STRUCT(
    COALESCE(SUM(page_view.uv), 0) AS uv,
    COALESCE(SUM(page_view.pv), 0) AS pv
  ) AS page_view,  -- 页面浏览
  STRUCT(
    COALESCE(SUM(inspiration_template_show.uv), 0) AS uv,
    COALESCE(SUM(inspiration_template_show.pv), 0) AS pv
  ) AS inspiration_template_show,  -- 灵感模板曝光
  STRUCT(
    COALESCE(SUM(generator_show.uv), 0) AS uv,
    COALESCE(SUM(generator_show.pv), 0) AS pv
  ) AS generator_show,  -- 生成器曝光
  STRUCT(
    COALESCE(SUM(video_generate_click.uv), 0) AS uv,
    COALESCE(SUM(video_generate_click.pv), 0) AS pv
  ) AS video_generate_click,  -- 视频生成点击
  STRUCT(
    COALESCE(SUM(generate_start.uv), 0) AS uv,
    COALESCE(SUM(generate_start.pv), 0) AS pv
  ) AS generate_start,  -- AI 服务开始
  STRUCT(
    COALESCE(SUM(generate_success.uv), 0) AS uv,
    COALESCE(SUM(generate_success.pv), 0) AS pv
  ) AS generate_success,  -- AI 服务成功
  STRUCT(
    COALESCE(SUM(ai_execute_start.uv), 0) AS uv,
    COALESCE(SUM(ai_execute_start.pv), 0) AS pv
  ) AS ai_execute_start,  -- AI 执行开始
  STRUCT(
    COALESCE(SUM(ai_execute_success.uv), 0) AS uv,
    COALESCE(SUM(ai_execute_success.pv), 0) AS pv
  ) AS ai_execute_success,  -- AI 执行成功
  STRUCT(
    COALESCE(SUM(fashion_ai_service_start.uv), 0) AS uv,
    COALESCE(SUM(fashion_ai_service_start.pv), 0) AS pv
  ) AS fashion_ai_service_start,  -- Fashion 视频服务开始
  STRUCT(
    COALESCE(SUM(fashion_ai_service_success.uv), 0) AS uv,
    COALESCE(SUM(fashion_ai_service_success.pv), 0) AS pv
  ) AS fashion_ai_service_success,  -- Fashion 视频服务成功
  STRUCT(
    COALESCE(SUM(vip_subscribe_succeed.uv), 0) AS uv,
    COALESCE(SUM(vip_subscribe_succeed.pv), 0) AS pv
  ) AS vip_subscribe_succeed  -- VIP 订阅成功
  FROM `aidata2025.ads_collartfashion.ads_oper_event_metric_country_di`
  WHERE event_date BETWEEN from_date AND to_date
  GROUP BY event_date
)

SELECT
  COALESCE(b.event_date, e.event_date) AS event_date,
  'collart_fashion' AS package_name,
  COALESCE(b.dau, 0) AS dau,
  COALESCE(b.vip_dau, 0) AS vip_dau,
  COALESCE(b.churn_uv, 0) AS churn_uv,
  COALESCE(b.return_uv, 0) AS return_uv,
  COALESCE(b.dnu, 0) AS dnu,
  COALESCE(b.dnu_vip, 0) AS dnu_vip,
  COALESCE(b.dnu_free, 0) AS dnu_free,
  b.dnu_by_source,
  b.retain,
  COALESCE(b.retain2_vip, 0) AS retain2_vip,
  COALESCE(b.retain2_free, 0) AS retain2_free,
  b.retain2_by_source,
  b.rolling_retain,
  b.subscription,
  b.revenue,
  b.delivery,
  COALESCE(e.welcome_show, STRUCT(0, 0)) AS welcome_show,
  COALESCE(e.page_view, STRUCT(0, 0)) AS page_view,
  COALESCE(e.inspiration_template_show, STRUCT(0, 0)) AS inspiration_template_show,
  COALESCE(e.generator_show, STRUCT(0, 0)) AS generator_show,
  COALESCE(e.video_generate_click, STRUCT(0, 0)) AS video_generate_click,
  COALESCE(e.generate_start, STRUCT(0, 0)) AS generate_start,
  COALESCE(e.generate_success, STRUCT(0, 0)) AS generate_success,
  COALESCE(e.ai_execute_start, STRUCT(0, 0)) AS ai_execute_start,
  COALESCE(e.ai_execute_success, STRUCT(0, 0)) AS ai_execute_success,
  COALESCE(e.fashion_ai_service_start, STRUCT(0, 0)) AS fashion_ai_service_start,
  COALESCE(e.fashion_ai_service_success, STRUCT(0, 0)) AS fashion_ai_service_success,
  COALESCE(e.vip_subscribe_succeed, STRUCT(0, 0)) AS vip_subscribe_succeed
FROM basic AS b
FULL OUTER JOIN evt AS e
  USING (event_date);

COMMIT TRANSACTION;
