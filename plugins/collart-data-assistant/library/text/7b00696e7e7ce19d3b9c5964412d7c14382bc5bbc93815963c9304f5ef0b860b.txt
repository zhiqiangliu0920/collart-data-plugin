-- ============================================================================
-- Collart iOS（VidArt）每日 ASA 投放成本
--
-- 成本源：aidata2025.dws.dws_oper_asa_delivery_di
--         （definitions/dws/daily/h_dws_oper_asa_delivery_di.sqlx）
--
-- 新表已内置以下口径，故不再需要手工处理：
--   · 整行去重（源 ODS 自 2026-05 起有 2~9 倍重复行）
--   · CNY → USD 换算（ASA 账户币种是人民币，直接给 cost_usd / cost_cny 两列）
--   · Max Conversion 等无关键词投放的成本（is_keyword_attributed = FALSE）
--   · 多产品隔离（同一 ASA 账号还有 Collart / RoomAI / RemoveBG / Imagix 等）
--
-- 实测 2026-07-18 ~ 08-06 与 Apple campaign 报表花费、安装均 diff = 0
-- 数据起点：2026-05-19
-- ============================================================================

SELECT
  event_date,
  ROUND(SUM(cost_usd), 2) AS cost_usd,     -- 美元（与收入同币种）
  ROUND(SUM(cost_cny), 2) AS cost_cny,     -- 账户原币（人民币）
  SUM(impressions) AS impression,
  SUM(clicks)      AS clicks,
  SUM(installs)    AS install,
  ROUND(SAFE_DIVIDE(SUM(cost_usd), SUM(installs)), 2) AS cpi_usd
FROM `aidata2025.dws.dws_oper_asa_delivery_di`
WHERE event_date >= DATE '2026-05-19'
  AND package_name = 'ai.photo.video.generator.fotos.ai.image.picture.editor.app.free'
GROUP BY event_date
ORDER BY event_date DESC
