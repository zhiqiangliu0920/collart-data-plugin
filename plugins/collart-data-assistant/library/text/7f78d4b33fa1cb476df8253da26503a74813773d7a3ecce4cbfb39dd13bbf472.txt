-- ============================================
-- 紧急验证：订阅追踪是否失效
-- ============================================

-- 验证1: 检查 subscription_first 事件是否存在
SELECT 
  'subscription_first事件存在性' as check_item,
  COUNT(*) as event_count,
  MIN(dt) as earliest_date,
  MAX(dt) as latest_date
FROM `storytemplate-10a27.analytics_232977577.events_2026*`
WHERE _TABLE_SUFFIX BETWEEN '0401' AND '0421'
  AND event_name = 'subscription_first'
  AND platform = 'ANDROID';

-- 验证2: 检查是否有任何订阅相关事件
SELECT 
  event_name,
  COUNT(*) as event_count,
  COUNT(DISTINCT user_pseudo_id) as user_count
FROM `storytemplate-10a27.analytics_232977577.events_2026*`
WHERE _TABLE_SUFFIX BETWEEN '0401' AND '0421'
  AND platform = 'ANDROID'
  AND (event_name LIKE '%subscription%' 
       OR event_name LIKE '%purchase%'
       OR event_name LIKE '%pay%'
       OR event_name LIKE '%subscribe%')
GROUP BY event_name
ORDER BY event_count DESC;

-- 验证3: 检查新增用户中是否有后续付费行为（跨天追踪）
WITH new_users AS (
  SELECT 
    user_pseudo_id,
    MIN(dt) as first_open_date
  FROM `storytemplate-10a27.analytics_232977577.events_2026*`
  WHERE _TABLE_SUFFIX BETWEEN '0401' AND '0421'
    AND event_name = 'first_open'
    AND platform = 'ANDROID'
  GROUP BY user_pseudo_id
),
paid_users AS (
  SELECT 
    user_pseudo_id,
    MIN(dt) as first_pay_date
  FROM `storytemplate-10a27.analytics_232977577.events_2026*`
  WHERE _TABLE_SUFFIX BETWEEN '0401' AND '0421'
    AND platform = 'ANDROID'
    AND event_name IN ('subscription_first', 'purchase', 'subscription_success')
  GROUP BY user_pseudo_id
)
SELECT 
  COUNT(DISTINCT n.user_pseudo_id) as total_new_users,
  COUNT(DISTINCT p.user_pseudo_id) as paid_users_count,
  ROUND(100.0 * COUNT(DISTINCT p.user_pseudo_id) / NULLIF(COUNT(DISTINCT n.user_pseudo_id), 0), 2) as actual_pay_rate,
  -- 次日付费占比
  COUNT(DISTINCT CASE WHEN DATE_DIFF(p.first_pay_date, n.first_open_date, DAY) = 0 THEN p.user_pseudo_id END) as same_day_payers,
  COUNT(DISTINCT CASE WHEN DATE_DIFF(p.first_pay_date, n.first_open_date, DAY) = 1 THEN p.user_pseudo_id END) as next_day_payers
FROM new_users n
LEFT JOIN paid_users p ON n.user_pseudo_id = p.user_pseudo_id;

-- 验证4: 按天查看付费事件分布（看是否有数据）
SELECT 
  dt,
  COUNT(DISTINCT user_pseudo_id) as payer_count,
  COUNT(*) as pay_event_count
FROM `storytemplate-10a27.analytics_232977577.events_2026*`
WHERE _TABLE_SUFFIX BETWEEN '0401' AND '0421'
  AND platform = 'ANDROID'
  AND event_name IN ('subscription_first', 'purchase', 'subscription_success')
GROUP BY dt
ORDER BY dt DESC;
