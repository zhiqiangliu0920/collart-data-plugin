#!/usr/bin/env python3
"""
验证订阅追踪是否存在
"""

from google.cloud import bigquery

client = bigquery.Client(project="storytemplate-10a27")

# 检查 subscription_first 事件
sql = """
SELECT 
  'subscription_first事件' as check_item,
  COUNT(*) as event_count,
  COUNT(DISTINCT user_pseudo_id) as user_count,
  MIN(dt) as earliest_date,
  MAX(dt) as latest_date
FROM `storytemplate-10a27.analytics_232977577.events_2026*`
WHERE _TABLE_SUFFIX BETWEEN '0401' AND '0421'
  AND event_name = 'subscription_first'
  AND platform = 'ANDROID'
"""

print("=" * 60)
print("验证 subscription_first 事件是否存在")
print("=" * 60)
print()

try:
    df = client.query(sql).to_dataframe()
    if not df.empty and df['event_count'].iloc[0] > 0:
        print(f"✅ 事件存在!")
        print(f"   事件数: {df['event_count'].iloc[0]:,}")
        print(f"   用户数: {df['user_count'].iloc[0]:,}")
        print(f"   日期范围: {df['earliest_date'].iloc[0]} 至 {df['latest_date'].iloc[0]}")
    else:
        print("❌ 未找到 subscription_first 事件!")
        print("   这说明首订追踪完全失效")
except Exception as e:
    print(f"查询失败: {e}")

print()
print("=" * 60)
