-- =============================================
-- Vibedance 新产品数据探索 - Step 1
-- 目的: 摸清事件命名、数据范围、核心事件
-- 执行: 拿到权限后先跑这组，5分钟摸清底
-- =============================================
-- 项目: vibedance-540a0
-- 数据集: analytics_533044528
-- 表: events_* (日分区)

-- ---------------------------------------------
-- 1. 查看数据时间范围 & 数据量
-- ---------------------------------------------
SELECT
  MIN(_TABLE_SUFFIX) AS earliest_date,
  MAX(_TABLE_SUFFIX) AS latest_date,
  COUNT(DISTINCT _TABLE_SUFFIX) AS total_days
FROM `vibedance-540a0.analytics_533044528.events_*`
WHERE _TABLE_SUFFIX >= '20260101';

-- ---------------------------------------------
-- 2. 查看所有事件名称 (TOP 50)
-- ---------------------------------------------
SELECT
  event_name,
  COUNT(*) AS event_count,
  COUNT(DISTINCT user_pseudo_id) AS user_count
FROM `vibedance-540a0.analytics_533044528.events_*`
WHERE _TABLE_SUFFIX >= '20260101'
GROUP BY event_name
ORDER BY event_count DESC
LIMIT 50;

-- ---------------------------------------------
-- 3. 查看事件参数有哪些 (抽样最近1天)
-- ---------------------------------------------
SELECT
  event_name,
  ep.key AS param_key,
  COUNT(*) AS param_count
FROM `vibedance-540a0.analytics_533044528.events_*`,
UNNEST(event_params) AS ep
WHERE _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 3 DAY))
GROUP BY event_name, ep.key
ORDER BY event_name, param_count DESC;

-- ---------------------------------------------
-- 4. 查看用户属性字段 (device / geo / app_info)
-- ---------------------------------------------
SELECT
  device.operating_system AS os,
  device.mobile_brand_name AS brand,
  COUNT(DISTINCT user_pseudo_id) AS users
FROM `vibedance-540a0.analytics_533044528.events_*`
WHERE _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 7 DAY))
GROUP BY device.operating_system, device.mobile_brand_name
ORDER BY users DESC
LIMIT 20;

-- ---------------------------------------------
-- 5. 查看崩溃/错误事件 (如果有的话)
-- ---------------------------------------------
SELECT
  event_name,
  COUNT(*) AS count
FROM `vibedance-540a0.analytics_533044528.events_*`
WHERE _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 7 DAY))
  AND (
    event_name LIKE '%error%'
    OR event_name LIKE '%crash%'
    OR event_name LIKE '%exception%'
    OR event_name LIKE '%failed%'
  )
GROUP BY event_name
ORDER BY count DESC;
