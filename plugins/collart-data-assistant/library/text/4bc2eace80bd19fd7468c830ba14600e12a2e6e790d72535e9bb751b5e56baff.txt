"""Validate local audit evidence and report links; no network or cloud mutation."""
import hashlib,json,re
from pathlib import Path
R=Path(__file__).resolve().parent
def read(name):return json.loads((R/name).read_text(encoding='utf-8'))
checks={}
cat=read('task_catalog.json')
checks['14_fashion_actions']=sum(x['category']=='Fashion专属' for x in cat)==14
checks['2_direct_shared_actions']=sum(x['category']=='直接相关共享' for x in cat)==2
checks['4_reference_actions']=sum(x['category'].startswith('参考') for x in cat)==4
sources=read('workspace_sources.json')+read('additional_sources.json')
checks['23_source_hashes_match']=len(sources)==23 and all(hashlib.sha256(s['content'].encode('utf-8')).hexdigest()==s['sha256'] for s in sources)
rr=read('reconciliation.json')['rows']
checks['new_daily_dau_reconciles']=all(r['active_dau']==r['attr_dau']==r['country_dau']==r['daily_dau'] for r in rr)
checks['legacy_nature_mismatch_proved']=([r['active_nature_dnu'] for r in rr]==[70,87,91] and all(r['legacy_nature_dnu']==0 for r in rr))
checks['active_user_day_keys_unique_sample']=all(r['active_rows']==r['pseudo_keys'] for r in rr)
p=read('profile_consistency.json')['rows'][0]
checks['profile_3348_stale_7d']=p['active_days7_mismatch_profiles']==p['inactive_user_days7_mismatch']==3348
checks['profile_revenue_reconciles']=p['revenue_mismatch_profiles']==0
d=[r for r in read('dm_quality.json')['rows'] if r['event_date']=='2026-09-08' and r['package_name']=='collart_fashion'][0]
checks['internal_success_9_of_20']=d['internal_fashion_success']==9 and d['external_fashion_success']==11 and rr[2]['daily_success']['pv']==20
s=read('shared_revenue_reconciliation.json')['rows'][0]
checks['web_contains_fashion_revenue']=abs(s['stripe_web_source_revenue']-s['shared_web_sub_revenue'])<1e-8 and abs(s['fashion_matched_in_stripe_web']-s['fashion_attributed_revenue'])<1e-8 and s['shared_fashion_sub_revenue']==25.34
checks['shared_fashion_all_us']=s['shared_fashion_us_revenue']==25.34 and s['shared_fashion_other_country_revenue']==0
checks['userday_revenue_has_indonesia']=any(x['country']=='Indonesia' and x['revenue_usd']>0 for x in read('revenue_country.json')['rows'])
ledger=[]
for name in ['jobs','reconciliation','raw_table_overlap','dm_quality','profile_consistency','revenue_country','shared_revenue_reconciliation']:
    j=read(name+'.json')
    ledger.append({'sql':name+'.sql','result':name+'.json','job_id':j['job_id'],'bytes_processed':j['bytes_processed'],'bytes_billed':j['bytes_billed'],'slot_ms':j['slot_millis']})
missing=[]
for name in ['fashion_warehouse_review.md','task_inventory.md']:
    content=(R/name).read_text(encoding='utf-8')
    for target in re.findall(r'\]\(<([^>]+)>\)',content):
        path=re.sub(r':\d+$','',target)
        if not Path(path).exists() and not path.endswith('/qa.json'): missing.append(path)
checks['all_local_report_links_exist']=not missing
out={'status':'PASS' if all(checks.values()) else 'FAIL','checks':checks,'missing_links':missing,
     'query_ledger_note':'每份 SQL 的最新成功作业；不包括被覆盖的早次运行，不能用来计算整次审计费用。',
     'latest_query_evidence':ledger,'cloud_changes':'none: read-only metadata/GET/SELECT and dry runs only','old_project_changes':'none'}
(R/'qa.json').write_text(json.dumps(out,ensure_ascii=False,indent=2),encoding='utf-8')
print(json.dumps({'status':out['status'],'checks':checks,'missing_links':missing},ensure_ascii=False,indent=2))
if not all(checks.values()): raise SystemExit(1)
