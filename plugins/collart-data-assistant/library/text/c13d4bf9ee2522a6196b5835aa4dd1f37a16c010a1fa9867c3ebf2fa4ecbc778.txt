-- 新老用户触发 explore 转化率分析
-- 项目：collart web
-- 输出：日期、新用户数、新用户触发explore uv；老用户数、老用户触发explore uv
-- 注意：BigQuery 不支持中文别名，以下使用英文别名

WITH 
new_user_pool AS (
  SELECT 
    event_date,
    user_pseudo_id
  FROM `aidata2025.ads_collart.ads_oper_user_new_collartweb_di`
  WHERE event_date >= DATE_SUB(CURRENT_DATE(), INTERVAL 30 DAY)
),

all_user_behavior AS (
  SELECT 
    PARSE_DATE('%Y%m%d', event_date) AS event_date,
    user_pseudo_id,
    MAX(CASE WHEN event_name = 'explore_page_show' THEN 1 ELSE 0 END) AS has_explored
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 30 DAY))
    AND FORMAT_DATE('%Y%m%d', CURRENT_DATE())
    AND app_info.id IS NULL
  GROUP BY event_date, user_pseudo_id
)

SELECT 
  a.event_date AS date,
  COUNT(DISTINCT CASE WHEN n.user_pseudo_id IS NOT NULL THEN a.user_pseudo_id END) AS new_user_cnt,
  COUNT(DISTINCT CASE WHEN n.user_pseudo_id IS NOT NULL AND a.has_explored = 1 THEN a.user_pseudo_id END) AS new_user_explore_uv,
  COUNT(DISTINCT CASE WHEN n.user_pseudo_id IS NULL THEN a.user_pseudo_id END) AS old_user_cnt,
  COUNT(DISTINCT CASE WHEN n.user_pseudo_id IS NULL AND a.has_explored = 1 THEN a.user_pseudo_id END) AS old_user_explore_uv,
  ROUND(COUNT(DISTINCT CASE WHEN n.user_pseudo_id IS NOT NULL AND a.has_explored = 1 THEN a.user_pseudo_id END) * 100.0 / 
        NULLIF(COUNT(DISTINCT CASE WHEN n.user_pseudo_id IS NOT NULL THEN a.user_pseudo_id END), 0), 2) AS new_user_explore_rate,
  ROUND(COUNT(DISTINCT CASE WHEN n.user_pseudo_id IS NULL AND a.has_explored = 1 THEN a.user_pseudo_id END) * 100.0 / 
        NULLIF(COUNT(DISTINCT CASE WHEN n.user_pseudo_id IS NULL THEN a.user_pseudo_id END), 0), 2) AS old_user_explore_rate
FROM all_user_behavior a
LEFT JOIN new_user_pool n ON a.user_pseudo_id = n.user_pseudo_id AND a.event_date = n.event_date
GROUP BY a.event_date
ORDER BY a.event_date DESC
