-- DAU 查询（Android）
-- 更新时间: 2026-03-23

-- 方法1：使用 ads_oper_basic_indicator_collart_di（推荐）
SELECT 
  dt,
  dau
FROM `aidata2025.ads_collart.ads_oper_basic_indicator_collart_di`
WHERE dt >= '2026-03-01' AND dt <= '2026-03-07'
ORDER BY dt;

-- 方法2：使用 dwd_oper_user_collart_di
SELECT 
  dt,
  COUNT(DISTINCT user_id) AS dau
FROM `aidata2025.dwd.dwd_oper_user_collart_di`
WHERE dt >= '2026-03-01' AND dt <= '2026-03-07'
GROUP BY dt
ORDER BY dt;
