-- Collart Web：Fashion 用户 vs 主站用户划分
--
-- 口径（2026-07-22）：
--   【events 原始表】Fashion：page_location 含 studio 或 fashion（不区分大小写）
--                     主站：  从未出现上述 page_location
--   【DWD/ADS 加工表】主站：  package_name = 'collart_web'
--                     Fashion：package_name = 'collart_fashion'
--
-- 文档：collart_web/indicators/fashion_vs_main_site_user.md

CREATE TEMP FUNCTION is_fashion_page(url STRING) AS (
  REGEXP_CONTAINS(LOWER(COALESCE(url, '')), r'(studio|fashion)')
);
