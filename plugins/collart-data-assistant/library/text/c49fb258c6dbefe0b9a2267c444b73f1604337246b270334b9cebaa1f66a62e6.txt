-- Collart Web 新用户标准口径：user_pseudo_id 首次发现 + 首次落地页
-- 主键粒度: user_pseudo_id
-- 新用户日期: 该 pseudo 在 Web 埋点中最早 event_date（不用 first_visit）
-- 首次落地页: 最早 page_view 的 page_location（来自 GA4 events_*）
-- Web 过滤: app_info.id IS NULL / package_name = 'collart_web'
-- 凭证: aidata2025

-- ============================================================
-- 参数区（按需修改）
-- ============================================================
DECLARE scan_start DATE DEFAULT DATE '2026-01-01';  -- 首次发现扫描起点（历史越早越好，注意成本）
DECLARE scan_end   DATE DEFAULT DATE '2026-07-20';  -- 扫描结束日
DECLARE target_date DATE DEFAULT DATE '2026-07-20'; -- 示例：查询该日新用户

-- ============================================================
-- Step 1: pseudo 首次发现（来自 DWD 埋点轻汇总，成本低）
-- ============================================================
WITH first_seen AS (
  SELECT
    user_pseudo_id,
    MIN(event_date) AS new_user_date,
    MIN(event_time) AS first_seen_time,
    COUNT(*) AS total_events_to_date
  FROM `aidata2025.dwd.dwd_oper_user_event_di`
  WHERE package_name = 'collart_web'
    AND event_date BETWEEN scan_start AND scan_end
    AND user_pseudo_id IS NOT NULL
  GROUP BY user_pseudo_id
),

-- ============================================================
-- Step 2: 首次落地页（来自 GA4 events_*，DWD 无 page_location）
-- ============================================================
page_events AS (
  SELECT
    user_pseudo_id,
    event_timestamp,
    (
      SELECT ep.value.string_value
      FROM UNNEST(event_params) ep
      WHERE ep.key = 'page_location'
    ) AS page_location
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN FORMAT_DATE('%Y%m%d', scan_start)
                          AND FORMAT_DATE('%Y%m%d', scan_end)
    AND app_info.id IS NULL
    AND event_name = 'page_view'
    AND user_pseudo_id IS NOT NULL
),

first_landing AS (
  SELECT
    user_pseudo_id,
    ARRAY_AGG(page_location IGNORE NULLS ORDER BY event_timestamp LIMIT 1)[SAFE_OFFSET(0)] AS first_landing_page,
    REGEXP_EXTRACT(
      LOWER(ARRAY_AGG(page_location IGNORE NULLS ORDER BY event_timestamp LIMIT 1)[SAFE_OFFSET(0)]),
      r'https?://[^/]+(/[^?#]*)'
    ) AS first_landing_path
  FROM page_events
  WHERE COALESCE(page_location, '') != ''
  GROUP BY user_pseudo_id
),

-- ============================================================
-- Step 3: user_id 多对多映射（补充，不作新用户主键）
-- ============================================================
pseudo_user_map AS (
  SELECT
    user_pseudo_id,
    ARRAY_AGG(DISTINCT NULLIF(user_id, '') IGNORE NULLS) AS linked_user_ids,
    ARRAY_AGG(DISTINCT NULLIF(user_id, '') IGNORE NULLS ORDER BY NULLIF(user_id, '') LIMIT 1)[SAFE_OFFSET(0)] AS latest_user_id
  FROM `aidata2025.dwd.dwd_oper_user_event_di`
  WHERE package_name = 'collart_web'
    AND event_date BETWEEN scan_start AND scan_end
    AND user_pseudo_id IS NOT NULL
  GROUP BY user_pseudo_id
),

-- ============================================================
-- Step 4: 首次发现日属性（国家/设备/渠道，取当天 DWD 快照）
-- ============================================================
new_day_attrs AS (
  SELECT
    d.user_pseudo_id,
    d.event_date,
    ARRAY_AGG(d.country IGNORE NULLS ORDER BY d.event_time LIMIT 1)[SAFE_OFFSET(0)] AS country,
    ARRAY_AGG(d.device_category IGNORE NULLS ORDER BY d.event_time LIMIT 1)[SAFE_OFFSET(0)] AS device_category,
    ARRAY_AGG(d.traffic_src_source IGNORE NULLS ORDER BY d.event_time LIMIT 1)[SAFE_OFFSET(0)] AS traffic_src_source,
    ARRAY_AGG(d.traffic_src_name IGNORE NULLS ORDER BY d.event_time LIMIT 1)[SAFE_OFFSET(0)] AS traffic_src_name,
    ARRAY_AGG(d.traffic_src_type IGNORE NULLS ORDER BY d.event_time LIMIT 1)[SAFE_OFFSET(0)] AS traffic_src_type,
    LOGICAL_OR(COALESCE(d.is_login, '') = 'true') AS is_login_on_new_day
  FROM `aidata2025.dwd.dwd_oper_user_event_di` d
  INNER JOIN first_seen f
    ON d.user_pseudo_id = f.user_pseudo_id
   AND d.event_date = f.new_user_date
  WHERE d.package_name = 'collart_web'
  GROUP BY d.user_pseudo_id, d.event_date
),

-- ============================================================
-- Step 5: 新用户维表（主输出）
-- ============================================================
new_user_dim AS (
  SELECT
    f.user_pseudo_id,
    f.new_user_date,
    f.first_seen_time,
    fl.first_landing_page,
    fl.first_landing_path,
    a.country,
    a.device_category,
    a.traffic_src_source,
    a.traffic_src_name,
    a.traffic_src_type,
    a.is_login_on_new_day,
    m.linked_user_ids,
    m.latest_user_id,
    ARRAY_LENGTH(COALESCE(m.linked_user_ids, [])) AS linked_user_id_cnt
  FROM first_seen f
  LEFT JOIN first_landing fl ON f.user_pseudo_id = fl.user_pseudo_id
  LEFT JOIN new_day_attrs a
    ON f.user_pseudo_id = a.user_pseudo_id
   AND f.new_user_date = a.event_date
  LEFT JOIN pseudo_user_map m ON f.user_pseudo_id = m.user_pseudo_id
)

-- ============================================================
-- 用法 A: 某日新用户列表
-- ============================================================
SELECT *
FROM new_user_dim
WHERE new_user_date = target_date
ORDER BY first_seen_time, user_pseudo_id;
