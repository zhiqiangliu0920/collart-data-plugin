config {
  type: "operations",
  name: "ads_collartfashion_oper_event_metric_attr_di",
  schema: "ads_collartfashion",
  dependencies: ["ads_collartfashion_oper_user_event_metric_di", "ads_collartfashion_oper_user_active_di"]
}
DECLARE from_date DATE DEFAULT ${dataform.projectConfig.vars.biz_date} - 30;
DECLARE to_date   DATE DEFAULT ${dataform.projectConfig.vars.biz_date};
CREATE TABLE IF NOT EXISTS `aidata2025.ads_collartfashion.ads_oper_event_metric_attr_di`
(
  event_date DATE, is_new BOOL, is_vip BOOL, country STRING,
  traffic_src_type STRING, traffic_src_platform STRING, traffic_src_name STRING, app_version STRING,
  welcome_show STRUCT<uv INT64, pv INT64>,
  page_view STRUCT<uv INT64, pv INT64>,
  inspiration_template_show STRUCT<uv INT64, pv INT64>,
  generator_show STRUCT<uv INT64, pv INT64>,
  video_generate_click STRUCT<uv INT64, pv INT64>,
  generate_start STRUCT<uv INT64, pv INT64>,
  generate_success STRUCT<uv INT64, pv INT64>,
  ai_execute_start STRUCT<uv INT64, pv INT64>,
  ai_execute_success STRUCT<uv INT64, pv INT64>,
  fashion_ai_service_start STRUCT<uv INT64, pv INT64>,
  fashion_ai_service_success STRUCT<uv INT64, pv INT64>,
  vip_subscribe_succeed STRUCT<uv INT64, pv INT64>
)
PARTITION BY event_date CLUSTER BY country, is_new
OPTIONS(description='ads_collartfashion 日×维度事件指标（新建）');
BEGIN TRANSACTION;
DELETE FROM `aidata2025.ads_collartfashion.ads_oper_event_metric_attr_di` WHERE event_date BETWEEN from_date AND to_date;
INSERT INTO `aidata2025.ads_collartfashion.ads_oper_event_metric_attr_di`
(event_date, is_new, is_vip, country, traffic_src_type, traffic_src_platform, traffic_src_name, app_version,
  welcome_show,
  page_view,
  inspiration_template_show,
  generator_show,
  video_generate_click,
  generate_start,
  generate_success,
  ai_execute_start,
  ai_execute_success,
  fashion_ai_service_start,
  fashion_ai_service_success,
  vip_subscribe_succeed)
WITH active_base AS (
  SELECT event_date, user_pseudo_id, is_new, (is_vip IN ('1','2')) AS is_vip, country,
    traffic_src_type, traffic_src_platform, traffic_src_name, app_version
  FROM `aidata2025.ads_collartfashion.ads_oper_user_active_di`
  WHERE event_date BETWEEN from_date AND to_date
),
top_version AS (
  SELECT event_date, app_version FROM (
    SELECT event_date, app_version, ROW_NUMBER() OVER (
      PARTITION BY event_date ORDER BY COUNT(DISTINCT user_pseudo_id) DESC, app_version
    ) rn
    FROM active_base GROUP BY 1,2
  ) WHERE rn <= 5
),
joined AS (
  SELECT m.event_date, a.is_new, a.is_vip, a.country, a.traffic_src_type, a.traffic_src_platform,
    a.traffic_src_name, IF(tv.app_version IS NULL, 'other', a.app_version) AS app_version,
    m.* EXCEPT(event_date, user_pseudo_id)
  FROM `aidata2025.ads_collartfashion.ads_oper_user_event_metric_di` m
  JOIN active_base a USING (event_date, user_pseudo_id)
  LEFT JOIN top_version tv ON a.event_date=tv.event_date AND a.app_version=tv.app_version
  WHERE m.event_date BETWEEN from_date AND to_date
)
SELECT event_date, is_new, is_vip, country, traffic_src_type, traffic_src_platform, traffic_src_name, app_version,
  STRUCT(COUNTIF(welcome_show_pv>0), COALESCE(SUM(welcome_show_pv),0)) AS welcome_show,  -- 欢迎页曝光
  STRUCT(COUNTIF(page_view_pv>0), COALESCE(SUM(page_view_pv),0)) AS page_view,  -- 页面浏览
  STRUCT(COUNTIF(inspiration_template_show_pv>0), COALESCE(SUM(inspiration_template_show_pv),0)) AS inspiration_template_show,  -- 灵感模板曝光
  STRUCT(COUNTIF(generator_show_pv>0), COALESCE(SUM(generator_show_pv),0)) AS generator_show,  -- 生成器曝光
  STRUCT(COUNTIF(video_generate_click_pv>0), COALESCE(SUM(video_generate_click_pv),0)) AS video_generate_click,  -- 视频生成点击
  STRUCT(COUNTIF(generate_start_pv>0), COALESCE(SUM(generate_start_pv),0)) AS generate_start,  -- AI 服务开始
  STRUCT(COUNTIF(generate_success_pv>0), COALESCE(SUM(generate_success_pv),0)) AS generate_success,  -- AI 服务成功
  STRUCT(COUNTIF(ai_execute_start_pv>0), COALESCE(SUM(ai_execute_start_pv),0)) AS ai_execute_start,  -- AI 执行开始
  STRUCT(COUNTIF(ai_execute_success_pv>0), COALESCE(SUM(ai_execute_success_pv),0)) AS ai_execute_success,  -- AI 执行成功
  STRUCT(COUNTIF(fashion_ai_service_start_pv>0), COALESCE(SUM(fashion_ai_service_start_pv),0)) AS fashion_ai_service_start,  -- Fashion 视频服务开始
  STRUCT(COUNTIF(fashion_ai_service_success_pv>0), COALESCE(SUM(fashion_ai_service_success_pv),0)) AS fashion_ai_service_success,  -- Fashion 视频服务成功
  STRUCT(COUNTIF(vip_subscribe_succeed_pv>0), COALESCE(SUM(vip_subscribe_succeed_pv),0)) AS vip_subscribe_succeed  -- VIP 订阅成功
FROM joined
GROUP BY 1,2,3,4,5,6,7,8;
COMMIT TRANSACTION;
