-- Collart Android: Russia home template load failure diagnosis
-- Incident onset: approximately 2026-07-24 07:35 Asia/Shanghai
-- GA4 raw-event scan is limited to 2026-07-17 through 2026-07-26.
-- Run each numbered SELECT separately if the BigQuery client only displays one
-- result set.

-- 1. Daily impact and error buckets.
WITH base AS (
  SELECT
    DATE(TIMESTAMP_MICROS(event_timestamp), 'Asia/Shanghai') AS event_date_bj,
    user_pseudo_id,
    event_name,
    (
      SELECT ep.value.string_value
      FROM UNNEST(event_params) ep
      WHERE ep.key = 'error_msg'
    ) AS error_msg
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN '20260717' AND '20260727'
    AND app_info.id = 'free.ai.photo.generator.collart.ai'
    AND geo.country = 'Russia'
    AND event_name IN (
      'user_engagement',
      'screen_view',
      'home_temp_frame_show',
      'home_firsttemp_show',
      'home_category_show',
      'Home_Template_Show',
      'home_temp_show_failed'
    )
)
SELECT
  event_date_bj,
  COUNT(DISTINCT IF(
    event_name IN ('user_engagement', 'screen_view'),
    user_pseudo_id,
    NULL
  )) AS dau,
  COUNT(DISTINCT IF(
    event_name = 'home_category_show',
    user_pseudo_id,
    NULL
  )) AS category_uv,
  COUNT(DISTINCT IF(
    event_name = 'Home_Template_Show',
    user_pseudo_id,
    NULL
  )) AS template_uv,
  COUNT(DISTINCT IF(
    event_name = 'home_temp_show_failed',
    user_pseudo_id,
    NULL
  )) AS fail_uv,
  COUNT(DISTINCT IF(
    event_name = 'home_temp_show_failed'
      AND LOWER(error_msg) LIKE '%timeout%',
    user_pseudo_id,
    NULL
  )) AS timeout_fail_uv,
  COUNT(DISTINCT IF(
    event_name = 'home_temp_show_failed'
      AND LOWER(error_msg) LIKE '%unable to resolve host%',
    user_pseudo_id,
    NULL
  )) AS dns_fail_uv,
  ROUND(
    SAFE_DIVIDE(
      COUNT(DISTINCT IF(
        event_name = 'home_temp_show_failed',
        user_pseudo_id,
        NULL
      )),
      COUNT(DISTINCT IF(
        event_name = 'home_category_show',
        user_pseudo_id,
        NULL
      ))
    ) * 100,
    2
  ) AS fail_to_category_pct
FROM base
WHERE event_date_bj BETWEEN '2026-07-17' AND '2026-07-26'
GROUP BY ALL
ORDER BY event_date_bj;


-- 2. Five-minute onset versus the previous seven days at the same time.
WITH base AS (
  SELECT
    DATE(TIMESTAMP_MICROS(event_timestamp), 'Asia/Shanghai') AS event_date_bj,
    EXTRACT(
      HOUR
      FROM DATETIME(TIMESTAMP_MICROS(event_timestamp), 'Asia/Shanghai')
    ) * 60
      + EXTRACT(
        MINUTE
        FROM DATETIME(TIMESTAMP_MICROS(event_timestamp), 'Asia/Shanghai')
      ) AS minute_of_day_raw,
    user_pseudo_id,
    event_name
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN '20260717' AND '20260725'
    AND app_info.id = 'free.ai.photo.generator.collart.ai'
    AND geo.country = 'Russia'
    AND event_name IN (
      'home_category_show',
      'Home_Template_Show',
      'home_temp_show_failed'
    )
),
five_min AS (
  SELECT
    event_date_bj,
    DIV(minute_of_day_raw, 5) * 5 AS minute_of_day,
    COUNT(DISTINCT IF(
      event_name = 'home_category_show',
      user_pseudo_id,
      NULL
    )) AS category_uv,
    COUNT(DISTINCT IF(
      event_name = 'Home_Template_Show',
      user_pseudo_id,
      NULL
    )) AS template_uv,
    COUNT(DISTINCT IF(
      event_name = 'home_temp_show_failed',
      user_pseudo_id,
      NULL
    )) AS fail_uv
  FROM base
  GROUP BY ALL
)
SELECT
  FORMAT(
    '%02d:%02d',
    DIV(minute_of_day, 60),
    MOD(minute_of_day, 60)
  ) AS time_bj,
  MAX(IF(event_date_bj = '2026-07-24', category_uv, NULL))
    AS actual_category_uv,
  MAX(IF(event_date_bj = '2026-07-24', template_uv, NULL))
    AS actual_template_uv,
  MAX(IF(event_date_bj = '2026-07-24', fail_uv, NULL))
    AS actual_fail_uv,
  ROUND(AVG(IF(
    event_date_bj BETWEEN '2026-07-17' AND '2026-07-23',
    category_uv,
    NULL
  )), 1) AS baseline_category_uv,
  ROUND(AVG(IF(
    event_date_bj BETWEEN '2026-07-17' AND '2026-07-23',
    template_uv,
    NULL
  )), 1) AS baseline_template_uv,
  ROUND(AVG(IF(
    event_date_bj BETWEEN '2026-07-17' AND '2026-07-23',
    fail_uv,
    NULL
  )), 1) AS baseline_fail_uv
FROM five_min
WHERE minute_of_day BETWEEN 6 * 60 AND 15 * 60
GROUP BY minute_of_day
ORDER BY minute_of_day;


-- 3. Exact error-message composition before and after the incident.
WITH base AS (
  SELECT
    CASE
      WHEN TIMESTAMP_MICROS(event_timestamp)
        < TIMESTAMP('2026-07-24 07:35:00+08')
        THEN 'baseline'
      ELSE 'incident'
    END AS period,
    user_pseudo_id,
    COALESCE((
      SELECT ep.value.string_value
      FROM UNNEST(event_params) ep
      WHERE ep.key = 'error_msg'
    ), '(missing)') AS error_msg
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN '20260717' AND '20260727'
    AND app_info.id = 'free.ai.photo.generator.collart.ai'
    AND geo.country = 'Russia'
    AND event_name = 'home_temp_show_failed'
    AND TIMESTAMP_MICROS(event_timestamp)
      >= TIMESTAMP('2026-07-17 07:35:00+08')
    AND TIMESTAMP_MICROS(event_timestamp)
      < TIMESTAMP('2026-07-27 00:00:00+08')
),
agg AS (
  SELECT
    period,
    error_msg,
    COUNT(*) AS pv,
    COUNT(DISTINCT user_pseudo_id) AS uv
  FROM base
  GROUP BY ALL
)
SELECT
  *,
  ROUND(
    SAFE_DIVIDE(pv, SUM(pv) OVER (PARTITION BY period)) * 100,
    2
  ) AS pv_share_pct
FROM agg
QUALIFY ROW_NUMBER() OVER (
  PARTITION BY period
  ORDER BY pv DESC
) <= 20
ORDER BY period, pv DESC;


-- 4. User-segment comparison.
WITH base AS (
  SELECT
    CASE
      WHEN TIMESTAMP_MICROS(event_timestamp)
        < TIMESTAMP('2026-07-24 07:35:00+08')
        THEN 'baseline'
      ELSE 'incident'
    END AS period,
    event_name,
    user_pseudo_id,
    COALESCE(app_info.version, '(unknown)') AS app_version,
    COALESCE(device.mobile_brand_name, '(unknown)') AS device_brand,
    COALESCE(device.mobile_model_name, '(unknown)') AS device_model,
    COALESCE(device.operating_system_version, '(unknown)') AS os_version,
    COALESCE(device.language, '(unknown)') AS language,
    COALESCE(geo.city, '(unknown)') AS city,
    COALESCE(traffic_source.source, '(unknown)') AS traffic_source,
    CASE
      WHEN DATE(
        TIMESTAMP_MICROS(user_first_touch_timestamp),
        'Asia/Shanghai'
      ) = DATE(TIMESTAMP_MICROS(event_timestamp), 'Asia/Shanghai')
        THEN 'new_today'
      WHEN DATE_DIFF(
        DATE(TIMESTAMP_MICROS(event_timestamp), 'Asia/Shanghai'),
        DATE(
          TIMESTAMP_MICROS(user_first_touch_timestamp),
          'Asia/Shanghai'
        ),
        DAY
      ) <= 7
        THEN 'age_1_7d'
      WHEN DATE_DIFF(
        DATE(TIMESTAMP_MICROS(event_timestamp), 'Asia/Shanghai'),
        DATE(
          TIMESTAMP_MICROS(user_first_touch_timestamp),
          'Asia/Shanghai'
        ),
        DAY
      ) <= 30
        THEN 'age_8_30d'
      ELSE 'age_31d_plus'
    END AS user_age_bucket
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN '20260717' AND '20260727'
    AND app_info.id = 'free.ai.photo.generator.collart.ai'
    AND geo.country = 'Russia'
    AND event_name IN ('home_category_show', 'home_temp_show_failed')
    AND TIMESTAMP_MICROS(event_timestamp)
      >= TIMESTAMP('2026-07-17 07:35:00+08')
    AND TIMESTAMP_MICROS(event_timestamp)
      < TIMESTAMP('2026-07-27 00:00:00+08')
),
long AS (
  SELECT
    period,
    event_name,
    user_pseudo_id,
    dim.dimension,
    dim.segment
  FROM base,
  UNNEST([
    STRUCT('app_version' AS dimension, app_version AS segment),
    STRUCT('device_brand', device_brand),
    STRUCT('device_model', device_model),
    STRUCT('os_version', os_version),
    STRUCT('language', language),
    STRUCT('city', city),
    STRUCT('traffic_source', traffic_source),
    STRUCT('user_age_bucket', user_age_bucket)
  ]) dim
),
agg AS (
  SELECT
    period,
    dimension,
    segment,
    COUNT(DISTINCT IF(
      event_name = 'home_category_show',
      user_pseudo_id,
      NULL
    )) AS category_uv,
    COUNT(DISTINCT IF(
      event_name = 'home_temp_show_failed',
      user_pseudo_id,
      NULL
    )) AS fail_uv
  FROM long
  GROUP BY ALL
),
pivoted AS (
  SELECT
    dimension,
    segment,
    MAX(IF(period = 'baseline', category_uv, NULL))
      AS baseline_category_uv,
    MAX(IF(period = 'baseline', fail_uv, NULL)) AS baseline_fail_uv,
    MAX(IF(period = 'incident', category_uv, NULL))
      AS incident_category_uv,
    MAX(IF(period = 'incident', fail_uv, NULL)) AS incident_fail_uv
  FROM agg
  GROUP BY ALL
)
SELECT
  *,
  ROUND(
    SAFE_DIVIDE(baseline_fail_uv, baseline_category_uv) * 100,
    2
  ) AS baseline_fail_rate_pct,
  ROUND(
    SAFE_DIVIDE(incident_fail_uv, incident_category_uv) * 100,
    2
  ) AS incident_fail_rate_pct,
  ROUND(
    SAFE_DIVIDE(
      SAFE_DIVIDE(incident_fail_uv, incident_category_uv),
      SAFE_DIVIDE(baseline_fail_uv, baseline_category_uv)
    ),
    2
  ) AS rate_ratio
FROM pivoted
WHERE incident_category_uv >= 30 OR incident_fail_uv >= 20
QUALIFY ROW_NUMBER() OVER (
  PARTITION BY dimension
  ORDER BY incident_fail_uv DESC
) <= 20
ORDER BY dimension, incident_fail_uv DESC;


-- 5. Country comparison for July 23 versus July 24.
WITH base AS (
  SELECT
    DATE(TIMESTAMP_MICROS(event_timestamp), 'Asia/Shanghai') AS event_date_bj,
    COALESCE(geo.country, '(unknown)') AS country,
    user_pseudo_id,
    event_name,
    (
      SELECT ep.value.string_value
      FROM UNNEST(event_params) ep
      WHERE ep.key = 'error_msg'
    ) AS error_msg
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN '20260723' AND '20260725'
    AND app_info.id = 'free.ai.photo.generator.collart.ai'
    AND event_name IN (
      'home_category_show',
      'Home_Template_Show',
      'home_temp_show_failed'
    )
),
daily AS (
  SELECT
    event_date_bj,
    country,
    COUNT(DISTINCT IF(
      event_name = 'home_category_show',
      user_pseudo_id,
      NULL
    )) AS category_uv,
    COUNT(DISTINCT IF(
      event_name = 'home_temp_show_failed',
      user_pseudo_id,
      NULL
    )) AS fail_uv,
    COUNT(DISTINCT IF(
      event_name = 'home_temp_show_failed'
        AND LOWER(error_msg) LIKE '%timeout%',
      user_pseudo_id,
      NULL
    )) AS timeout_fail_uv
  FROM base
  WHERE event_date_bj IN ('2026-07-23', '2026-07-24')
  GROUP BY ALL
),
pivoted AS (
  SELECT
    country,
    MAX(IF(event_date_bj = '2026-07-23', category_uv, NULL))
      AS category_uv_23,
    MAX(IF(event_date_bj = '2026-07-23', fail_uv, NULL))
      AS fail_uv_23,
    MAX(IF(event_date_bj = '2026-07-24', category_uv, NULL))
      AS category_uv_24,
    MAX(IF(event_date_bj = '2026-07-24', fail_uv, NULL))
      AS fail_uv_24,
    MAX(IF(event_date_bj = '2026-07-24', timeout_fail_uv, NULL))
      AS timeout_fail_uv_24
  FROM daily
  GROUP BY country
)
SELECT
  *,
  ROUND(SAFE_DIVIDE(fail_uv_23, category_uv_23) * 100, 2)
    AS fail_rate_23_pct,
  ROUND(SAFE_DIVIDE(fail_uv_24, category_uv_24) * 100, 2)
    AS fail_rate_24_pct,
  ROUND(
    (
      SAFE_DIVIDE(fail_uv_24, category_uv_24)
      - SAFE_DIVIDE(fail_uv_23, category_uv_23)
    ) * 100,
    2
  ) AS fail_rate_change_pp,
  ROUND(SAFE_DIVIDE(timeout_fail_uv_24, fail_uv_24) * 100, 2)
    AS timeout_share_24_pct
FROM pivoted
WHERE category_uv_24 >= 100
ORDER BY fail_rate_change_pp DESC;
