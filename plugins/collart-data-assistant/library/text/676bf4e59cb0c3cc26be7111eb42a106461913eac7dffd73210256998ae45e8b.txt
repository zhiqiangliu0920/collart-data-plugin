-- 新老用户触发 explore 转化率分析
-- 日期: 2026-04-09
-- 定义: 新用户 = 当日 first_visit，老用户 = 非首次访问

WITH 
-- 当日活跃用户
active_users AS (
  SELECT 
    PARSE_DATE('%Y%m%d', event_date) AS stats_date,
    user_pseudo_id
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 8 DAY))
    AND app_info.id IS NULL
    AND event_name = 'session_start'
  GROUP BY stats_date, user_pseudo_id
),
-- 当日新用户（first_visit）
new_users AS (
  SELECT 
    PARSE_DATE('%Y%m%d', event_date) AS stats_date,
    user_pseudo_id
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 8 DAY))
    AND app_info.id IS NULL
    AND event_name = 'first_visit'
  GROUP BY stats_date, user_pseudo_id
),
-- 触发 explore 的用户
explore_users AS (
  SELECT 
    PARSE_DATE('%Y%m%d', event_date) AS stats_date,
    user_pseudo_id
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX >= FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 8 DAY))
    AND app_info.id IS NULL
    AND event_name = 'explore_page_show'
  GROUP BY stats_date, user_pseudo_id
)
SELECT 
  a.stats_date AS date,
  COUNT(DISTINCT n.user_pseudo_id) AS new_users,
  COUNT(DISTINCT CASE WHEN n.user_pseudo_id IS NOT NULL AND e.user_pseudo_id IS NOT NULL THEN n.user_pseudo_id END) AS new_users_explore,
  ROUND(SAFE_DIVIDE(
    COUNT(DISTINCT CASE WHEN n.user_pseudo_id IS NOT NULL AND e.user_pseudo_id IS NOT NULL THEN n.user_pseudo_id END),
    COUNT(DISTINCT n.user_pseudo_id)
  ) * 100, 2) AS new_user_rate,
  COUNT(DISTINCT CASE WHEN n.user_pseudo_id IS NULL THEN a.user_pseudo_id END) AS old_users,
  COUNT(DISTINCT CASE WHEN n.user_pseudo_id IS NULL AND e.user_pseudo_id IS NOT NULL THEN a.user_pseudo_id END) AS old_users_explore,
  ROUND(SAFE_DIVIDE(
    COUNT(DISTINCT CASE WHEN n.user_pseudo_id IS NULL AND e.user_pseudo_id IS NOT NULL THEN a.user_pseudo_id END),
    COUNT(DISTINCT CASE WHEN n.user_pseudo_id IS NULL THEN a.user_pseudo_id END)
  ) * 100, 2) AS old_user_rate
FROM active_users a
LEFT JOIN new_users n USING (stats_date, user_pseudo_id)
LEFT JOIN explore_users e USING (stats_date, user_pseudo_id)
GROUP BY a.stats_date
ORDER BY a.stats_date DESC
