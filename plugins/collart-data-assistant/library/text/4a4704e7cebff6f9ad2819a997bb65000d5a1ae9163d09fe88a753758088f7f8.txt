-- 功能使用分布
SELECT 
  (
    SELECT ep.value.string_value 
    FROM UNNEST(event_params) ep 
    WHERE ep.key = 'service_type'
  ) AS service_type,
  COUNT(*) AS task_count,
  COUNT(DISTINCT user_pseudo_id) AS user_count
FROM `storytemplate-10a27.analytics_232977577.events_*`
WHERE _TABLE_SUFFIX >= '20260306' AND _TABLE_SUFFIX <= '20260312'
  AND app_info.id = 'free.ai.photo.generator.collart.ai'
  AND event_name = 'ai_service_task_create'
GROUP BY service_type
ORDER BY task_count DESC;
