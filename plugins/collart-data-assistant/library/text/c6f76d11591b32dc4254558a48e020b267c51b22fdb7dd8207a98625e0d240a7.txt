-- Collart Web 导航栏 & Banner 点击率分析
-- 时间范围：过去7天
-- 分母：DAU (session_start 去重用户)

WITH dau_base AS (
  -- DAU 计算：每日活跃用户数（分母）
  SELECT 
    PARSE_DATE('%Y%m%d', event_date) AS dt,
    COUNT(DISTINCT user_pseudo_id) AS dau
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 7 DAY))
                          AND FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 1 DAY))
    AND app_info.id IS NULL
    AND event_name = 'session_start'
  GROUP BY dt
),

nav_click AS (
  -- 左边导航栏点击数
  SELECT 
    PARSE_DATE('%Y%m%d', event_date) AS dt,
    (
      SELECT ep.value.string_value
      FROM UNNEST(event_params) ep
      WHERE ep.key = 'feature'
    ) AS feature,
    COUNT(*) AS click_cnt,
    COUNT(DISTINCT user_pseudo_id) AS click_uv
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 7 DAY))
                          AND FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 1 DAY))
    AND app_info.id IS NULL
    AND event_name = 'navigation_click'
  GROUP BY dt, feature
),

banner_click AS (
  -- Banner 点击数（探索页功能模块 Tab 点击）
  SELECT 
    PARSE_DATE('%Y%m%d', event_date) AS dt,
    (
      SELECT ep.value.string_value
      FROM UNNEST(event_params) ep
      WHERE ep.key = 'type'
    ) AS banner_type,
    COUNT(*) AS click_cnt,
    COUNT(DISTINCT user_pseudo_id) AS click_uv
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 7 DAY))
                          AND FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 1 DAY))
    AND app_info.id IS NULL
    AND event_name = 'explore_module_tab_click'
  GROUP BY dt, banner_type
)

-- 汇总结果
SELECT 
  '导航栏' AS click_type,
  n.feature AS click_target,
  SUM(n.click_cnt) AS total_clicks,
  SUM(n.click_uv) AS total_click_uv,
  SUM(d.dau) AS total_dau,
  ROUND(SUM(n.click_cnt) * 100.0 / SUM(d.dau), 2) AS ctr_by_click,
  ROUND(SUM(n.click_uv) * 100.0 / SUM(d.dau), 2) AS ctr_by_uv
FROM nav_click n
JOIN dau_base d ON n.dt = d.dt
GROUP BY click_type, click_target

UNION ALL

SELECT 
  'Banner' AS click_type,
  b.banner_type AS click_target,
  SUM(b.click_cnt) AS total_clicks,
  SUM(b.click_uv) AS total_click_uv,
  SUM(d.dau) AS total_dau,
  ROUND(SUM(b.click_cnt) * 100.0 / SUM(d.dau), 2) AS ctr_by_click,
  ROUND(SUM(b.click_uv) * 100.0 / SUM(d.dau), 2) AS ctr_by_uv
FROM banner_click b
JOIN dau_base d ON b.dt = d.dt
GROUP BY click_type, click_target

ORDER BY click_type, total_clicks DESC;
