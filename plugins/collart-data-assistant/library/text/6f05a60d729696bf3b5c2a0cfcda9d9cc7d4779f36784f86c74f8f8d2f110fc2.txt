-- Collart Android 搜索关键词分布分析
-- 事件：action_search_template_start
-- 参数：text（搜索词）
-- 时间范围：过去 7 天

WITH search_events AS (
  SELECT
    PARSE_DATE('%Y%m%d', event_date) AS stats_date,
    user_pseudo_id,
    (SELECT value.string_value FROM UNNEST(event_params) WHERE key = 'text') AS search_text
  FROM `storytemplate-10a27.analytics_232977577.events_*`
  WHERE _TABLE_SUFFIX BETWEEN FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 7 DAY))
    AND FORMAT_DATE('%Y%m%d', DATE_SUB(CURRENT_DATE(), INTERVAL 1 DAY))
    AND app_info.id = 'free.ai.photo.generator.collart.ai'
    AND event_name = 'action_search_template_start'
)

SELECT
  search_text,
  COUNT(*) AS search_count,
  COUNT(DISTINCT user_pseudo_id) AS user_count,
  ROUND(COUNT(*) * 1.0 / COUNT(DISTINCT user_pseudo_id), 2) AS avg_search_per_user
FROM search_events
WHERE search_text IS NOT NULL
GROUP BY search_text
ORDER BY search_count DESC
LIMIT 1000
