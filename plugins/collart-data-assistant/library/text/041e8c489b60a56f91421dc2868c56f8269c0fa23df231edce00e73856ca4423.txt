/*
  Collart Fashion 用户×日事件宽表
  Dataform: definitions/ads/collart/collart_fashion/ads_oper_user_event_metric_di.sqlx
  列由 aidata2025.ads_collart.ads_dim_metric_rule is_materialized=true 驱动
  （本文件由 _scratch/gen_event_metric_sqlx_fashion.py 产出，勿手改）
*/

DECLARE from_date DATE DEFAULT ${dataform.projectConfig.vars.biz_date} - 30;
DECLARE to_date   DATE DEFAULT ${dataform.projectConfig.vars.biz_date};

CREATE TABLE IF NOT EXISTS `aidata2025.ads_collartfashion.ads_oper_user_event_metric_di`
(
  package_name STRING,
  app_name STRING,
  event_date DATE,
  user_pseudo_id STRING,
  welcome_show_pv INT64,
  page_view_pv INT64,
  inspiration_template_show_pv INT64,
  generator_show_pv INT64,
  video_generate_click_pv INT64,
  generate_start_pv INT64,
  generate_success_pv INT64,
  ai_execute_start_pv INT64,
  ai_execute_success_pv INT64,
  fashion_ai_service_start_pv INT64,
  fashion_ai_service_success_pv INT64,
  vip_subscribe_succeed_pv INT64
)
PARTITION BY event_date
CLUSTER BY user_pseudo_id
OPTIONS (description = 'Collart Fashion 用户×日事件宽表：ads_dim_metric_rule 驱动');

BEGIN TRANSACTION;
DELETE FROM `aidata2025.ads_collartfashion.ads_oper_user_event_metric_di`
WHERE event_date BETWEEN from_date AND to_date;
INSERT INTO `aidata2025.ads_collartfashion.ads_oper_user_event_metric_di`
(
  package_name,
  app_name,
  event_date,
  user_pseudo_id,
  welcome_show_pv,
  page_view_pv,
  inspiration_template_show_pv,
  generator_show_pv,
  video_generate_click_pv,
  generate_start_pv,
  generate_success_pv,
  ai_execute_start_pv,
  ai_execute_success_pv,
  fashion_ai_service_start_pv,
  fashion_ai_service_success_pv,
  vip_subscribe_succeed_pv
)
WITH act AS (
  SELECT event_date, user_pseudo_id
  FROM `aidata2025.ads_collartfashion.ads_oper_user_active_di`
  WHERE event_date BETWEEN from_date AND to_date
    AND user_pseudo_id IS NOT NULL
    AND is_active
),
evt AS (
  SELECT
    e.event_date,
    e.user_pseudo_id,
    e.event_name,
    e.web_params_info.task_id AS task_id,
    LOWER(e.web_params_info.service_type) AS service_type
  FROM `aidata2025.dm.dm_collart_web_user_event_di` AS e
  INNER JOIN act USING (event_date, user_pseudo_id)
  WHERE e.event_date BETWEEN from_date AND to_date
    AND e.package_name = 'collart_fashion'
    AND (e.event_name IN ('ai_service_execute_start', 'ai_service_execute_success', 'ai_service_start', 'ai_service_success', 'fashion_generator_show', 'fashion_inspiration_template_show', 'fashion_video_generate_click', 'fashion_welcome_page_show', 'page_view', 'vip_subscribe_succeed'))
)
SELECT
  'collart_fashion' AS package_name,
  'collart_fashion' AS app_name,
  event_date,
  user_pseudo_id,
  COUNTIF(event_name = 'fashion_welcome_page_show') AS welcome_show_pv,  -- 欢迎页曝光
  COUNTIF(event_name = 'page_view') AS page_view_pv,  -- 页面浏览
  COUNTIF(event_name = 'fashion_inspiration_template_show') AS inspiration_template_show_pv,  -- 灵感模板曝光
  COUNTIF(event_name = 'fashion_generator_show') AS generator_show_pv,  -- 生成器曝光
  COUNTIF(event_name = 'fashion_video_generate_click') AS video_generate_click_pv,  -- 视频生成点击
  COUNTIF(event_name = 'ai_service_start') AS generate_start_pv,  -- AI 服务开始
  COUNTIF(event_name = 'ai_service_success') AS generate_success_pv,  -- AI 服务成功
  COUNTIF(event_name = 'ai_service_execute_start') AS ai_execute_start_pv,  -- AI 执行开始
  COUNTIF(event_name = 'ai_service_execute_success') AS ai_execute_success_pv,  -- AI 执行成功
  COUNTIF(event_name = 'ai_service_start' AND service_type = 'fashion_video') AS fashion_ai_service_start_pv,  -- Fashion 视频服务开始
  COUNTIF(event_name = 'ai_service_success' AND service_type = 'fashion_video') AS fashion_ai_service_success_pv,  -- Fashion 视频服务成功
  COUNTIF(event_name = 'vip_subscribe_succeed') AS vip_subscribe_succeed_pv  -- VIP 订阅成功
FROM evt
GROUP BY event_date, user_pseudo_id;
COMMIT TRANSACTION;
