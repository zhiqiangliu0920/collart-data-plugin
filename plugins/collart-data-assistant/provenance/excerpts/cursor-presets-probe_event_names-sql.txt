-- 改：表名 / _TABLE_SUFFIX / app_info 过滤 / LIMIT
SELECT
  event_name,
  COUNT(*) AS pv,
  COUNT(DISTINCT user_pseudo_id) AS uv
FROM `storytemplate-10a27.analytics_232977577.events_*`
WHERE _TABLE_SUFFIX BETWEEN '20260801' AND '20260807'
  AND app_info.id = 'free.ai.photo.generator.collart.ai'  -- Android
  -- Web/Fashion: AND app_info.id IS NULL
  -- iOS: 改用 `vidart-8b8ca.analytics_528583115.events_*` + bundle 过滤
GROUP BY event_name
ORDER BY pv DESC
LIMIT 50;
