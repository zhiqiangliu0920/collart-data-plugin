-- Web 新用户 cohort 明细骨架
-- 新用户名单优先用 ADS profile；需要完整历史（含从未进 active 的用户）时回落旧 new 表
-- 记得排除内部用户（见 internal_user_filter.sql）
WITH new_users AS (
  SELECT user_pseudo_id, user_id, first_open_date AS cohort_date, country, traffic_src_type
  FROM `aidata2025.ads_collartweb.ads_oper_user_profile_df`
  WHERE first_open_date BETWEEN DATE_SUB(CURRENT_DATE(), INTERVAL 30 DAY) AND CURRENT_DATE()
),
acts AS (
  SELECT user_pseudo_id, event_date, is_active
  FROM `aidata2025.ads_collartweb.ads_oper_user_active_di`
  WHERE event_date BETWEEN DATE_SUB(CURRENT_DATE(), INTERVAL 37 DAY) AND CURRENT_DATE()
)
SELECT
  n.cohort_date,
  n.country,
  n.traffic_src_type,
  COUNT(DISTINCT n.user_pseudo_id) AS new_users,
  COUNT(DISTINCT IF(a.event_date = DATE_ADD(n.cohort_date, INTERVAL 1 DAY) AND a.is_active, a.user_pseudo_id, NULL)) AS d2_retained
FROM new_users n
LEFT JOIN acts a USING (user_pseudo_id)
GROUP BY ALL
ORDER BY n.cohort_date, new_users DESC;
